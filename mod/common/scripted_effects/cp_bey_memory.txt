# --- Refresh SoL from pop cohort -------------------------------------------
cp_bey_refresh_sol = {
	cp_refresh_sol_from_pop = {
		person = bey
		culture = misri
		religion = sunni
		poptype = aristocrats
	}
}


# --- Dispatcher branches ----------------------------------------------------
cp_bey_dispatch_startup = {
	cp_person_hook_noop = yes
}

cp_bey_dispatch_monthly = {
	cp_person_hook_noop = yes
}

cp_bey_dispatch_yearly = {
	cp_age_increment = { person = bey }
	cp_bey_refresh_sol = yes
	cp_person_yearly_lifecycle = { person = bey country = EGY }
}

cp_bey_dispatch_law_enacted = {
	if = {
		limit = {
			ROOT.currently_enacting_law.type ?= law_type:law_commercialized_agriculture
			NOT = { has_variable = cp_bey_seen_commercialized }
		}
		cp_try_fire_law_reaction = { event = cp_bey.22 person = bey }
	}
}

cp_bey_dispatch_war_started = {
	cp_person_hook_noop = yes
}

cp_bey_dispatch_war_end = {
	cp_person_hook_noop = yes
}

cp_bey_dispatch_tech = {
	if = {
		limit = {
			cp_person_home_has_manufacturing = { person = bey }
			NOT = { has_variable = cp_bey_seen_factory_boundary }
		}
		cp_try_fire_world_response = { event = cp_bey.30 person = bey }
	}
}

cp_bey_dispatch_building = {
	if = {
		limit = {
			is_building_group = bg_manufacturing
			cp_building_affects_person_workplace = { person = bey }
			owner ?= {
				THIS = c:EGY
				cp_person_is_alive = { person = bey }
				NOT = { has_variable = cp_bey_seen_factory_boundary }
			}
		}
		owner ?= {
			cp_try_fire_world_response = { event = cp_bey.30 person = bey }
		}
	}
}

cp_bey_dispatch_revolution = {
	cp_person_hook_noop = yes
}
