# character. He enters conditionally when manufacturing and urban pull turn

# --- Refresh SoL from pop cohort -------------------------------------------
cp_mansur_refresh_sol = {
	cp_refresh_sol_from_pop = {
		person = mansur
		culture = misri
		religion = sunni
		poptype = laborers
	}
}


# --- Conditional entry ------------------------------------------------------
cp_mansur_try_spawn = {
	if = {
		limit = {
			c:EGY ?= this
			NOT = { has_global_variable = cp_person_mansur_alive }
			NOT = { has_variable = cp_mansur_alive }
			cp_country_has_factory_migration = yes
		}
		random_list = {
			42 = {
				cp_mark_person_spawn_home_from_manufacturing = { person = mansur }
				cp_register_person = {
					person = mansur
					country = EGY
				}
				cp_mansur_dispatch_startup = yes
				trigger_event = { id = cp_mansur.1 }
			}
			58 = { }
		}
	}
}

			cp_person_home_has_manufacturing = { person = mansur }

		}
		random_list = {
			35 = { cp_try_fire_world_response = { event = cp_mansur.20 person = mansur } }
			65 = { }
		}
	}
	if = {
		limit = {
			OR = {
				has_technology_researched = mechanical_tools
				has_technology_researched = rotary_valve_engine
				has_technology_researched = combustion_engine
			}
			NOT = { has_variable = cp_mansur_seen_belt }
		}
		cp_try_fire_world_response = { event = cp_mansur.30 person = mansur }
	}
}
