cp_compact_active_slot = {
	if = {
		limit = {
			NOT = { cp_active_roster_has_slot = { slot = $to$ } }
			has_variable = cp_$person$_active_slot
			var:cp_$person$_active_slot = $from$
		}
		set_variable = { name = cp_$person$_active_slot value = $to$ }
	}
}

cp_recover_active_roster_person = {
	if = {
		limit = {
			cp_person_is_alive = { person = $person$ }
			cp_$person$_has_visible_history = yes
			NOT = { has_variable = cp_$person$_active_slot }
		}
		if = {
			limit = { NOT = { cp_active_roster_has_slot = { slot = 1 } } }
			set_variable = { name = cp_$person$_active_slot value = 1 }
		}
		else_if = {
			limit = { NOT = { cp_active_roster_has_slot = { slot = 2 } } }
			set_variable = { name = cp_$person$_active_slot value = 2 }
		}
		else_if = {
			limit = { NOT = { cp_active_roster_has_slot = { slot = 3 } } }
			set_variable = { name = cp_$person$_active_slot value = 3 }
		}
	}
}

	cp_clear_invalid_active_people = yes
	cp_clear_dead_active_people = yes
	cp_compact_active_roster = yes
	cp_recover_active_roster_from_seen_people = yes
}

cp_spotlight_person = {
	if = {
		limit = { cp_person_is_alive = { person = $person$ } }
		cp_refresh_active_roster = yes
		if = {
			limit = {
				has_variable = cp_$person$_active_slot
				var:cp_$person$_active_slot = 1
			}
			set_variable = { name = cp_$person$_active_slot value = 1 }
		}
		else = {
			remove_variable = cp_$person$_active_slot
			cp_clear_active_slot_three = yes
			cp_shift_active_slot_two_to_three = yes
			cp_shift_active_slot_one_to_two = yes
			set_variable = { name = cp_$person$_active_slot value = 1 }
		}
	}
}
