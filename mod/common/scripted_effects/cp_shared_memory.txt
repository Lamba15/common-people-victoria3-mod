###############################################################################
# Common People -- shared person-parameterised primitives (r5)
#
# Every primitive here takes `person = <name>` and writes to
# cp_$person$_<attr>. Callsites look like:
#
#   cp_shift_hope = { person = layla delta = 2 }
#
# Complex effects that read the person's pop cohort (SoL drift, literacy
# drift, radicalism drift) also take culture/religion/poptype because V3
# cannot dereference cu:$var$ at runtime — these must be literals at the
# macro-substitution point. Each person gets a thin wrapper in their own
# cp_<person>_memory.txt that bakes in their culture/religion.
#
# See documentation/characters/_shared/registry.md for the full spec.
###############################################################################


# ============================================================================
# cp_person_hook_noop -- intentionally empty person hook placeholder
# ============================================================================
# Used by scaffolded per-person dispatchers so every registered person exposes
# the same hook surface even before they have content for every hook.
cp_person_hook_noop = {
	random_list = {
		1 = { }
	}
}

	change_variable = { name = cp_$person$_hope add = $delta$ }
	clamp_variable  = { name = cp_$person$_hope min = 0 max = 10 }

# Radicalism / loyalism are mutually exclusive on V3 pops; we mirror that
# by zeroing the other whenever we bump one.
cp_shift_radical = {
	change_variable = { name = cp_$person$_radical add = $delta$ }
	clamp_variable  = { name = cp_$person$_radical min = 0 max = 100 }
	if = {
		limit = { var:cp_$person$_radical > 0 }
		set_variable = { name = cp_$person$_loyalist value = 0 }
	}
}

cp_shift_loyalist = {
	change_variable = { name = cp_$person$_loyalist add = $delta$ }
	clamp_variable  = { name = cp_$person$_loyalist min = 0 max = 100 }
	if = {
		limit = { var:cp_$person$_loyalist > 0 }
		set_variable = { name = cp_$person$_radical value = 0 }
	}
}

# ============================================================================
# Shared non-Layla lifecycle
# ============================================================================
# Layla has a bespoke mortality curve because she has authored death and
# post-death events. Other persons use this common pressure dial so they do not
# live forever in long campaigns. Active JE people are protected from silent
# death; inactive people can leave the registry once age, low SoL, exhaustion,
# and recent hardship make that plausible.

cp_refresh_person_mortality_pressure = {
	set_variable = { name = cp_$person$_mortality_pressure value = 0 }
	if = {
		limit = { NOT = { has_variable = cp_$person$_recent_hardship } }
		set_variable = { name = cp_$person$_recent_hardship value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = cp_$person$_recent_joy } }
		set_variable = { name = cp_$person$_recent_joy value = 0 }
	}

	if = {
		limit = { var:cp_$person$_age >= 80 }
		change_variable = { name = cp_$person$_mortality_pressure add = 7 }
	}
	else_if = {
		limit = { var:cp_$person$_age >= 70 }
		change_variable = { name = cp_$person$_mortality_pressure add = 5 }
	}
	else_if = {
		limit = { var:cp_$person$_age >= 60 }
		change_variable = { name = cp_$person$_mortality_pressure add = 3 }
	}
	else_if = {
		limit = { var:cp_$person$_age >= 50 }
		change_variable = { name = cp_$person$_mortality_pressure add = 1 }
	}

	if = {
		limit = { var:cp_$person$_sol <= 4 }
		change_variable = { name = cp_$person$_mortality_pressure add = 3 }
	}
	else_if = {
		limit = { var:cp_$person$_sol <= 9 }
		change_variable = { name = cp_$person$_mortality_pressure add = 1 }
	}

	if = {
		limit = { var:cp_$person$_exhaustion >= 8 }
		change_variable = { name = cp_$person$_mortality_pressure add = 2 }
	}
	else_if = {
		limit = { var:cp_$person$_exhaustion >= 5 }
		change_variable = { name = cp_$person$_mortality_pressure add = 1 }
	}

	if = {
		limit = { var:cp_$person$_recent_hardship > 0 }
		change_variable = { name = cp_$person$_mortality_pressure add = 1 }
	}
	if = {
		limit = { var:cp_$person$_recent_joy > 0 }
		change_variable = { name = cp_$person$_mortality_pressure add = -1 }
	}

	if = {
		limit = { has_law = law_type:law_public_health_insurance }
		change_variable = { name = cp_$person$_mortality_pressure add = -2 }
	}
	else_if = {
		limit = {
			has_law = law_type:law_private_health_insurance
			var:cp_$person$_sol >= 15
		}
		change_variable = { name = cp_$person$_mortality_pressure add = -1 }
	}
	else_if = {
		limit = {
			has_law = law_type:law_charitable_health_system
			var:cp_$person$_sol <= 10
		}
		change_variable = { name = cp_$person$_mortality_pressure add = -1 }
	}

	if = {
		limit = {
			has_law = law_type:law_poor_laws
			var:cp_$person$_sol <= 10
		}
		change_variable = { name = cp_$person$_mortality_pressure add = -1 }
	}
	if = {
		limit = {
			has_law = law_type:law_old_age_pension
			var:cp_$person$_age >= 60
		}
		change_variable = { name = cp_$person$_mortality_pressure add = -1 }
	}

	clamp_variable = { name = cp_$person$_mortality_pressure min = 0 max = 10 }
}

cp_person_yearly_lifecycle = {
	if = {
		limit = {
			cp_person_is_alive = { person = $person$ }
			NOT = { has_variable = cp_$person$_active_slot }
		}
		cp_refresh_person_mortality_pressure = { person = $person$ }
		if = {
			limit = { var:cp_$person$_mortality_pressure >= 8 }
			random_list = {
				12 = { cp_person_dies = { person = $person$ country = $country$ } }
				88 = { }
			}
		}
		else_if = {
			limit = { var:cp_$person$_mortality_pressure >= 5 }
			random_list = {
				6 = { cp_person_dies = { person = $person$ country = $country$ } }
				94 = { }
			}
		}
		else_if = {
			limit = { var:cp_$person$_mortality_pressure >= 3 }
			random_list = {
				3 = { cp_person_dies = { person = $person$ country = $country$ } }
				97 = { }
			}
		}
	}
}


# ============================================================================
# Home-state assignment
# ============================================================================
# A person has exactly one home state marker: cp_<person>_lives_here. Local
# building events and pop-cohort reads use that marker, so person setup must go
# through this shared effect instead of hand-writing per-person state blocks.

cp_clear_person_home_state = {
	every_country = {
		every_scope_state = {
			limit = { has_variable = cp_$person$_lives_here }
			remove_variable = cp_$person$_lives_here
		}
	}
}

cp_set_person_home_state = {
	cp_clear_person_home_state = { person = $person$ }
	s:$state$ ?= {
		if = {
			limit = {
				any_scope_state = { owner = c:$country$ }
			}
			random_scope_state = {
				limit = { owner = c:$country$ }
				set_variable = cp_$person$_lives_here
				root = {
					set_variable = { name = cp_$person$_state_pointer value = prev }
				}
			}
		}
		else = {
			random_scope_state = {
				set_variable = cp_$person$_lives_here
				root = {
					set_variable = { name = cp_$person$_state_pointer value = prev }
				}
			}
		}
	}
}

	cp_clear_person_home_state = { person = $person$ }
	if = {
		limit = {
			any_scope_state = { has_variable = cp_$person$_spawn_home_candidate }
		}
		random_scope_state = {
			limit = { has_variable = cp_$person$_spawn_home_candidate }
			set_variable = cp_$person$_lives_here

				set_variable = { name = cp_$person$_state_pointer value = prev }

	}

# ============================================================================
# Pop-cohort reads (require culture + religion + poptype literals)
# ============================================================================
# Reads the largest pop in the person's home state that matches their
# culture + religion + profession. Writes the cohort's SoL into
# cp_$person$_sol.
#
# Callsite must supply person + culture + religion + poptype as LITERALS
# (not variables) because V3 macro-substitution is load-time. See
# cp_layla_memory.txt::cp_layla_refresh_sol for the Layla wrapper.

cp_refresh_sol_from_pop = {
	every_scope_state = {
		limit = { has_variable = cp_$person$_lives_here }
		ordered_scope_pop = {
			limit = {
				culture     = cu:$culture$
				religion    = rel:$religion$
				is_pop_type = $poptype$
			}
			order_by = workforce
			position = 0
			save_temporary_scope_value_as = {
				name = tmp_cohort_sol
				value = standard_of_living
			}
		}
	}
	if = {
		limit = { exists = scope:tmp_cohort_sol }
		set_variable = { name = cp_$person$_sol value = scope:tmp_cohort_sol }
		clamp_variable = { name = cp_$person$_sol min = 1 max = 99 }
	}
}
