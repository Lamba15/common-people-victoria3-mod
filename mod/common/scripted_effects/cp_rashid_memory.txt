# Common People -- Rashid al-Katib's memory
#
# Person 11 scaffold. Rashid is registered as `rashid`, not as a V3
# character. He enters conditionally when government administration, central
# archives, or identification papers make the state visible at the counter.
###############################################################################


# --- Refresh SoL from pop cohort -------------------------------------------
cp_rashid_refresh_sol = {
	cp_refresh_sol_from_pop = {
		person = rashid
		culture = misri
		religion = sunni
		poptype = bureaucrats
	}
}


# --- Conditional entry ------------------------------------------------------
cp_rashid_try_spawn = {
	if = {
		limit = {
			c:EGY ?= this
			NOT = { has_global_variable = cp_person_rashid_alive }
			NOT = { has_variable = cp_rashid_alive }
			OR = {
				cp_country_has_government_office = yes
				cp_country_has_modern_records = yes
				has_technology_researched = tech_bureaucracy
			}
		}
		cp_mark_person_spawn_home_from_government_office = { person = rashid }
		cp_register_person = {
			person = rashid
			country = EGY
		}
		cp_rashid_dispatch_startup = yes
		trigger_event = { id = cp_rashid.1 }
	}
}


# --- Dispatcher branches ----------------------------------------------------
cp_rashid_dispatch_startup = {
	cp_person_hook_noop = yes
}

cp_rashid_dispatch_monthly = {
	cp_person_hook_noop = yes
}

cp_rashid_dispatch_yearly = {
	cp_age_increment = { person = rashid }
	cp_rashid_refresh_sol = yes
	cp_person_yearly_lifecycle = { person = rashid country = EGY }
	if = {
		limit = {
			has_technology_researched = central_archives
			NOT = { has_variable = cp_rashid_seen_archives }
		}
		random_list = {
			35 = { cp_try_fire_world_response = { event = cp_rashid.20 person = rashid } }
			65 = { }
		}
	}
}

cp_rashid_dispatch_law_enacted = {
	if = {
		limit = {
			OR = {
				ROOT.currently_enacting_law.type ?= law_type:law_hereditary_bureaucrats
				ROOT.currently_enacting_law.type ?= law_type:law_appointed_bureaucrats
				ROOT.currently_enacting_law.type ?= law_type:law_elected_bureaucrats
			}
			NOT = { has_variable = cp_rashid_seen_new_seal }
		}
		cp_try_fire_law_reaction = { event = cp_rashid.50 person = rashid }
	}
}

cp_rashid_dispatch_tech = {
	if = {
		limit = {
			has_technology_researched = central_archives
			NOT = { has_variable = cp_rashid_seen_archives }
		}
		cp_try_fire_world_response = { event = cp_rashid.20 person = rashid }
	}
	if = {
		limit = {

			NOT = { has_variable = cp_rashid_seen_identification }
		}

	}
}

cp_rashid_dispatch_building = {
	if = {
		limit = {
			is_building_type = building_government_administration
			cp_building_affects_person_workplace = { person = rashid }
			owner ?= {
				THIS = c:EGY
				cp_person_is_alive = { person = rashid }
				NOT = { has_variable = cp_rashid_seen_intro }
			}
		}
		owner ?= {
			random_list = {
				60 = { cp_try_fire_world_response = { event = cp_rashid.10 person = rashid } }
				40 = { }
			}
		}
	}
	if = {
		limit = {
			is_building_type = building_government_administration
			cp_building_affects_person_workplace = { person = rashid }
			owner ?= {
				THIS = c:EGY
				cp_person_is_alive = { person = rashid }
				has_variable = cp_rashid_seen_intro
				NOT = { has_variable = cp_rashid_seen_numbered_drawer }
			}
		}
		owner ?= {
			cp_try_fire_world_response = { event = cp_rashid.60 person = rashid }
		}
	}
}
