cp_roll_war_start_response = {
	random_list = {
		55 = {
			trigger = {
				cp_person_is_alive = { person = layla }
				has_variable = cp_layla_ahmed_alive
				var:cp_layla_ahmed_alive > 0
				NOT = { has_variable = cp_layla_ahmed_at_war }
				NOT = { has_variable = cp_layla_seen_ahmed_conscripted }
				var:cp_layla_ahmed_age >= 18
				var:cp_layla_ahmed_age <= 45
			}
			set_variable = { name = cp_shared_war_start_target_layla days = 30 }
		}
		45 = {
			trigger = {
				cp_person_is_alive = { person = soldier }
				NOT = { has_variable = cp_soldier_at_war }
				NOT = { has_variable = cp_soldier_seen_notice }
				var:cp_soldier_age >= 18
				var:cp_soldier_age <= 45
			}
			set_variable = { name = cp_shared_war_start_target_soldier days = 30 }
		}
		15 = { }
	}
}
