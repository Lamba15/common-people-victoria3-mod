cp_mina_dispatch_tech = {
	if = {
		limit = {
			cp_country_has_modern_records = yes
			NOT = { has_variable = cp_mina_seen_notice_sheet }
		}
		cp_try_fire_world_response = { event = cp_mina.30 person = mina }
	}
}

cp_mina_dispatch_building = {
	if = {
		limit = {
			OR = {
				is_building_type = building_university
				is_building_type = building_government_administration
			}
			cp_building_affects_person_workplace = { person = mina }
			owner ?= {
				THIS = c:EGY
				cp_person_is_alive = { person = mina }
				NOT = { has_variable = cp_mina_seen_catalogue_room }
			}
		}
		owner ?= {
			cp_try_fire_world_response = { event = cp_mina.50 person = mina }
		}
	}
}
