###############################################################################
# Common People -- shared firing gatekeepers (r5)
#
# Every event fire in the mod flows through one of five gatekeepers, picked
# by the event's tier:
#
#   AMBIENT        — small life beats, conversations, seasonal pulses.
#                    Gated by a 183-day global cooldown and a 365-day
#                    per-person cooldown. Target: 1–2 fires per in-game year
#                    across ALL persons combined, with a hard cap of 2.
#
#   LAW-REACTION   — events tied to a specific law enactment. Uses the same
#                    visible-story cooldown as ambient/startup beats, so law
#                    reforms cannot flood the player with Common People popups.
#                    Already one-shot via the event's own seen flag.
#
#   WORLD-RESPONSE — routine automatic reactions to technology, buildings,
#                    conditional person entry, and other state changes. Uses
#                    the same visible-story cooldown as ambient/startup/law
#                    beats so modernization does not create popup bursts.
#
#   MILESTONE      — critical life beats: death, Cairo offer, suitor, first
#                    welfare envelope, revolution. Bypasses cooldowns; fires
#                    on event-specific probability rolls in the dispatcher.
#
#   BUTTON         — player-initiated via a person journal/check button or QA
#                    button. Always fires. Per-event seen-flags still apply
#                    (no repeats of the same beat within its cooldown).
#
# See documentation/characters/_shared/firing.md for the full protocol.
###############################################################################


# ============================================================================
# cp_try_fire_ambient
# ============================================================================
# Fire an ambient event if both the global and per-person cooldowns are clear.
# Sets both cooldown flags after a successful fire so subsequent ambient
# attempts no-op until the cooldowns expire.
#
# Parameters:
#   event   = event id (e.g. cp_layla.13)
#   person  = name token (e.g. layla)
# ============================================================================

cp_try_fire_ambient = {
	if = {
		limit = {
			cp_person_is_alive = { person = $person$ }
			NOT = { has_global_variable = cp_global_ambient_cooldown }
			NOT = { has_global_variable = cp_person_$person$_ambient_cooldown }
		}
		cp_spotlight_person = { person = $person$ }
		trigger_event = { id = $event$ }
		set_global_variable = {
			name = cp_global_ambient_cooldown
			days = 183
		}
		set_global_variable = {
			name = cp_person_$person$_ambient_cooldown
			days = 365
		}
	}
}

# ============================================================================
# cp_roll_revolution_response
# ============================================================================
# Shared resolver for revolution-start reactions. Revolution can plausibly
# belong to several lives at once, but the player should not get a stack of
# Common People windows from a single rupture. This picks one eligible person
# by setting a short-lived target flag; person dispatchers still own the
# actual event fire and state mutation.
# ============================================================================

cp_roll_revolution_response = {
	random_list = {
		55 = {
			trigger = {
				cp_person_is_alive = { person = layla }
				NOT = { has_variable = cp_layla_seen_revolution }
			}
			set_variable = { name = cp_shared_revolution_target_layla days = 30 }
		}
		45 = {
			trigger = {
				cp_person_is_alive = { person = nabil }
				NOT = { has_variable = cp_nabil_just_spawned }
				has_variable = cp_nabil_seen_intro
				NOT = { has_variable = cp_nabil_seen_square_afterward }
			}
			set_variable = { name = cp_shared_revolution_target_nabil days = 30 }
		}
		25 = { }
	}
}

# ============================================================================
# cp_try_fire_milestone
# ============================================================================
# Fire a milestone event (death, major crossroads, first welfare). Bypasses
# cooldowns. The dispatcher decides probability via random_list; by the time
# this gatekeeper is called the roll has succeeded.
#
# Parameters:
#   event   = event id
#   person  = name token
# ============================================================================

cp_try_fire_milestone = {
	if = {
		limit = {
			cp_person_is_alive = { person = $person$ }
		}
		cp_spotlight_person = { person = $person$ }
		trigger_event = { id = $event$ }
	}
}


# ============================================================================
# cp_button_fire
# ============================================================================
# Fire an event in response to a player click on a scripted button. Always
# fires — a player click is a promise to show them something. Per-event
# seen-flags still gate repeats; that's handled by the button's own random_list
# trigger clauses.
#
# Parameters:
#   event   = event id
#   person  = name token
# ============================================================================

cp_button_fire = {
	if = {
		limit = {
			cp_person_is_alive = { person = $person$ }
		}
		cp_spotlight_person = { person = $person$ }
		trigger_event = { id = $event$ popup = yes }
	}
}

# automatic popups at 1–2 per in-game year while letting the roster feel alive.
#
