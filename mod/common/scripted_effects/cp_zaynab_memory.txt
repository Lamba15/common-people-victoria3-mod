# --- Dispatcher branches ----------------------------------------------------

	cp_person_hook_noop = yes
}

	cp_person_hook_noop = yes
}

cp_zaynab_dispatch_yearly = {
	cp_age_increment = { person = zaynab }
	cp_zaynab_refresh_sol = yes
	cp_person_yearly_lifecycle = { person = zaynab country = EGY }
	if = {
		limit = {
			NOT = { cp_country_public_health = yes }
			NOT = { has_variable = cp_zaynab_seen_fever_bowl }
		}
		cp_try_fire_world_response = { event = cp_zaynab.30 person = zaynab }
	}
}

cp_zaynab_dispatch_tech = {
	if = {
		limit = {
			OR = {
				has_technology_researched = medical_degrees
				has_technology_researched = pharmaceuticals
				has_technology_researched = modern_sewerage
			}
			NOT = { has_variable = cp_zaynab_seen_lime_bucket }
		}
		cp_try_fire_world_response = { event = cp_zaynab.50 person = zaynab }
	}
}

cp_zaynab_dispatch_building = {
	if = {
		limit = {
			OR = {
				is_building_type = building_government_administration
				is_building_type = building_university
			}
			cp_building_affects_person_workplace = { person = zaynab }
			owner ?= {
				THIS = c:EGY
				cp_person_is_alive = { person = zaynab }
				NOT = { has_variable = cp_zaynab_seen_lime_bucket }
			}
		}
		owner ?= {
			cp_try_fire_world_response = { event = cp_zaynab.50 person = zaynab }
		}
	}
}
