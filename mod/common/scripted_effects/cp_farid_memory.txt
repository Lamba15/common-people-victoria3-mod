###############################################################################
# Common People -- Farid al-Haddad's memory
#
# Person 9 scaffold. Farid is registered as `farid`, not as a V3 character.
# He enters conditionally when railways or telegraphy make an ordinary
# station worker's life mechanically legible.
###############################################################################


# --- Refresh SoL from pop cohort -------------------------------------------
cp_farid_refresh_sol = {
	cp_refresh_sol_from_pop = {
		person = farid
		culture = misri
		religion = sunni
		poptype = laborers
	}
}


# --- Conditional entry ------------------------------------------------------
cp_farid_try_spawn = {
	if = {
		limit = {
			c:EGY ?= this
			NOT = { has_global_variable = cp_person_farid_alive }
			NOT = { has_variable = cp_farid_alive }
			OR = {
				has_technology_researched = railways
				has_technology_researched = electric_telegraph
				cp_country_has_rail_infrastructure = yes
			}
		}
		cp_mark_person_spawn_home_from_rail_infrastructure = { person = farid }
		cp_register_person = {
			person = farid
			country = EGY
		}
		cp_farid_dispatch_startup = yes
		trigger_event = { id = cp_farid.1 }
	}
}


# --- Dispatcher branches ----------------------------------------------------
cp_farid_dispatch_startup = {
	cp_person_hook_noop = yes
}

cp_farid_dispatch_monthly = {
	cp_person_hook_noop = yes
}

cp_farid_dispatch_yearly = {
	cp_age_increment = { person = farid }
	cp_farid_refresh_sol = yes
	cp_person_yearly_lifecycle = { person = farid country = EGY }
	if = {
		limit = {
			has_technology_researched = electric_telegraph
			NOT = { has_variable = cp_farid_seen_telegraph }
		}
		random_list = {
			35 = { cp_try_fire_world_response = { event = cp_farid.20 person = farid } }
			65 = { }
		}
	}
	if = {
		limit = {
			OR = {
				has_technology_researched = steel_railway_cars
				has_technology_researched = electric_railway
			}
			NOT = { has_variable = cp_farid_seen_new_carriage }
		}
		random_list = {
			35 = { cp_try_fire_world_response = { event = cp_farid.50 person = farid } }
			65 = { }
		}
	}
}

cp_farid_dispatch_building = {
	if = {
		limit = {
			is_building_type = building_railway
			cp_building_affects_person_workplace = { person = farid }
			owner ?= {
				THIS = c:EGY
				cp_person_is_alive = { person = farid }
				NOT = { has_variable = cp_farid_seen_rail_yard }
			}
		}
		owner ?= {
			random_list = {
				70 = { cp_try_fire_world_response = { event = cp_farid.30 person = farid } }
				30 = { }
			}
		}
	}
	if = {
		limit = {
			is_building_type = building_railway
			cp_building_affects_person_workplace = { person = farid }
			owner ?= {
				THIS = c:EGY
				cp_person_is_alive = { person = farid }
				has_variable = cp_farid_seen_intro
				NOT = { has_variable = cp_farid_seen_parcel_bag }
			}
		}
		owner ?= {
			cp_try_fire_world_response = { event = cp_farid.60 person = farid }
		}
	}
}
