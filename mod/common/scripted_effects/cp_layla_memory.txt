###############################################################################
# Common People -- memory
#
# The only layer that writes to Layla's country variables. Events mutate state
# by calling effects here; they never call set_variable / change_variable
# directly inside an option block. This is the single source of truth for
# "what changes hope / SoL / class / anchor".
#
# Conventions:
#   - Every effect is named cp_<verb>_<field>.
#   - All run in country scope (root = c:EGY).
#   - Clamping uses V3's built-in clamp_variable primitive.
#   - Composite effects call primitives so the library is layered.
###############################################################################


# ============================================================================
# cp_layla_refresh_sol — Layla's pop-cohort SoL read (wraps shared primitive)
# ============================================================================
# Wraps cp_refresh_sol_from_pop (in cp_shared_memory.txt) with Layla's
# literal culture/religion/profession baked in. V3 macro substitution is
# load-time, not runtime, so each person needs a thin wrapper with literals.
# Called from cp_on_yearly via `cp_layla_refresh_sol = yes`.
cp_layla_refresh_sol = {
	cp_refresh_sol_from_pop = {
		person = layla
		culture = misri
		religion = sunni
		poptype = peasants
	}
}


# ============================================================================
# cp_layla_dispatch_startup -- Layla's startup/save-migration hook
# ============================================================================
# Called from the shared startup event before registration/setup. New games
# pass through cleanly; old saves get the Layla-owned backfills here instead
# of having shared on_actions write directly to cp_layla_* variables.
cp_layla_dispatch_startup = {
	if = {
		limit = { has_variable = cp_layla_alive  var:cp_layla_alive > 0  NOT = { has_variable = cp_layla_status_variant } }
		set_variable = { name = cp_layla_status_variant value = 1 }
	}
	if = {
		limit = { has_variable = cp_layla_alive  var:cp_layla_alive > 0  NOT = { has_variable = cp_layla_sol_prev } }
		set_variable = { name = cp_layla_sol_prev value = var:cp_layla_sol }
		set_variable = { name = cp_layla_sol_delta value = 0 }
	}
	# Saves from before the data-driven home-state refactor lack the
	# region_state marker. Backfill to Lower Egypt (her start) so the shared
	# pop-cohort reader has something real to inspect.
	if = {
		limit = {
			has_variable = cp_layla_alive
			var:cp_layla_alive > 0
			NOT = {
				any_country = {
					any_scope_state = { has_variable = cp_layla_lives_here }
				}
			}
		}
		cp_move_to_state = { state = STATE_LOWER_EGYPT }
	}
}


# ============================================================================
# cp_layla_dispatch_monthly -- Layla's JE display-state pulse
# ============================================================================
# The shared monthly on_action only guards and delegates. The actual Layla
# mood roll and JE tick live here with the rest of her person-owned state.
cp_layla_dispatch_monthly = {
	# Self-heal: backfill cp_layla_status_variant on saves that predate it.
	if = {
		limit = { NOT = { has_variable = cp_layla_status_variant } }
		set_variable = { name = cp_layla_status_variant value = 1 }
	}
	cp_debug_tick_increment = yes
	# Mood bucket (1..10 -> low/mid/high bucket)
	random_list = {
		10 = { set_variable = { name = cp_layla_mood_roll value = 1 } }
		10 = { set_variable = { name = cp_layla_mood_roll value = 2 } }
		10 = { set_variable = { name = cp_layla_mood_roll value = 3 } }
		10 = { set_variable = { name = cp_layla_mood_roll value = 4 } }
		10 = { set_variable = { name = cp_layla_mood_roll value = 5 } }
		10 = { set_variable = { name = cp_layla_mood_roll value = 6 } }
		10 = { set_variable = { name = cp_layla_mood_roll value = 7 } }
		10 = { set_variable = { name = cp_layla_mood_roll value = 8 } }
		10 = { set_variable = { name = cp_layla_mood_roll value = 9 } }
		10 = { set_variable = { name = cp_layla_mood_roll value = 10 } }
	}
	# Variant within bucket (1..3)
	random_list = {
		33 = { set_variable = { name = cp_layla_status_variant value = 1 } }
		33 = { set_variable = { name = cp_layla_status_variant value = 2 } }
		34 = { set_variable = { name = cp_layla_status_variant value = 3 } }
	}
}


# ============================================================================
# cp_layla_dispatch_yearly -- Layla's yearly lifecycle and milestone router
# ============================================================================
# Called from cp_on_yearly after the same shared alive guard used by the rest
# of the roster. This keeps Layla's heavy life logic person-owned instead of
# living as a special case in the global on_action dispatcher.
cp_layla_dispatch_yearly = {
	cp_age_increment = { person = layla }
	# Self-heal: backfill snapshot variables on older saves.
	if = {
		limit = { NOT = { has_variable = cp_layla_sol_prev } }
		set_variable = { name = cp_layla_sol_prev value = var:cp_layla_sol }
		set_variable = { name = cp_layla_sol_delta value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = cp_layla_tax_burden } }
		set_variable = { name = cp_layla_tax_burden value = 0 }
		cp_refresh_tax_burden = yes
		set_variable = { name = cp_layla_tax_burden_prev value = var:cp_layla_tax_burden }
	}
	if = {
		limit = { NOT = { has_variable = cp_layla_welfare_tier } }
		set_variable = { name = cp_layla_welfare_tier value = 0 }
		set_variable = { name = cp_layla_welfare_tier_prev value = 0 }
		cp_refresh_welfare_tier = yes
	}
	# Resample cp_layla_sol from her pop group BEFORE computing delta.
	cp_layla_refresh_sol = yes

	# Pop-tick: literacy drifts toward cohort, political mood drifts
	# from expected-SoL gap. Both mirror the SoL resample pattern.
	cp_pop_tick_literacy = yes
	cp_pop_tick_political_mood = yes
	# Delta = current - prev. Computed via multiply-by-(-1)
	# because `change_variable = { subtract = var:X }` is not
	# reliably supported; `add = var:X` is. So negate prev, then add.
	set_variable = { name = cp_layla_sol_delta value = var:cp_layla_sol_prev }
	change_variable = { name = cp_layla_sol_delta multiply = -1 }
	change_variable = { name = cp_layla_sol_delta add = var:cp_layla_sol }
	set_variable = { name = cp_layla_sol_prev value = var:cp_layla_sol }

	# Tax burden: same pattern. Recompute, then fire the tax-spike
	# event if burden jumped by ≥3 since last year (the tax
	# collector now comes for more than he did).
	cp_refresh_tax_burden = yes
	if = {
		limit = {
			NOT = { has_variable = cp_layla_seen_tax_spike }
			var:cp_layla_tax_burden >= 7
			# current - prev >= 3  via multiply-by-(-1) trick
			var:cp_layla_tax_burden >= var:cp_layla_tax_burden_prev
			# We cannot express var - var >= 3 directly; approximate
			# with absolute threshold (>=7) + the trend guard above.
		}
		cp_try_fire_milestone = { event = cp_layla.7 person = layla }
		set_variable = { name = cp_layla_seen_tax_spike days = 1800 }
	}
	# First-bite event: consumption tax on grain lands for the
	# first time since she's been alive.
	if = {
		limit = {
			NOT = { has_variable = cp_layla_seen_grain_tax }
			has_consumption_tax = g:grain
		}
		cp_try_fire_milestone = { event = cp_layla.8 person = layla }
		set_variable = cp_layla_seen_grain_tax
	}
	set_variable = { name = cp_layla_tax_burden_prev value = var:cp_layla_tax_burden }

	# First-child milestone. This is not ambient gossip: it opens
	# the whole children/daughter branch, so it bypasses the ambient
	# budget but stays probabilistic and one-shot.
	if = {
		limit = {
			NOT = { has_variable = cp_layla_seen_childbirth }
			NOT = { var:cp_layla_children >= 1 }
			has_variable = cp_layla_married
			has_variable = cp_layla_ahmed_alive
			var:cp_layla_ahmed_alive > 0
			NOT = { has_variable = cp_layla_ahmed_at_war }
			var:cp_layla_age >= 23
			var:cp_layla_age <= 32
		}
		random_list = {
			35 = { cp_try_fire_milestone = { event = cp_layla.5 person = layla } }
			65 = { }
		}
	}

	# Crossroads: Cairo offers work. Fires once per life when
	# prerequisites align — a mill exists in her state, she's
	# still a village peasant, SoL is pressed, and migration is
	# legally possible (NOT serfdom). 25% roll per qualifying
	# year so it feels like a choice arriving rather than an
	# obligation on a timer.
	if = {
		limit = {
			has_variable = cp_layla_seen_first_textile_mill
			NOT = { has_variable = cp_layla_seen_cairo_offer }
			NOT = { has_law = law_type:law_serfdom }
			has_variable = cp_layla_profession_peasants
			var:cp_layla_sol <= 12
		}
		random_list = {
			25 = {
				cp_try_fire_milestone = { event = cp_layla.70 person = layla }
				set_variable = cp_layla_seen_cairo_offer
			}
			75 = { }
		}
	}

	# Mortality pressures — computed yearly before the rolls.
	cp_refresh_mortality_pressure = yes
	cp_refresh_worker_safety_pressure = yes

	# First welfare receipt. Fires once per life when a welfare
	# law is active AND she plausibly qualifies. Each rung has
	# its own eligibility test; any match triggers the event.
	if = {
		limit = {
			NOT = { has_variable = cp_layla_seen_first_envelope }
			OR = {
				# Old-age pension and she's of age.
				AND = {
					has_law = law_type:law_old_age_pension
					var:cp_layla_age >= 60
				}
				# Poor laws and she's poor.
				AND = {
					has_law = law_type:law_poor_laws
					var:cp_layla_sol <= 10
				}
				# Wage subsidies and her household earns wages.
				AND = {
					has_law = law_type:law_wage_subsidies
					var:cp_layla_sol <= 15
					OR = {
						has_variable = cp_layla_profession_laborers
						has_variable = cp_layla_profession_machinists
						has_variable = cp_layla_profession_farmers
						has_variable = cp_layla_profession_clerks
					}
				}
			}
		}
		# 50% per qualifying year — welfare rolls are slow but
		# the first envelope feels like an arrival, not a raffle.
		random_list = {
			50 = {
				cp_try_fire_milestone = { event = cp_layla.90 person = layla }
				set_variable = cp_layla_seen_first_envelope
			}
			50 = { }
		}
	}

	# Layla's annual mortality roll. Buckets her 0..10 pressure
	# into three rough tiers with per-year death probabilities
	# calibrated to 19th-century reality (a 70yo peasant without
	# public health dies at ~10%/yr; a 40yo at peak health ~1%).
	# Age gate: no roll before 35, because events-before-then
	# should kill her (childbirth, famine, factory), not timers.
	if = {
		limit = {
			var:cp_layla_age >= 35
			var:cp_layla_mortality_pressure >= 1
		}
		if = {
			limit = { var:cp_layla_mortality_pressure >= 7 }
			random_list = {
				10 = { cp_try_fire_milestone = { event = cp_layla.80 person = layla } }
				90 = { }
			}
		}
		else_if = {
			limit = { var:cp_layla_mortality_pressure >= 4 }
			random_list = {
				4 = { cp_try_fire_milestone = { event = cp_layla.80 person = layla } }
				96 = { }
			}
		}
		else = {
			random_list = {
				1 = { cp_try_fire_milestone = { event = cp_layla.80 person = layla } }
				99 = { }
			}
		}
	}

	# Ahmed's worker-accident roll. Gated on cp_layla_worker_risk which
	# is only non-zero when he works a shop floor. Worker-
	# protections law clamps the worst away; no-workers-rights
	# (or early industrialisation under regulatory_bodies) makes
	# accidents real and annual.
	if = {
		limit = {
			has_variable = cp_layla_ahmed_alive
			var:cp_layla_ahmed_alive > 0
			var:cp_layla_worker_risk >= 3
		}
		if = {
			limit = { var:cp_layla_worker_risk >= 7 }
			random_list = {
				6 = { cp_try_fire_milestone = { event = cp_layla.81 person = layla } }
				94 = { }
			}
		}
		else = {
			random_list = {
				2 = { cp_try_fire_milestone = { event = cp_layla.81 person = layla } }
				98 = { }
			}
		}
	}

	# Child mortality roll. V3 models this as state-level, but
	# we capture one household's risk: if she has children AND
	# conditions are grim (pressure 5+), a child may die this
	# year. Lower bar than Layla's roll because children are
	# more vulnerable to what the adults can survive.
	if = {
		limit = {
			var:cp_layla_children >= 1
			var:cp_layla_mortality_pressure >= 5
		}
		random_list = {
			3 = { cp_try_fire_milestone = { event = cp_layla.82 person = layla } }
			97 = { }
		}
	}

	# Crossroads: the Small Concern. An urban household with
	# savings and literacy is offered a partnership in a small
	# workshop. Fires once per life. Eligibility covers lower-
	# to-middle strata (laborers, machinists, clerks, engineers)
	# — per V3's profession ladder, these are the pops with
	# enough income + literacy to plausibly buy capital. She
	# still needs a permissive economy law (laissez-faire or
	# interventionism). 20% roll per qualifying year.
	if = {
		limit = {
			has_variable = cp_layla_in_city
			NOT = { has_variable = cp_layla_seen_ownership_offer }
			OR = {
				has_variable = cp_layla_profession_laborers
				has_variable = cp_layla_profession_machinists
				has_variable = cp_layla_profession_clerks
				has_variable = cp_layla_profession_engineers
			}
			var:cp_layla_sol >= 20
			var:cp_layla_literacy >= 10
			OR = {
				has_law = law_type:law_laissez_faire
				has_law = law_type:law_interventionism
			}
		}
		random_list = {
			20 = {
				cp_try_fire_milestone = { event = cp_layla.71 person = layla }
				set_variable = cp_layla_seen_ownership_offer
			}
			80 = { }
		}
	}

	# Crossroads: self-employment (Slice D). Gated on cp_layla_may_work
	# (set by cp_layla.95 when law_women_in_the_workplace enacts).
	# Two paths:
	#   - FORCED: she has to. Widowhood + pressed SoL, or
	#     unemployed + pressed SoL. 50% roll per qualifying year
	#     because survival arrives faster than vocation.
	#   - CHOSEN: law active, household OK, small yearly chance.
	#     15% roll per qualifying year.
	# One-shot per life via cp_layla_seen_self_employment; flag is set
	# inside the event's immediate.
	if = {
		limit = {
			has_variable = cp_layla_may_work
			NOT = { has_variable = cp_layla_seen_self_employment }
			NOT = { has_variable = cp_layla_self_employed }
			# Forced: widow with SoL <=10, or low SoL without
			# Ahmed's profession carrying them.
			OR = {
				AND = {
					has_variable = cp_layla_ahmed_alive
					var:cp_layla_ahmed_alive = 0
					var:cp_layla_sol <= 10
				}
				AND = {
					var:cp_layla_sol <= 8
					NOT = { has_variable = cp_layla_profession_shopkeepers }
					NOT = { has_variable = cp_layla_profession_capitalists }
				}
			}
		}
		random_list = {
			50 = {
				cp_try_fire_milestone = { event = cp_layla.97 person = layla }
			}
			50 = { }
		}
	}
	else_if = {
		# Chosen path: law active, decent SoL, not already
		# self-employed or in a profession that makes this
		# redundant. Low roll so it feels optional.
		limit = {
			has_variable = cp_layla_may_work
			NOT = { has_variable = cp_layla_seen_self_employment }
			NOT = { has_variable = cp_layla_self_employed }
			NOT = { has_variable = cp_layla_profession_shopkeepers }
			NOT = { has_variable = cp_layla_profession_capitalists }
			var:cp_layla_sol >= 9
		}
		random_list = {
			15 = {
				cp_try_fire_milestone = { event = cp_layla.97 person = layla }
			}
			85 = { }
		}
	}

	# Crossroads: daughter's marriage. Fires once per life when
	# the mother is old enough that a first child born around
	# cp_layla_age 22 would now be of marrying age. 20% roll per
	# qualifying year so the crossroads arrives when it does.
	if = {
		limit = {
			var:cp_layla_age >= 38
			var:cp_layla_children >= 1
			NOT = { has_variable = cp_layla_seen_daughter_marriage }
			NOT = { has_variable = cp_layla_daughter_married }
		}
		random_list = {
			20 = {
				cp_try_fire_milestone = { event = cp_layla.72 person = layla }
				set_variable = cp_layla_seen_daughter_marriage
			}
			80 = { }
		}
	}

	# Crossroads: the mill closes. Fires when she's a city
	# laborer household AND SoL has fallen to struggling. 30%
	# roll per qualifying year — the downturn feels arriving,
	# not scheduled.
	if = {
		limit = {
			has_variable = cp_layla_in_city
			NOT = { has_variable = cp_layla_seen_mill_closed }
			OR = {
				has_variable = cp_layla_ahmed_profession_laborer
				has_variable = cp_layla_profession_laborers
			}
			var:cp_layla_sol <= 9
		}
		random_list = {
			30 = {
				cp_try_fire_milestone = { event = cp_layla.73 person = layla }
			}
			70 = { }
		}
	}

	# The District Office (cp_layla.125). Per the editor audit:
	# she makes the real trip that the fantasy audience is a
	# survival mechanism for. Fires once per life, 10% per
	# qualifying year, gated on age and at least one prior
	# law-touching experience.
	if = {
		limit = {
			var:cp_layla_age >= 35
			NOT = { has_variable = cp_layla_seen_district_office }
			OR = {
				has_variable = cp_layla_receives_welfare
				has_variable = cp_layla_has_voted
				has_variable = cp_layla_owns_land
				has_variable = cp_layla_self_employed
			}
		}
		random_list = {
			10 = {
				cp_try_fire_milestone = { event = cp_layla.125 person = layla }
			}
			90 = { }
		}
	}
}

# ============================================================================
# cp_layla_dispatch_law_enacted -- Layla-only law reactions
# ============================================================================
# Shared law contests stay in cp_on_law_enacted so one historical moment can
# choose between eligible people. This effect owns only Layla's single-person
# law reactions.
cp_layla_dispatch_law_enacted = {
	# Welfare-ladder transition. Any lawgroup_welfare change reaches
	# here. We cache old tier, refresh to the new tier, compare, and
	# fire up/down events.
	if = {
		limit = {
			OR = {
				ROOT.currently_enacting_law.type ?= law_type:law_no_social_security
				ROOT.currently_enacting_law.type ?= law_type:law_chiefs_distribute_aid
				ROOT.currently_enacting_law.type ?= law_type:law_poor_laws
				ROOT.currently_enacting_law.type ?= law_type:law_wage_subsidies
				ROOT.currently_enacting_law.type ?= law_type:law_old_age_pension
			}
		}
		# Snapshot the prior tier into a temp var before refresh.
		set_variable = { name = cp_layla_welfare_tier_prev value = var:cp_layla_welfare_tier }
		cp_refresh_welfare_tier = yes
		# Up transition: new > prev.
		if = {
			limit = { var:cp_layla_welfare_tier > var:cp_layla_welfare_tier_prev }
			cp_try_fire_law_reaction = { event = cp_layla.91 person = layla }
		}
		else_if = {
			# Down transition: new < prev. (Repeal / regression.)
			limit = { var:cp_layla_welfare_tier < var:cp_layla_welfare_tier_prev }
			cp_try_fire_law_reaction = { event = cp_layla.92 person = layla }
		}
	}


	# Women's rights enactments. First-time-only per rung; the one-shot
	# gate is the cp_layla_seen_womens_rights_* flag set inside each event's
	# immediate block. Rollback events are not fired (user-design: each
	# rung is a "first enactment" memory, not a ladder mirror).
	if = {
		limit = {
			ROOT.currently_enacting_law.type ?= law_type:law_women_in_the_fields
			NOT = { has_variable = cp_layla_seen_womens_rights_fields }
		}
		cp_try_fire_law_reaction = { event = cp_layla.93 person = layla }
	}

	else_if = {
		limit = {
			ROOT.currently_enacting_law.type ?= law_type:law_women_in_the_workplace
			NOT = { has_variable = cp_layla_seen_womens_rights_workplace }
		}
		cp_try_fire_law_reaction = { event = cp_layla.95 person = layla }
	}

	else_if = {
		limit = {
			ROOT.currently_enacting_law.type ?= law_type:law_womens_suffrage
			NOT = { has_variable = cp_layla_seen_womens_rights_suffrage }
		}
		cp_try_fire_law_reaction = { event = cp_layla.96 person = layla }
	}

	if = {
		limit = {
			ROOT.currently_enacting_law.type ?= law_type:law_compulsory_primary_school
			NOT = { has_variable = cp_layla_seen_compulsory_school }
		}
		cp_try_fire_law_reaction = { event = cp_layla.101 person = layla }
	}


	# Slavery: emancipation is the big beat. The starting states
	# (slave_trade / legacy / debt) are her baseline — no reaction
	# event for those; they're handled via the ambient pulse
	# cp_layla.104 which fires while those laws are active.
	if = {
		limit = {
			ROOT.currently_enacting_law.type ?= law_type:law_slavery_banned
			NOT = { has_variable = cp_layla_seen_emancipation }
		}
		cp_try_fire_law_reaction = { event = cp_layla.103 person = layla }
	}


	# Regime: universal suffrage + single-party state. One-shot each.
	if = {
		limit = {
			ROOT.currently_enacting_law.type ?= law_type:law_universal_suffrage
			NOT = { has_variable = cp_layla_seen_universal_suffrage }
		}
		cp_try_fire_law_reaction = { event = cp_layla.105 person = layla }
	}

	else_if = {
		limit = {
			ROOT.currently_enacting_law.type ?= law_type:law_single_party_state
			NOT = { has_variable = cp_layla_seen_single_party }
		}
		cp_try_fire_law_reaction = { event = cp_layla.106 person = layla }
	}


	# Citizenship: multicultural. One-shot.
	if = {
		limit = {
			ROOT.currently_enacting_law.type ?= law_type:law_multicultural
			NOT = { has_variable = cp_layla_seen_multicultural }
		}
		cp_try_fire_law_reaction = { event = cp_layla.107 person = layla }
	}


	# Health system rungs. Each one-shot.
	if = {
		limit = {
			ROOT.currently_enacting_law.type ?= law_type:law_charitable_health_system
			NOT = { has_variable = cp_layla_seen_charitable_health }
		}

# Called from the shared war helpers after entering Egypt country scope.
cp_layla_dispatch_war_started = {
	if = {
		limit = {
			has_variable = cp_shared_war_start_target_layla
			has_variable = cp_layla_ahmed_alive
			var:cp_layla_ahmed_alive > 0
			NOT = { has_variable = cp_layla_ahmed_at_war }
			NOT = { has_variable = cp_layla_seen_ahmed_conscripted }
			var:cp_layla_ahmed_age >= 18
			var:cp_layla_ahmed_age <= 45
		}
		remove_variable = cp_shared_war_start_target_layla
		cp_try_fire_milestone = { event = cp_layla.4 person = layla }
	}
}

cp_layla_dispatch_war_end = {
	if = {
		limit = {
			has_variable = cp_layla_ahmed_alive
			var:cp_layla_ahmed_alive > 0
			has_variable = cp_layla_ahmed_at_war
		}
		# Clear the conscription gate so the next war can fire cp_layla.4
		# again. cp_layla_ahmed_at_war itself is cleared by the outcome effect
		# in cp_memory (cp_ahmed_homecoming / cp_ahmed_killed).
		remove_variable = cp_layla_seen_ahmed_conscripted

		# Outcome weights by war result. recently_won_war and recently_lost_war
		# are vanilla-set on country scope at on_capitulation (days = 1825
		# timer, so they're fresh here).
		if = {
			limit = { has_variable = recently_won_war }
			random_list = {
				75 = { cp_try_fire_milestone = { event = cp_layla.40 person = layla } }
				25 = { cp_try_fire_milestone = { event = cp_layla.41 person = layla } }
			}
		}
		else_if = {
			limit = { has_variable = recently_lost_war }
			random_list = {
				40 = { cp_try_fire_milestone = { event = cp_layla.40 person = layla } }
				60 = { cp_try_fire_milestone = { event = cp_layla.41 person = layla } }
			}
		}
		else = {
			# Neither side fully capitulated (white peace, stalemate).
			# Still brutal; still a coin flip with a slight pull toward home.
			random_list = {
				55 = { cp_try_fire_milestone = { event = cp_layla.40 person = layla } }
				45 = { cp_try_fire_milestone = { event = cp_layla.41 person = layla } }
			}
		}
	}
}

# ============================================================================
# Taxation -- cp_layla_tax_burden 0..10, how hard the tax regime hits HER
# ============================================================================
# V3 tracks five tax levels (very_low/low/medium/high/very_high) and five
# tax methods (land / per_capita / proportional / graduated / consumption).
# A peasant under Land-Based Taxation at Very High is crushed; the same
# peasant under Per-Capita Taxation at Very High is exempt. Consumption
# taxes on basic goods (grain, fish, fabric) hurt the poor more than the
# rich because peasants spend a larger share of wages on them.
#
# Vanilla triggers used:
#   tax_level >= high                    (country scope)
#   has_law = law_type:law_X             (country scope)
#   has_consumption_tax = g:grain        (country scope)
# ============================================================================
cp_refresh_tax_burden = {
	set_variable = { name = cp_layla_tax_burden value = 0 }

	# Who pays the main tax method? Peasants + subsistence farmers are
	# land-taxed; middle strata pay per-capita / proportional; laborers
	# pay a sliver through proportional + consumption.
	if = {
		limit = {
			OR = {
				has_variable = cp_layla_profession_peasants
				has_variable = cp_layla_profession_farmers
			}
		}
		# Land-Based Taxation hits her directly as a peasant.
		if = {
			limit = { has_law = law_type:law_land_based_taxation }
			if = {
				limit = { tax_level >= very_high }
				change_variable = { name = cp_layla_tax_burden add = 8 }
			}
			else_if = {
				limit = { tax_level >= high }
				change_variable = { name = cp_layla_tax_burden add = 6 }
			}
			else_if = {
				limit = { tax_level >= medium }
				change_variable = { name = cp_layla_tax_burden add = 4 }
			}
			else_if = {
				limit = { tax_level >= low }
				change_variable = { name = cp_layla_tax_burden add = 2 }
			}
			# very_low = +0 (effectively untaxed)
		}
		# Per-Capita taxation exempts peasants entirely — she pays nothing.
		# Proportional/graduated tax wages; peasants have none.
		# So no other tax-method branch fires for her under the peasant arc.
	}
	else = {
		# Middle/upper strata pay per-capita, proportional, or graduated.
		if = {
			limit = {
				OR = {
					has_law = law_type:law_per_capita_based_taxation
					has_law = law_type:law_proportional_taxation
					has_law = law_type:law_graduated_taxation
				}
			}
			if = {
				limit = { tax_level >= very_high }
				change_variable = { name = cp_layla_tax_burden add = 6 }
			}
			else_if = {
				limit = { tax_level >= high }
				change_variable = { name = cp_layla_tax_burden add = 4 }
			}
			else_if = {
				limit = { tax_level >= medium }
				change_variable = { name = cp_layla_tax_burden add = 2 }
			}
			else_if = {
				limit = { tax_level >= low }
				change_variable = { name = cp_layla_tax_burden add = 1 }
			}
		}
	}

	# Consumption taxes on goods she actually consumes. Weights reflect
	# how central each good is to a peasant household's buy package:
	# grain is the loaf; fabric is clothes; fish/meat sometimes; sugar
	# and coffee are later-game luxuries.
	if = {
		limit = { has_consumption_tax = g:grain }
		change_variable = { name = cp_layla_tax_burden add = 2 }
	}
	if = {
		limit = { has_consumption_tax = g:fish }
		change_variable = { name = cp_layla_tax_burden add = 1 }
	}
	if = {
		limit = { has_consumption_tax = g:fabric }
		change_variable = { name = cp_layla_tax_burden add = 1 }
	}
	if = {
		limit = { has_consumption_tax = g:meat }
		change_variable = { name = cp_layla_tax_burden add = 0.5 }
	}
	if = {
		limit = { has_consumption_tax = g:sugar }
		change_variable = { name = cp_layla_tax_burden add = 0.5 }
	}
	if = {
		limit = { has_consumption_tax = g:coffee }
		change_variable = { name = cp_layla_tax_burden add = 0.3 }
	}

	clamp_variable = { name = cp_layla_tax_burden min = 0 max = 10 }
}

cp_pop_tick_literacy = {
	# Try local cohort first.
	every_scope_state = {
		limit = { has_variable = cp_layla_lives_here }
		cp_read_cohort_literacy_only = yes
	}
	# Diaspora fallback.
	if = {
		limit = { NOT = { exists = scope:tmp_cohort_literacy } }
		cp_read_diaspora_cohort_literacy = yes
	}

	if = {
		limit = { exists = scope:tmp_cohort_literacy }
		# Convert cohort fraction (0..1) to percent (0..100) for comparison.
		set_variable = { name = cp_layla_cohort_literacy_target value = scope:tmp_cohort_literacy }
		change_variable = { name = cp_layla_cohort_literacy_target multiply = 100 }

		# Drift UP only when cohort is ahead of her. Age-banded.
		if = {
			limit = {
				var:cp_layla_cohort_literacy_target > var:cp_layla_literacy
				var:cp_layla_age < 60
			}
			if = {
				limit = { var:cp_layla_age < 30 }
				change_variable = { name = cp_layla_literacy add = 3 }
			}
			else_if = {
				limit = { var:cp_layla_age < 40 }
				change_variable = { name = cp_layla_literacy add = 2 }
			}
			else = {
				change_variable = { name = cp_layla_literacy add = 1 }
			}
		}
		clamp_variable = { name = cp_layla_literacy min = 0 max = 100 }
	}
}

# ----------------------------------------------------------------------------
# Political-mood pop-tick -- expected-SoL gap drives radicalism/loyalism.
#
# V3's own formula (standardofliving.txt): a pop below expected SoL gains
# +0.2% radicalism per level-gap per month (cap 15%). Above expected SoL
# produces loyalism. Peasants have sol_change_impact = 0.01 so year-over-
# year SoL SWINGS barely move them — what radicalises peasants is the
# chronic gap between actual and expected.
#
# Expected SoL per the plan §0.5: lower strata = 5, middle = 10.
# On a yearly tick we scale roughly 12× monthly: a 1-point gap ≈ +2.4%/yr;
# we bucket it as +1 per small gap, +2 per mid, +3 per wide.
#
# cp_shift_radical / cp_shift_loyalist zero the opposite automatically
# (V3's mutual-exclusion rule encoded in the library).
# ----------------------------------------------------------------------------
cp_pop_tick_political_mood = {
	# Expected SoL for her current profession. Lower strata 5, middle 10,
	# upper 20 (capitalists / aristocrats per V3 profession.txt).
	set_variable = { name = cp_layla_expected_sol value = 5 }
	if = {
		limit = {
			OR = {
				has_variable = cp_layla_profession_clerks
				has_variable = cp_layla_profession_shopkeepers
				has_variable = cp_layla_profession_machinists
				has_variable = cp_layla_profession_engineers
				has_variable = cp_layla_profession_academics
				has_variable = cp_layla_profession_bureaucrats
				has_variable = cp_layla_profession_farmers
			}
		}
		set_variable = { name = cp_layla_expected_sol value = 10 }
	}
	if = {
		limit = {
			has_variable = cp_layla_profession_capitalists
		}
		set_variable = { name = cp_layla_expected_sol value = 20 }
	}

	# cp_layla_sol_gap = expected - cp_layla_sol. Positive → she's below expected.
	# multiply-by-(-1) trick because `subtract = var:X` is unreliable.
	set_variable = { name = cp_layla_sol_gap value = var:cp_layla_sol }
	change_variable = { name = cp_layla_sol_gap multiply = -1 }
	change_variable = { name = cp_layla_sol_gap add = var:cp_layla_expected_sol }

	# Below expected → radicalism drift.
	if = {
		limit = { var:cp_layla_sol_gap >= 5 }
		cp_shift_radical = { person = layla delta = 3 }
	}
	else_if = {
		limit = { var:cp_layla_sol_gap >= 2 }
		cp_shift_radical = { person = layla delta = 2 }
	}
	else_if = {
		limit = { var:cp_layla_sol_gap >= 1 }
		cp_shift_radical = { person = layla delta = 1 }
	}
	# Above expected → loyalism drift.
	else_if = {
		limit = { var:cp_layla_sol_gap <= -5 }
		cp_shift_loyalist = { person = layla delta = 3 }
	}
	else_if = {
		limit = { var:cp_layla_sol_gap <= -2 }
		cp_shift_loyalist = { person = layla delta = 2 }
	}
	else_if = {
		limit = { var:cp_layla_sol_gap <= -1 }
		cp_shift_loyalist = { person = layla delta = 1 }
	}
	# Gap ≈ 0 → no drift; the mutually-exclusive pair decays slowly via
	# V3's own tick and through event-driven cp_shift_* calls.
}

# Clear the cp_layla_lives_here marker from wherever it currently sits.
# Run from country scope (root = c:EGY). We search across EVERY country's
# states in case she's been displaced by border changes and now lives in
# a state EGY no longer owns.
cp_clear_home_state = {
	every_country = {
		every_scope_state = {
			limit = { has_variable = cp_layla_lives_here }
			remove_variable = cp_layla_lives_here
		}
	}
	# Urban sub-locators (cities within her state — Cairo, Alexandria,
	# Damascus, etc.) are still flag-per-hub on the country because
	# they name the hub, not the state.
	remove_variable = cp_layla_in_city
	remove_variable = cp_layla_in_cairo
	remove_variable = cp_layla_in_alexandria
}

# Move Layla to the specified state. Takes $state$ as a V3 state region
# id. Muhammad Ali's Egypt at 1836 owns a huge arc of states:
#   Nile region:  STATE_LOWER_EGYPT, STATE_MIDDLE_EGYPT, STATE_UPPER_EGYPT,
#                 STATE_EGYPTIAN_DESERT, STATE_SINAI, STATE_MATRUH,
#                 STATE_DONGOLA, STATE_KORDOFAN, STATE_BLUE_NILE
#   Arabia:       STATE_ALEPPO, STATE_LEBANON, STATE_PALESTINE, STATE_SYRIA,
#                 STATE_TRANSJORDAN
#   Anatolia:     STATE_ADANA (shared with Turkey)
#   Ethiopia:     STATE_ERITREA
#   Balkans:      STATE_CRETE
# Any of these is a valid $state$ input; see gamestates.txt for the full
# V3 state catalogue.
#
# Sets the marker on whichever region_state in that region is owned by
# EGY; if EGY doesn't own any of it, falls back to the first region_state
# so the marker still lands (she's living under foreign rule).
#
# Also stores a pointer to her state as a country variable, so the JE
# label can render the state's vanilla name via
# [Country.Var('cp_layla_state_pointer').GetState.GetStateRegion.GetName]
# (vanilla pattern: expedition_location_var in 00_expedition_effects.txt).
cp_move_to_state = {
	cp_clear_home_state = yes
	s:$state$ ?= {
		# Prefer the EGY-owned region_state when present (her home country).
		if = {
			limit = {
				any_scope_state = { owner = c:EGY }
			}
			random_scope_state = {
				limit = { owner = c:EGY }
				set_variable = cp_layla_lives_here
				root = {
					set_variable = { name = cp_layla_state_pointer value = prev }
				}
			}
		}
		else = {
			# EGY doesn't own any of this region — she's under foreign
			# rule. Pick any region_state so the marker still lands.
			random_scope_state = {
				set_variable = cp_layla_lives_here
				root = {
					set_variable = { name = cp_layla_state_pointer value = prev }
				}
			}
		}
	}
}

# ============================================================================
# Mortality pressure -- yearly risk synthesised from age, health laws, SoL
# ============================================================================
# cp_layla_mortality_pressure 0..10 captures how close she sits to the edge of her
# life in any given year. Computed yearly from:
#   - age (the main driver; rises sharply past 60)
#   - SoL (low SoL = malnutrition risk)
#   - country health law (public > private > charitable for the poor)
#   - wealth + private insurance access
#   - pollution in her state (industrial cities erode the lungs)
#
# V3 baselines (standardofliving.txt wiki): mild starvation +30% mortality,
# severe +100%; pollution +0.5% per impact. We compress these into a 0..10
# dial that downstream random rolls read.
# ============================================================================
cp_refresh_mortality_pressure = {
	set_variable = { name = cp_layla_mortality_pressure value = 0 }

	# Age bands. Peasants in 1836 did not typically see 70; a 70-yr-old is
	# already living on borrowed time.
	if = {
		limit = { var:cp_layla_age >= 70 }
		change_variable = { name = cp_layla_mortality_pressure add = 4 }
	}
	else_if = {
		limit = { var:cp_layla_age >= 60 }
		change_variable = { name = cp_layla_mortality_pressure add = 3 }
	}
	else_if = {
		limit = { var:cp_layla_age >= 50 }
		change_variable = { name = cp_layla_mortality_pressure add = 2 }
	}
	else_if = {
		limit = { var:cp_layla_age >= 40 }
		change_variable = { name = cp_layla_mortality_pressure add = 1 }
	}

	# Low SoL: malnutrition / exposure. V3's food-security numbers suggest
	# mortality rises sharply below Struggling (SoL 5).
	if = {
		limit = { var:cp_layla_sol <= 3 }
		change_variable = { name = cp_layla_mortality_pressure add = 3 }
	}
	else_if = {
		limit = { var:cp_layla_sol <= 5 }
		change_variable = { name = cp_layla_mortality_pressure add = 2 }
	}
	else_if = {
		limit = { var:cp_layla_sol <= 8 }
		change_variable = { name = cp_layla_mortality_pressure add = 1 }
	}

	# Health-care law. V3 ships three: charitable (default, worst), private
	# (helps those who can pay), public (helps everyone).
	if = {
		limit = { has_law = law_type:law_public_health_insurance }
		change_variable = { name = cp_layla_mortality_pressure add = -2 }
	}
	else_if = {
		limit = { has_law = law_type:law_private_health_insurance }
		# Private insurance only helps households that can afford it.
		if = {
			limit = { var:cp_layla_sol >= 25 }
			change_variable = { name = cp_layla_mortality_pressure add = -2 }
		}
		else_if = {
			limit = { var:cp_layla_sol >= 15 }
			change_variable = { name = cp_layla_mortality_pressure add = -1 }
		}
		# Below SoL 15 private insurance is out of reach — no bonus.
	}
	else_if = {
		limit = { has_law = law_type:law_charitable_health_system }
		# Charitable system covers the poorest a little.
		if = {
			limit = { var:cp_layla_sol <= 10 }
			change_variable = { name = cp_layla_mortality_pressure add = -1 }
		}
	}

	# Urban pollution: an indutrial city erodes the lungs. Approximation
	# while state_pollution trigger names are being finalised — we gate
	# on cp_layla_in_city + cp_layla_seen_first_textile_mill as a proxy for "she
	# breathes industrial air".
	if = {
		limit = {
			has_variable = cp_layla_in_city
			has_variable = cp_layla_seen_first_textile_mill
		}
		change_variable = { name = cp_layla_mortality_pressure add = 1 }
	}

	# Welfare. V3 ships a 5-rung lawgroup_welfare AND an accompanying
	# institution_social_security whose investment level (0..5) scales the
	# law's real effect. A nominally pension-law country at institution
	# level 0 is barely better than no_social_security; at level 5 it is
	# meaningfully protective. We stack the modifiers: the law decides WHICH
	# cohort is eligible, the institution level decides HOW MUCH help.
	#
	# Pattern verified via vanilla 06_portugal_politics.txt:656 — the
	# institution_investment_level trigger takes institution + value.
	if = {
		limit = {
			has_law = law_type:law_poor_laws
			var:cp_layla_sol <= 10
		}
		change_variable = { name = cp_layla_mortality_pressure add = -1 }
		if = {
			limit = {
				institution_investment_level = { institution = institution_social_security value >= 3 }
			}
			change_variable = { name = cp_layla_mortality_pressure add = -1 }
		}
	}
	if = {
		limit = {
			has_law = law_type:law_wage_subsidies
			var:cp_layla_sol <= 15
			OR = {
				has_variable = cp_layla_profession_laborers
				has_variable = cp_layla_profession_machinists
				has_variable = cp_layla_profession_farmers
				has_variable = cp_layla_profession_clerks
			}
		}
		change_variable = { name = cp_layla_mortality_pressure add = -1 }
		if = {
			limit = {
				institution_investment_level = { institution = institution_social_security value >= 3 }
			}
			change_variable = { name = cp_layla_mortality_pressure add = -1 }
		}
	}
	if = {
		limit = {
			has_law = law_type:law_old_age_pension
			var:cp_layla_age >= 60
		}
		change_variable = { name = cp_layla_mortality_pressure add = -1 }
		if = {
			limit = {
				institution_investment_level = { institution = institution_social_security value >= 2 }
			}
			change_variable = { name = cp_layla_mortality_pressure add = -1 }
		}
		if = {
			limit = {
				institution_investment_level = { institution = institution_social_security value >= 4 }
			}
			change_variable = { name = cp_layla_mortality_pressure add = -1 }
		}
	}

	clamp_variable = { name = cp_layla_mortality_pressure min = 0 max = 10 }

# ============================================================================
# Worker-safety pressure -- Ahmed's annual accident risk at the mill
# ============================================================================
# cp_layla_worker_risk 0..10 drives the Ahmed-at-work-dies random roll. Weighted
# by whether Ahmed has an industrial job, the country's worker-rights law,
# and whether the household owns the mill (owners don't tend machines).
# ============================================================================
cp_refresh_worker_safety_pressure = {
	set_variable = { name = cp_layla_worker_risk value = 0 }

	# Only if Ahmed is on a shop floor. Owners (shopkeepers/capitalists)
	# don't take mill-accident risk, even if they used to.
	if = {
		limit = {
			has_variable = cp_layla_ahmed_alive
			var:cp_layla_ahmed_alive > 0
			OR = {
				has_variable = cp_layla_ahmed_profession_laborer
			}
		}
		# Base industrial risk.
		change_variable = { name = cp_layla_worker_risk add = 3 }

		# Law gates. No worker rights = worst; protections cut risk.
		if = {
			limit = { has_law = law_type:law_no_workers_rights }
			change_variable = { name = cp_layla_worker_risk add = 3 }
		}
		else_if = {
			limit = { has_law = law_type:law_worker_protections }
			change_variable = { name = cp_layla_worker_risk add = -2 }
		}
		# law_regulatory_bodies is the implicit baseline; no modifier.
	}

	clamp_variable = { name = cp_layla_worker_risk min = 0 max = 10 }
}

# ============================================================================
# Welfare tier tracking -- which rung of the welfare ladder is EGY on
# ============================================================================
# cp_layla_welfare_tier 0..4 captures the country's current position on the
# lawgroup_welfare ladder. Updated on law enactment; compared to its prior
# value to fire up/down transition events. Ordering is the canonical V3
# rung order (no_social_security lowest, old_age_pension highest).
#
#   0 = law_no_social_security (default)
#   1 = law_chiefs_distribute_aid (Mongolia variant, ~= poor_laws)
#   2 = law_poor_laws
#   3 = law_wage_subsidies
#   4 = law_old_age_pension
# ============================================================================
cp_refresh_welfare_tier = {
	if = {
		limit = { has_law = law_type:law_old_age_pension }
		set_variable = { name = cp_layla_welfare_tier value = 4 }
	}
	else_if = {
		limit = { has_law = law_type:law_wage_subsidies }
		set_variable = { name = cp_layla_welfare_tier value = 3 }
	}
	else_if = {
		limit = { has_law = law_type:law_poor_laws }
		set_variable = { name = cp_layla_welfare_tier value = 2 }
	}
	else_if = {
		limit = { has_law = law_type:law_chiefs_distribute_aid }
		set_variable = { name = cp_layla_welfare_tier value = 1 }
	}
	else = {
		set_variable = { name = cp_layla_welfare_tier value = 0 }
	}
}
