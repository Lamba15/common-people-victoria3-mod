# --- Refresh SoL from pop cohort -------------------------------------------
cp_nour_refresh_sol = {
	cp_refresh_sol_from_pop = {
		person = nour
		culture = misri
		religion = sunni
		poptype = shopkeepers
	}
}


# --- Dispatcher branches ----------------------------------------------------
cp_nour_dispatch_startup = {
	cp_person_hook_noop = yes
}

cp_nour_dispatch_monthly = {
	cp_person_hook_noop = yes
}

cp_nour_dispatch_yearly = {
	cp_age_increment = { person = nour }
	cp_nour_refresh_sol = yes
	cp_person_yearly_lifecycle = { person = nour country = EGY }
	if = {
		limit = {
			cp_country_women_property_rights = yes
			NOT = { has_variable = cp_nour_seen_rent_receipt }
		}
		cp_try_fire_world_response = { event = cp_nour.30 person = nour }
	}
}

cp_nour_dispatch_tech = {
	if = {
		limit = {
			cp_country_has_modern_records = yes
			NOT = { has_variable = cp_nour_seen_register_shelf }
		}
		cp_try_fire_world_response = { event = cp_nour.50 person = nour }
	}
}

cp_nour_dispatch_building = {
	if = {
		limit = {
			is_building_type = building_government_administration
			cp_building_affects_person_workplace = { person = nour }
			owner ?= {
				THIS = c:EGY
				cp_person_is_alive = { person = nour }
				NOT = { has_variable = cp_nour_seen_register_shelf }
			}
		}
		owner ?= {
			cp_try_fire_world_response = { event = cp_nour.50 person = nour }
		}
	}
}
