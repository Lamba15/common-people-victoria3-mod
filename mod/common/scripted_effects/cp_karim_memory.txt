# Common People -- Karim al-Nahhas's memory
#
# Person 13 scaffold. Karim is registered as `karim`, not as a V3 character.
# He enters conditionally when machine tools, steel, or engines make factory
# skill visible through ordinary hands.
###############################################################################


# --- Refresh SoL from pop cohort -------------------------------------------
cp_karim_refresh_sol = {
	cp_refresh_sol_from_pop = {
		person = karim
		culture = misri
		religion = sunni
		poptype = machinists
	}
}


# --- Conditional entry ------------------------------------------------------
cp_karim_try_spawn = {
	if = {
		limit = {
			c:EGY ?= this
			NOT = { has_global_variable = cp_person_karim_alive }
			NOT = { has_variable = cp_karim_alive }
			OR = {
				cp_country_has_machine_industry = yes
				has_technology_researched = mechanical_tools
				AND = {
					cp_country_has_manufacturing = yes
					cp_era_1850_to_1880 = yes
				}
				AND = {
					cp_country_has_manufacturing = yes
					cp_era_after_1880 = yes
				}
			}
		}
		cp_mark_person_spawn_home_from_machine_industry = { person = karim }
		cp_mark_person_spawn_home_from_manufacturing = { person = karim }
		cp_register_person = {
			person = karim
			country = EGY
		}
		cp_karim_dispatch_startup = yes
		trigger_event = { id = cp_karim.1 }
	}
}

			10 = { cp_try_fire_world_response = { event = cp_karim.30 person = karim } }

			NOT = { has_variable = cp_karim_seen_night_gauge }

		}

			12 = { cp_try_fire_world_response = { event = cp_karim.60 person = karim } }

	}
}

cp_karim_dispatch_tech = {
	if = {
		limit = {
			has_technology_researched = mechanical_tools
			cp_person_home_has_machine_industry = { person = karim }
			NOT = { has_variable = cp_karim_seen_tool_bench }
		}
		cp_try_fire_world_response = { event = cp_karim.20 person = karim }
	}
	if = {
		limit = {
			OR = {
				has_technology_researched = rotary_valve_engine
				has_technology_researched = combustion_engine
			}
			cp_person_home_has_machine_industry = { person = karim }
			NOT = { has_variable = cp_karim_seen_new_engine }
		}
		cp_try_fire_world_response = { event = cp_karim.40 person = karim }
	}
}

	}
