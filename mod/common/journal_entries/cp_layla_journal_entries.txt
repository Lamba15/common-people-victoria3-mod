###############################################################################
# Common People -- journal entries
#
# cp_je_layla -- Layla's life. Pinned by default for Egypt at game start.
# Not a goal; does not complete on success. It exists while she does.
#
# Scope: country (c:EGY). All of Layla's state lives as country variables on
# this scope, so every read here is `var:cp_X` and every write is a call to
# a scripted_effect that ultimately resolves to country-scope mutations.
#
# UI composition:
#   - Title / icon (her face as a scene image, not a portrait)
#   - Status line: 12 triggered_desc variants (4 life situations x 3 moods)
#   - Body prose: cp_je_layla_reason -- lasting "who she is" paragraphs
#   - Button: cp_layla_check_button
#   - Current-value: a monthly debug tick during Phase A verification; will
#     be removed when the pulse is proven to fire.
#
# Pulses:
#   - on_monthly_pulse: cp_debug_tick_increment + reroll cp_mood_roll
#   - on_yearly_pulse: cp_age_increment
###############################################################################

cp_je_layla = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	group = je_group_internal_affairs

	# Monthly-updating status. first_valid picks the first matching branch,
	# so more specific conditions (situation + mood) come before general
	# (situation only, then mood only, then default).
	status_desc = {
		first_valid = {
			# --- Ahmed at war ---
			triggered_desc = {
				trigger = {
					has_variable = cp_ahmed_at_war
					var:cp_mood_roll >= 7
				}
				desc = cp_je_layla_status_war_high
			}
			triggered_desc = {
				trigger = {
					has_variable = cp_ahmed_at_war
					var:cp_mood_roll <= 3
				}
				desc = cp_je_layla_status_war_low
			}
			triggered_desc = {
				trigger = { has_variable = cp_ahmed_at_war }
				desc = cp_je_layla_status_war_mid
			}

			# --- Under the restored serfdom ---
			triggered_desc = {
				trigger = {
					has_variable = cp_under_serfdom
					var:cp_mood_roll >= 7
				}
				desc = cp_je_layla_status_serfdom_high
			}
			triggered_desc = {
				trigger = {
					has_variable = cp_under_serfdom
					var:cp_mood_roll <= 3
				}
				desc = cp_je_layla_status_serfdom_low
			}
			triggered_desc = {
				trigger = { has_variable = cp_under_serfdom }
				desc = cp_je_layla_status_serfdom_mid
			}

			# --- She owns the field ---
			triggered_desc = {
				trigger = {
					has_variable = cp_owns_land
					var:cp_mood_roll >= 7
				}
				desc = cp_je_layla_status_homesteader_high
			}
			triggered_desc = {
				trigger = {
					has_variable = cp_owns_land
					var:cp_mood_roll <= 3
				}
				desc = cp_je_layla_status_homesteader_low
			}
			triggered_desc = {
				trigger = { has_variable = cp_owns_land }
				desc = cp_je_layla_status_homesteader_mid
			}

			# --- Default: tenant's wife ---
			triggered_desc = {
				trigger = { var:cp_mood_roll >= 7 }
				desc = cp_je_layla_status_tenant_high
			}
			triggered_desc = {
				trigger = { var:cp_mood_roll <= 3 }
				desc = cp_je_layla_status_tenant_low
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = cp_je_layla_status_tenant_mid
			}
		}
	}

	scripted_button = cp_layla_check_button

	# Monthly: tick the debug counter + reroll her mood (1..10).
	on_monthly_pulse = {
		effect = {
			cp_debug_tick_increment = yes
			random_list = {
				10 = { set_variable = { name = cp_mood_roll value = 1 } }
				10 = { set_variable = { name = cp_mood_roll value = 2 } }
				10 = { set_variable = { name = cp_mood_roll value = 3 } }
				10 = { set_variable = { name = cp_mood_roll value = 4 } }
				10 = { set_variable = { name = cp_mood_roll value = 5 } }
				10 = { set_variable = { name = cp_mood_roll value = 6 } }
				10 = { set_variable = { name = cp_mood_roll value = 7 } }
				10 = { set_variable = { name = cp_mood_roll value = 8 } }
				10 = { set_variable = { name = cp_mood_roll value = 9 } }
				10 = { set_variable = { name = cp_mood_roll value = 10 } }
			}
		}
	}

	# Yearly: increment age.
	on_yearly_pulse = {
		effect = {
			cp_age_increment = yes
		}
	}

	weight = 100
	should_be_pinned_by_default = yes
}
