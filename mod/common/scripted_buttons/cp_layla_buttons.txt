###############################################################################
# Common People -- scripted buttons
#
# cp_layla_check_button -- the "Check on Layla" action on her JE. Dispatches
# to one of several reflection events. Each branch:
#   - gates on her current life situation
#   - gates on a "seen recently" timed flag so the same beat does not repeat
#     within 6 months
#   - sets the seen-recently flag when chosen
#
# Scope at effect time: country (c:EGY, the JE owner). All Layla state reads
# as `has_variable = X` / `var:X` directly.
###############################################################################

cp_layla_check_button = {
	name = "cp_layla_check_button_name"
	desc = "cp_layla_check_button_desc"

	visible = {
		always = yes
	}

	possible = {
		has_variable = cp_layla_alive
		NOT = { has_variable = cp_layla_button_cooldown }
	}

	# The cooldown is set by each reflection event's immediate block
	# (cp_layla.10..23), so clicking the button AND the JE's monthly
	# random_events auto-fire use the same 90-day lockout.
	effect = {
		random_list = {
			# Situation-specific: letter that has not come (Ahmed at war)
			15 = {
				trigger = {
					has_variable = cp_layla_ahmed_at_war
					NOT = { has_variable = cp_layla_seen_letter }
				}
				set_variable = { name = cp_layla_seen_letter days = 180 }
				cp_button_fire = { event = cp_layla.10 person = layla }
			}
			# Situation-specific: walking the field (homesteader)
			15 = {
				trigger = {
					has_variable = cp_layla_owns_land
					NOT = { has_variable = cp_layla_seen_walking_field }
				}
				set_variable = { name = cp_layla_seen_walking_field days = 180 }
				cp_button_fire = { event = cp_layla.11 person = layla }
			}
			# Situation-specific: under the bey's eye (serfdom restored)
			15 = {
				trigger = {
					has_variable = cp_layla_under_serfdom
					NOT = { has_variable = cp_layla_seen_under_beys_eye }
				}
				set_variable = { name = cp_layla_seen_under_beys_eye days = 180 }
				cp_button_fire = { event = cp_layla.12 person = layla }
			}
			# Always-on: a day with no wind
			10 = {
				trigger = {
					NOT = { has_variable = cp_layla_seen_no_wind }
				}
				set_variable = { name = cp_layla_seen_no_wind days = 180 }
				cp_button_fire = { event = cp_layla.13 person = layla }
			}
			# Always-on: the neighbour's boy in the lane
			10 = {
				trigger = {
					NOT = { has_variable = cp_layla_seen_neighbours_boy }
				}
				set_variable = { name = cp_layla_seen_neighbours_boy days = 180 }
				cp_button_fire = { event = cp_layla.14 person = layla }
			}
			# Always-on: her mother's left hand
			10 = {
				trigger = {
					NOT = { has_variable = cp_layla_seen_mothers_hand }
				}
				set_variable = { name = cp_layla_seen_mothers_hand days = 180 }
				cp_button_fire = { event = cp_layla.15 person = layla }
			}
			# Always-on: the hen that will not lay
			10 = {
				trigger = { NOT = { has_variable = cp_layla_seen_hen } }
				set_variable = { name = cp_layla_seen_hen days = 180 }
				cp_button_fire = { event = cp_layla.16 person = layla }
			}
			# Always-on: the merchant's scale
			10 = {
				trigger = { NOT = { has_variable = cp_layla_seen_scale } }
				set_variable = { name = cp_layla_seen_scale days = 180 }
				cp_button_fire = { event = cp_layla.17 person = layla }
			}
			# Always-on: Friday, the river low
			10 = {
				trigger = { NOT = { has_variable = cp_layla_seen_river_low } }
				set_variable = { name = cp_layla_seen_river_low days = 180 }
				cp_button_fire = { event = cp_layla.18 person = layla }
			}
			# Always-on: the old woman who sees through walls
			10 = {
				trigger = { NOT = { has_variable = cp_layla_seen_old_woman } }
				set_variable = { name = cp_layla_seen_old_woman days = 180 }
				cp_button_fire = { event = cp_layla.19 person = layla }
			}
			# Always-on: the bread that did not rise
			10 = {
				trigger = { NOT = { has_variable = cp_layla_seen_bread } }
				set_variable = { name = cp_layla_seen_bread days = 180 }
				cp_button_fire = { event = cp_layla.20 person = layla }
			}
			# Always-on: dust on the water
			10 = {
				trigger = { NOT = { has_variable = cp_layla_seen_dust } }
				set_variable = { name = cp_layla_seen_dust days = 180 }
				cp_button_fire = { event = cp_layla.21 person = layla }
			}
			# Always-on: the younger brother's letter
			10 = {
				trigger = { NOT = { has_variable = cp_layla_seen_brother } }
				set_variable = { name = cp_layla_seen_brother days = 180 }
				cp_button_fire = { event = cp_layla.22 person = layla }
			}
			# Always-on: the mule that will not move
			10 = {
				trigger = { NOT = { has_variable = cp_layla_seen_mule } }
				set_variable = { name = cp_layla_seen_mule days = 180 }
				cp_button_fire = { event = cp_layla.23 person = layla }
			}
			# Safety net: if every branch has been seen recently, fire a
			# generic reflection (cp_layla.6 "A Moment") so the button
			# never misfires.
			1 = {
				cp_button_fire = { event = cp_layla.6 person = layla }
			}
		}
	}
}


###############################################################################
# cp_layla_conversation_button -- "Imagine a conversation"
#
# Opens cp_conversation.1 (the root), which offers topic choices as options.
# Framing: she is imagining it. The player is a dream visitor. See
# documentation/characters/layla/conversation.md for the full design.
#
# Separate cooldown from the Check button: cp_conversation_cooldown (90 days).
# Both buttons can be used in the same window.
###############################################################################

cp_layla_conversation_button = {
	name = "cp_layla_conversation_button_name"
	desc = "cp_layla_conversation_button_desc"

	visible = {
		always = yes
	}

	possible = {
		has_variable = cp_layla_alive
		NOT = { has_variable = cp_layla_audience_cooldown }
	}

	# Cooldown is set by cp_conversation.90 (the closer), not here, so
	# that if the player quits mid-audience the button is not locked.
	effect = {
		cp_button_fire = { event = cp_conversation.1 person = layla }
	}
}
