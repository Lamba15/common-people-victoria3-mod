###############################################################################
# Common People -- shared scripted buttons
#
# Player-facing journal buttons should not belong to a single person unless
# that person has explicitly been promoted to a major, persistent arc. The only
# shared button shipped for now is the hidden-when-idle QA popup bridge used by
# generated console selectors.
###############################################################################

cp_shared_debug_fire_selected_button = {
	name = "cp_shared_debug_fire_selected_button_name"
	desc = "cp_shared_debug_fire_selected_button_desc"

	visible = {
		has_variable = cp_debug_selected_event
	}

	possible = {
		has_variable = cp_debug_selected_event
	}

	effect = {
		if = {
			limit = { has_variable = cp_debug_selected_event }
			cp_debug_fire_selected = yes
			remove_variable = cp_debug_selected_event
		}
	}
}


# ============================================================================
# Roster action buttons
# ============================================================================
# These buttons are shared JE controls, not person-owned journal entries. They
# bypass the automatic 1-2/year ambient budget because a player click is an
# explicit request to check in, but each person still gets a short cooldown and
# the selected event must have an unseen/currently-eligible beat.

cp_shared_action_layla_button = {
	name = "cp_shared_action_layla_button_name"
	desc = "cp_shared_action_layla_button_desc"

	visible = {
		cp_person_is_alive = { person = layla }
		cp_active_roster_has_person = { person = layla }
	}
	possible = {
		NOT = { has_variable = cp_layla_action_cooldown }
		cp_shared_action_layla_available = yes
	}
	effect = { cp_shared_action_layla = yes }
}
