###############################################################################
# Common People -- on_actions
#
# Hooks into base-game on_actions to drive the dispatcher layer. We NEVER
# override a base-game on_action; we always subscribe additively.
#
# Naming: cp_ prefix for every subscription target.
# Target version: Victoria 3 1.13 line.
#
# Current subscriptions:
#   on_game_started              -> cp_on_start
#   on_monthly_pulse_country     -> cp_on_monthly       (mood + ambient router)
#   on_yearly_pulse_country      -> cp_on_yearly        (person lifecycle)
#   on_law_enactment_pass        -> cp_on_law_enacted   (law reactions)
#   on_diplo_play_war_start      -> cp_on_war_started   (war routing)
#   on_war_end                   -> cp_on_war_end       (homecoming / death)
#   on_revolution_start          -> cp_on_revolution    (shared rupture resolver)
#   on_acquired_technology       -> cp_on_tech          (modernization routing)
#   on_building_built            -> cp_on_building_built (modernization routing)
#
# Planned subscriptions (wired in later phases):
#   on_production_method_changed -> mechanisation reactions
###############################################################################


# ============================================================================
# Game start: seed Egypt with the Common People roster
# ============================================================================

on_game_started = {
	on_actions = {
		cp_on_start
	}
}

cp_on_start = {
	effect = {
		set_global_variable = {
			name = cp_is_loaded
			value = yes
		}

		c:EGY ?= {
			# Shared person startup/register dispatch. Person-owned files
			# handle their own setup, migration, and state backfills.
			trigger_event = { id = cp_shared_startup.1 }
		}
	}
}

# ============================================================================
# Law enactment -- dispatch law-reaction events across the roster
# ============================================================================
# on_law_enactment_pass fires with root = country; the just-passed law is
# readable as currently_enacting_law (pattern confirmed in vanilla, e.g.
# je_corn_laws and vanilla law-reaction events).
# ============================================================================

on_law_enactment_pass = {
	on_actions = {
		cp_on_law_enacted
	}
}

cp_on_law_enacted = {
	effect = {
		if = {
			limit = {
				c:EGY ?= this
			}
			cp_nabil_try_spawn = yes
			cp_roll_public_order_law_reaction = yes
		}

		# Homesteading granted. Shared-trigger resolution decides whether
		# this beat belongs to Layla's deed or the Bey's opened gate.
		# Pattern mirrors vanilla 00_monarchy.txt: c:TAG ?= this + ROOT.currently_enacting_law.type ?= law_type:X.
		if = {
			limit = {
				c:EGY ?= this
				ROOT.currently_enacting_law.type ?= law_type:law_homesteading
				OR = {
					AND = {
						cp_person_is_alive = { person = layla }
						NOT = { has_variable = cp_layla_seen_homesteading }
					}
					AND = {
						cp_person_is_alive = { person = bey }
						NOT = { has_variable = cp_bey_seen_homesteading }
					}
				}
			}
			random_list = {
				60 = {
					trigger = {
						cp_person_is_alive = { person = layla }
						NOT = { has_variable = cp_layla_seen_homesteading }
					}
					cp_try_fire_law_reaction = { event = cp_layla.2 person = layla }
				}
				30 = {
					trigger = {
						cp_person_is_alive = { person = bey }
						NOT = { has_variable = cp_bey_seen_homesteading }
					}
					cp_try_fire_law_reaction = { event = cp_bey.20 person = bey }
				}
				10 = { }
			}
		}
		# Serfdom restored (regressed from Tenant Farmers or Homesteading).
		else_if = {
			limit = {
				c:EGY ?= this
				ROOT.currently_enacting_law.type ?= law_type:law_serfdom
				OR = {
					AND = {
						cp_person_is_alive = { person = layla }
						NOT = { has_variable = cp_layla_seen_serfdom_restored }
					}
					AND = {
						cp_person_is_alive = { person = bey }
						NOT = { has_variable = cp_bey_seen_serfdom_restored }
					}
				}
			}
			random_list = {
				60 = {
					trigger = {
						cp_person_is_alive = { person = layla }
						NOT = { has_variable = cp_layla_seen_serfdom_restored }
					}
					cp_try_fire_law_reaction = { event = cp_layla.3 person = layla }
				}
				30 = {
					trigger = {
						cp_person_is_alive = { person = bey }
						NOT = { has_variable = cp_bey_seen_serfdom_restored }
					}
					cp_try_fire_law_reaction = { event = cp_bey.21 person = bey }
				}
				10 = { }
			}
		}

		if = {
			limit = {
				c:EGY ?= this
				cp_person_is_alive = { person = bey }
			}
			cp_bey_dispatch_law_enacted = yes
		}
		if = {
			limit = {
				c:EGY ?= this
				cp_person_is_alive = { person = soldier }
			}
			cp_soldier_dispatch_law_enacted = yes
		}
		if = {
			limit = {
				c:EGY ?= this
				ROOT.currently_enacting_law.type ?= law_type:law_women_own_property
				OR = {
					AND = {
						cp_person_is_alive = { person = layla }
						NOT = { has_variable = cp_layla_seen_womens_rights_property }
					}
					AND = {
						cp_person_is_alive = { person = nour }
						NOT = { has_variable = cp_nour_seen_property_rights }
					}
				}
			}
			random_list = {
				45 = {
					trigger = {
						cp_person_is_alive = { person = layla }
						NOT = { has_variable = cp_layla_seen_womens_rights_property }
					}
					cp_try_fire_law_reaction = { event = cp_layla.94 person = layla }
				}
				45 = {
					trigger = {
						cp_person_is_alive = { person = nour }
						NOT = { has_variable = cp_nour_seen_property_rights }
					}
					cp_try_fire_law_reaction = { event = cp_nour.20 person = nour }
				}
				10 = { }
			}
		}

		# Education ladder: public schools + compulsory primary school.
		# Each one-shot per life via cp_seen_* flag (set inside event).
		if = {
			limit = {
				c:EGY ?= this
				ROOT.currently_enacting_law.type ?= law_type:law_public_schools
			}
			random_list = {
				45 = {
					trigger = {
						cp_person_is_alive = { person = layla }
						NOT = { has_variable = cp_layla_seen_public_schools }
					}
					cp_try_fire_law_reaction = { event = cp_layla.100 person = layla }
				}
				45 = {
					trigger = {
						cp_person_is_alive = { person = mina }
						NOT = { has_variable = cp_mina_seen_public_schools }
					}
					cp_try_fire_law_reaction = { event = cp_mina.20 person = mina }
				}
				10 = { }
			}
		}
		if = {
			limit = {
				c:EGY ?= this
				ROOT.currently_enacting_law.type ?= law_type:law_public_health_insurance
			}
			random_list = {
				45 = {
					trigger = {
						cp_person_is_alive = { person = layla }
						NOT = { has_variable = cp_layla_seen_public_health }
					}
					cp_try_fire_law_reaction = { event = cp_layla.110 person = layla }
				}
				45 = {
					trigger = {
						cp_person_is_alive = { person = zaynab }

					}
					cp_try_fire_law_reaction = { event = cp_zaynab.20 person = zaynab }

			}
		}
		if = {
			limit = {
				c:EGY ?= this
				cp_person_is_alive = { person = nour }
			}
			cp_nour_dispatch_law_enacted = yes
		}
		if = {
			limit = {
				c:EGY ?= this
				cp_person_is_alive = { person = mina }
			}
			cp_mina_dispatch_law_enacted = yes
		}
		if = {
			limit = {
				c:EGY ?= this
				cp_person_is_alive = { person = zaynab }
			}
			cp_zaynab_dispatch_law_enacted = yes
		}
		# Layla-only law reactions live in her person-owned memory file.
		if = {
			limit = {
				c:EGY ?= this
				cp_person_is_alive = { person = layla }
			}
			cp_layla_dispatch_law_enacted = yes
		}
		if = {
			limit = {
				c:EGY ?= this
				OR = {
					ROOT.currently_enacting_law.type ?= law_type:law_regulatory_bodies
					ROOT.currently_enacting_law.type ?= law_type:law_worker_protections
				}

				NOT = { has_variable = cp_samier_seen_labor_law }
				NOT = { has_variable = cp_tarek_seen_labor_law }
			}
			random_list = {

# ============================================================================
# War started -- roster war dispatch
# ============================================================================
# on_diplo_play_war_start fires in diplomatic_play scope.
# scope:actor and scope:target are the two lead countries; enter EGY from
# whichever side of the play contains it.
# ============================================================================

on_diplo_play_war_start = {
	on_actions = {
		cp_on_war_started
	}
}

cp_on_war_started = {
	effect = {
		scope:actor ?= {
			if = {
				limit = { THIS = c:EGY }
				cp_on_war_started_for_egy = yes
			}
		}
		scope:target ?= {
			if = {
				limit = { THIS = c:EGY }
				cp_on_war_started_for_egy = yes
			}
		}
	}
}
