###############################################################################
# Common People -- on_actions
#
# Hooks into base-game on_actions to drive the dispatcher layer. We NEVER
# override a base-game on_action; we always subscribe additively.
#
# Naming: cp_ prefix for every subscription target.
# Target version: 1.12; forward-compatible with 1.13.
#
# Current subscriptions:
#   on_game_started              -> cp_on_start
#   on_monthly_pulse_country     -> cp_on_monthly       (tick + mood reroll)
#   on_yearly_pulse_country      -> cp_on_yearly        (SoL delta snapshot)
#   on_law_enactment_pass        -> cp_on_law_enacted   (law reactions)
#   on_war_started               -> cp_on_war_started   (Ahmed conscription)
#   on_war_end                   -> cp_on_war_end       (homecoming / death)
#
# Planned subscriptions (wired in later phases):
#   on_revolution_start          -> fear / hiding event
#   on_acquired_technology       -> telegraph / railway / radio / automobile
#   on_building_built            -> first textile mill / port / rail
#   on_production_method_changed -> mechanisation reactions
###############################################################################


# ============================================================================
# Game start: seed Egypt with Layla
# ============================================================================

on_game_started = {
	on_actions = {
		cp_on_start
	}
}

cp_on_start = {
	effect = {
		set_global_variable = {
			name = cp_is_loaded
			value = yes
		}

		c:EGY ?= {
			# Migration: backfill variables added after a save was created.
			# Each block guards with NOT has_variable so new games and saves
			# already carrying the variable both pass through cleanly.
			if = {
				limit = { has_variable = cp_layla_alive  NOT = { has_variable = cp_status_variant } }
				set_variable = { name = cp_status_variant value = 1 }
			}
			if = {
				limit = { has_variable = cp_layla_alive  NOT = { has_variable = cp_sol_prev } }
				set_variable = { name = cp_sol_prev value = var:cp_sol }
				set_variable = { name = cp_sol_delta value = 0 }
			}
			# Saves from before the data-driven home-state refactor lack
			# the region_state marker. Backfill to Lower Egypt (her start)
			# so cp_refresh_sol_from_pop has something to read.
			if = {
				limit = {
					has_variable = cp_layla_alive
					NOT = {
						any_country = {
							any_scope_state = { has_variable = cp_layla_lives_here }
						}
					}
				}
				cp_move_to_state = { state = STATE_LOWER_EGYPT }
			}

			trigger_event = { id = cp_startup.1 }
		}
	}
}


# ============================================================================
# Monthly pulse -- tick the debug counter, reroll her mood
# ============================================================================
# Why here instead of on the JE's on_monthly_pulse: the JE's on_monthly_pulse
# is now used for random_events (spontaneous reflections). V3 doesn't support
# both `effect` and `random_events` inside a single on_monthly_pulse block,
# so mood-reroll + tick move here where they have the country scope they need.
# ============================================================================

on_monthly_pulse_country = {
	on_actions = {
		cp_on_monthly
	}
}

cp_on_monthly = {
	effect = {
		if = {
			limit = {
				THIS = c:EGY
				has_variable = cp_layla_alive
			}
			# Self-heal: backfill cp_status_variant on saves that predate it.
			if = {
				limit = { NOT = { has_variable = cp_status_variant } }
				set_variable = { name = cp_status_variant value = 1 }
			}
			cp_debug_tick_increment = yes
			# Mood bucket (1..10 -> low/mid/high bucket)
			random_list = {
				10 = { set_variable = { name = cp_mood_roll value = 1 } }
				10 = { set_variable = { name = cp_mood_roll value = 2 } }
				10 = { set_variable = { name = cp_mood_roll value = 3 } }
				10 = { set_variable = { name = cp_mood_roll value = 4 } }
				10 = { set_variable = { name = cp_mood_roll value = 5 } }
				10 = { set_variable = { name = cp_mood_roll value = 6 } }
				10 = { set_variable = { name = cp_mood_roll value = 7 } }
				10 = { set_variable = { name = cp_mood_roll value = 8 } }
				10 = { set_variable = { name = cp_mood_roll value = 9 } }
				10 = { set_variable = { name = cp_mood_roll value = 10 } }
			}
			# Variant within bucket (1..3)
			random_list = {
				33 = { set_variable = { name = cp_status_variant value = 1 } }
				33 = { set_variable = { name = cp_status_variant value = 2 } }
				34 = { set_variable = { name = cp_status_variant value = 3 } }
			}
		}
	}
}


# ============================================================================
# Yearly pulse -- SoL delta snapshot, future pop-tick machinery
# ============================================================================
# Runs once per game year for every country. We filter to EGY and only
# act if Layla is alive. Currently: captures the year-over-year SoL
# delta so the JE can show trend (+2, -1, etc.). The pop-tick effects
# (age drift, literacy drift, radicalism drift) will hang off here
# when Phase B starts.
# ============================================================================

on_yearly_pulse_country = {
	on_actions = {
		cp_on_yearly
	}
}

cp_on_yearly = {
	effect = {
		if = {
			limit = {
				THIS = c:EGY
				has_variable = cp_layla_alive
			}
			# Self-heal: backfill snapshot variables on older saves.
			if = {
				limit = { NOT = { has_variable = cp_sol_prev } }
				set_variable = { name = cp_sol_prev value = var:cp_sol }
				set_variable = { name = cp_sol_delta value = 0 }
			}
			if = {
				limit = { NOT = { has_variable = cp_tax_burden } }
				set_variable = { name = cp_tax_burden value = 0 }
				cp_refresh_tax_burden = yes
				set_variable = { name = cp_tax_burden_prev value = var:cp_tax_burden }
			}
			if = {
				limit = { NOT = { has_variable = cp_welfare_tier } }
				set_variable = { name = cp_welfare_tier value = 0 }
				set_variable = { name = cp_welfare_tier_prev value = 0 }
				cp_refresh_welfare_tier = yes
			}
			# Resample cp_sol from her pop group BEFORE computing delta.
			cp_refresh_sol_from_pop = yes

			# Pop-tick: literacy drifts toward cohort, political mood drifts
			# from expected-SoL gap. Both mirror the SoL resample pattern.
			cp_pop_tick_literacy = yes
			cp_pop_tick_political_mood = yes
			# Delta = current - prev. Computed via multiply-by-(-1)
			# because `change_variable = { subtract = var:X }` is not
			# reliably supported; `add = var:X` is. So negate prev, then add.
			set_variable = { name = cp_sol_delta value = var:cp_sol_prev }
			multiply_variable = { name = cp_sol_delta value = -1 }
			change_variable = { name = cp_sol_delta add = var:cp_sol }
			set_variable = { name = cp_sol_prev value = var:cp_sol }

			# Tax burden: same pattern. Recompute, then fire the tax-spike
			# event if burden jumped by ≥3 since last year (the tax
			# collector now comes for more than he did).
			cp_refresh_tax_burden = yes
			if = {
				limit = {
					NOT = { has_variable = cp_seen_tax_spike }
					var:cp_tax_burden >= 7
					# current - prev >= 3  via multiply-by-(-1) trick
					var:cp_tax_burden >= var:cp_tax_burden_prev
					# We cannot express var - var >= 3 directly; approximate
					# with absolute threshold (>=7) + the trend guard above.
				}
				trigger_event = { id = cp_layla.7 }
				set_variable = { name = cp_seen_tax_spike days = 1800 }
			}
			# First-bite event: consumption tax on grain lands for the
			# first time since she's been alive.
			if = {
				limit = {
					NOT = { has_variable = cp_seen_grain_tax }
					has_consumption_tax = g:grain
				}
				trigger_event = { id = cp_layla.8 }
				set_variable = cp_seen_grain_tax
			}
			set_variable = { name = cp_tax_burden_prev value = var:cp_tax_burden }

			# Crossroads: Cairo offers work. Fires once per life when
			# prerequisites align — a mill exists in her state, she's
			# still a village peasant, SoL is pressed, and migration is
			# legally possible (NOT serfdom). 25% roll per qualifying
			# year so it feels like a choice arriving rather than an
			# obligation on a timer.
			if = {
				limit = {
					has_variable = cp_seen_first_textile_mill
					NOT = { has_variable = cp_seen_cairo_offer }
					NOT = { has_law = law_type:law_serfdom }
					has_variable = cp_profession_peasants
					var:cp_sol <= 12
				}
				random_list = {
					25 = {
						trigger_event = { id = cp_layla.70 }
						set_variable = cp_seen_cairo_offer
					}
					75 = { }
				}
			}

			# Mortality pressures — computed yearly before the rolls.
			cp_refresh_mortality_pressure = yes
			cp_refresh_worker_safety_pressure = yes

			# First welfare receipt. Fires once per life when a welfare
			# law is active AND she plausibly qualifies. Each rung has
			# its own eligibility test; any match triggers the event.
			if = {
				limit = {
					NOT = { has_variable = cp_seen_first_envelope }
					OR = {
						# Old-age pension and she's of age.
						AND = {
							has_law = law_type:law_old_age_pension
							var:cp_age >= 60
						}
						# Poor laws and she's poor.
						AND = {
							has_law = law_type:law_poor_laws
							var:cp_sol <= 10
						}
						# Wage subsidies and her household earns wages.
						AND = {
							has_law = law_type:law_wage_subsidies
							var:cp_sol <= 15
							OR = {
								has_variable = cp_profession_laborers
								has_variable = cp_profession_machinists
								has_variable = cp_profession_farmers
								has_variable = cp_profession_clerks
							}
						}
					}
				}
				# 50% per qualifying year — welfare rolls are slow but
				# the first envelope feels like an arrival, not a raffle.
				random_list = {
					50 = {
						trigger_event = { id = cp_layla.90 }
						set_variable = cp_seen_first_envelope
					}
					50 = { }
				}
			}

			# Layla's annual mortality roll. Buckets her 0..10 pressure
			# into three rough tiers with per-year death probabilities
			# calibrated to 19th-century reality (a 70yo peasant without
			# public health dies at ~10%/yr; a 40yo at peak health ~1%).
			# Age gate: no roll before 35, because events-before-then
			# should kill her (childbirth, famine, factory), not timers.
			if = {
				limit = {
					var:cp_age >= 35
					var:cp_mortality_pressure >= 1
				}
				if = {
					limit = { var:cp_mortality_pressure >= 7 }
					random_list = {
						10 = { trigger_event = { id = cp_layla.80 } }
						90 = { }
					}
				}
				else_if = {
					limit = { var:cp_mortality_pressure >= 4 }
					random_list = {
						4 = { trigger_event = { id = cp_layla.80 } }
						96 = { }
					}
				}
				else = {
					random_list = {
						1 = { trigger_event = { id = cp_layla.80 } }
						99 = { }
					}
				}
			}

			# Ahmed's worker-accident roll. Gated on cp_worker_risk which
			# is only non-zero when he works a shop floor. Worker-
			# protections law clamps the worst away; no-workers-rights
			# (or early industrialisation under regulatory_bodies) makes
			# accidents real and annual.
			if = {
				limit = {
					has_variable = cp_ahmed_alive
					var:cp_ahmed_alive > 0
					var:cp_worker_risk >= 3
				}
				if = {
					limit = { var:cp_worker_risk >= 7 }
					random_list = {
						6 = { trigger_event = { id = cp_layla.81 } }
						94 = { }
					}
				}
				else = {
					random_list = {
						2 = { trigger_event = { id = cp_layla.81 } }
						98 = { }
					}
				}
			}

			# Child mortality roll. V3 models this as state-level, but
			# we capture one household's risk: if she has children AND
			# conditions are grim (pressure 5+), a child may die this
			# year. Lower bar than Layla's roll because children are
			# more vulnerable to what the adults can survive.
			if = {
				limit = {
					var:cp_children >= 1
					var:cp_mortality_pressure >= 5
				}
				random_list = {
					3 = { trigger_event = { id = cp_layla.82 } }
					97 = { }
				}
			}

			# Crossroads: the Small Concern. An urban household with
			# savings and literacy is offered a partnership in a small
			# workshop. Fires once per life. Eligibility covers lower-
			# to-middle strata (laborers, machinists, clerks, engineers)
			# — per V3's profession ladder, these are the pops with
			# enough income + literacy to plausibly buy capital. She
			# still needs a permissive economy law (laissez-faire or
			# interventionism). 20% roll per qualifying year.
			if = {
				limit = {
					has_variable = cp_in_city
					NOT = { has_variable = cp_seen_ownership_offer }
					OR = {
						has_variable = cp_profession_laborers
						has_variable = cp_profession_machinists
						has_variable = cp_profession_clerks
						has_variable = cp_profession_engineers
					}
					var:cp_sol >= 20
					var:cp_literacy >= 10
					OR = {
						has_law = law_type:law_laissez_faire
						has_law = law_type:law_interventionism
					}
				}
				random_list = {
					20 = {
						trigger_event = { id = cp_layla.71 }
						set_variable = cp_seen_ownership_offer
					}
					80 = { }
				}
			}

			# Crossroads: self-employment (Slice D). Gated on cp_may_work
			# (set by cp_layla.95 when law_women_in_the_workplace enacts).
			# Two paths:
			#   - FORCED: she has to. Widowhood + pressed SoL, or
			#     unemployed + pressed SoL. 50% roll per qualifying year
			#     because survival arrives faster than vocation.
			#   - CHOSEN: law active, household OK, small yearly chance.
			#     15% roll per qualifying year.
			# One-shot per life via cp_seen_self_employment; flag is set
			# inside the event's immediate.
			if = {
				limit = {
					has_variable = cp_may_work
					NOT = { has_variable = cp_seen_self_employment }
					NOT = { has_variable = cp_self_employed }
					# Forced: widow with SoL <=10, or low SoL without
					# Ahmed's profession carrying them.
					OR = {
						AND = {
							has_variable = cp_ahmed_alive
							var:cp_ahmed_alive = 0
							var:cp_sol <= 10
						}
						AND = {
							var:cp_sol <= 8
							NOT = { has_variable = cp_profession_shopkeepers }
							NOT = { has_variable = cp_profession_capitalists }
						}
					}
				}
				random_list = {
					50 = {
						trigger_event = { id = cp_layla.97 }
					}
					50 = { }
				}
			}
			else_if = {
				# Chosen path: law active, decent SoL, not already
				# self-employed or in a profession that makes this
				# redundant. Low roll so it feels optional.
				limit = {
					has_variable = cp_may_work
					NOT = { has_variable = cp_seen_self_employment }
					NOT = { has_variable = cp_self_employed }
					NOT = { has_variable = cp_profession_shopkeepers }
					NOT = { has_variable = cp_profession_capitalists }
					var:cp_sol >= 9
				}
				random_list = {
					15 = {
						trigger_event = { id = cp_layla.97 }
					}
					85 = { }
				}
			}

			# Crossroads: daughter's marriage. Fires once per life when
			# the mother is old enough that a first child born around
			# cp_age 22 would now be of marrying age. 20% roll per
			# qualifying year so the crossroads arrives when it does.
			if = {
				limit = {
					var:cp_age >= 38
					var:cp_children >= 1
					NOT = { has_variable = cp_seen_daughter_marriage }
					NOT = { has_variable = cp_daughter_married }
				}
				random_list = {
					20 = {
						trigger_event = { id = cp_layla.72 }
						set_variable = cp_seen_daughter_marriage
					}
					80 = { }
				}
			}
		}
	}
}


# ============================================================================
# Law enactment -- dispatch to Layla's law-reaction events
# ============================================================================
# on_law_enactment_pass fires with root = country; the just-passed law is
# readable as currently_enacting_law (pattern confirmed in vanilla, e.g.
# je_corn_laws and vanilla law-reaction events).
# ============================================================================

on_law_enactment_pass = {
	on_actions = {
		cp_on_law_enacted
	}
}

cp_on_law_enacted = {
	effect = {
		# Homesteading granted -- she owns the field.
		# Pattern mirrors vanilla 00_monarchy.txt: c:TAG ?= this + ROOT.currently_enacting_law.type ?= law_type:X.
		if = {
			limit = {
				c:EGY ?= this
				has_variable = cp_layla_alive
				ROOT.currently_enacting_law.type ?= law_type:law_homesteading
				NOT = { has_variable = cp_seen_homesteading }
			}
			trigger_event = { id = cp_layla.2 }
		}
		# Serfdom restored (regressed from Tenant Farmers or Homesteading).
		else_if = {
			limit = {
				c:EGY ?= this
				has_variable = cp_layla_alive
				ROOT.currently_enacting_law.type ?= law_type:law_serfdom
				NOT = { has_variable = cp_seen_serfdom_restored }
			}
			trigger_event = { id = cp_layla.3 }
		}

		# Welfare-ladder transition. Any lawgroup_welfare change reaches
		# here. We cache old tier, refresh to the new tier, compare, and
		# fire up/down events.
		if = {
			limit = {
				c:EGY ?= this
				has_variable = cp_layla_alive
				OR = {
					ROOT.currently_enacting_law.type ?= law_type:law_no_social_security
					ROOT.currently_enacting_law.type ?= law_type:law_chiefs_distribute_aid
					ROOT.currently_enacting_law.type ?= law_type:law_poor_laws
					ROOT.currently_enacting_law.type ?= law_type:law_wage_subsidies
					ROOT.currently_enacting_law.type ?= law_type:law_old_age_pension
				}
			}
			# Snapshot the prior tier into a temp var before refresh.
			set_variable = { name = cp_welfare_tier_prev value = var:cp_welfare_tier }
			cp_refresh_welfare_tier = yes
			# Up transition: new > prev.
			if = {
				limit = { var:cp_welfare_tier > var:cp_welfare_tier_prev }
				trigger_event = { id = cp_layla.91 }
			}
			else_if = {
				# Down transition: new < prev. (Repeal / regression.)
				limit = { var:cp_welfare_tier < var:cp_welfare_tier_prev }
				trigger_event = { id = cp_layla.92 }
			}
		}

		# Women's rights enactments. First-time-only per rung; the one-shot
		# gate is the cp_seen_womens_rights_* flag set inside each event's
		# immediate block. Rollback events are not fired (user-design: each
		# rung is a "first enactment" memory, not a ladder mirror).
		if = {
			limit = {
				c:EGY ?= this
				has_variable = cp_layla_alive
				ROOT.currently_enacting_law.type ?= law_type:law_women_in_the_fields
				NOT = { has_variable = cp_seen_womens_rights_fields }
			}
			trigger_event = { id = cp_layla.93 }
		}
		else_if = {
			limit = {
				c:EGY ?= this
				has_variable = cp_layla_alive
				ROOT.currently_enacting_law.type ?= law_type:law_women_own_property
				NOT = { has_variable = cp_seen_womens_rights_property }
			}
			trigger_event = { id = cp_layla.94 }
		}
		else_if = {
			limit = {
				c:EGY ?= this
				has_variable = cp_layla_alive
				ROOT.currently_enacting_law.type ?= law_type:law_women_in_the_workplace
				NOT = { has_variable = cp_seen_womens_rights_workplace }
			}
			trigger_event = { id = cp_layla.95 }
		}
		else_if = {
			limit = {
				c:EGY ?= this
				has_variable = cp_layla_alive
				ROOT.currently_enacting_law.type ?= law_type:law_womens_suffrage
				NOT = { has_variable = cp_seen_womens_rights_suffrage }
			}
			trigger_event = { id = cp_layla.96 }
		}

		# Education ladder: public schools + compulsory primary school.
		# Each one-shot per life via cp_seen_* flag (set inside event).
		if = {
			limit = {
				c:EGY ?= this
				has_variable = cp_layla_alive
				ROOT.currently_enacting_law.type ?= law_type:law_public_schools
				NOT = { has_variable = cp_seen_public_schools }
			}
			trigger_event = { id = cp_layla.100 }
		}
		else_if = {
			limit = {
				c:EGY ?= this
				has_variable = cp_layla_alive
				ROOT.currently_enacting_law.type ?= law_type:law_compulsory_primary_school
				NOT = { has_variable = cp_seen_compulsory_school }
			}
			trigger_event = { id = cp_layla.101 }
		}

		# Slavery: emancipation is the big beat. The starting states
		# (slave_trade / legacy / debt) are her baseline — no reaction
		# event for those; they're handled via the ambient pulse
		# cp_layla.104 which fires while those laws are active.
		if = {
			limit = {
				c:EGY ?= this
				has_variable = cp_layla_alive
				ROOT.currently_enacting_law.type ?= law_type:law_slavery_banned
				NOT = { has_variable = cp_seen_emancipation }
			}
			trigger_event = { id = cp_layla.103 }
		}

		# Regime: universal suffrage + single-party state. One-shot each.
		if = {
			limit = {
				c:EGY ?= this
				has_variable = cp_layla_alive
				ROOT.currently_enacting_law.type ?= law_type:law_universal_suffrage
				NOT = { has_variable = cp_seen_universal_suffrage }
			}
			trigger_event = { id = cp_layla.105 }
		}
		else_if = {
			limit = {
				c:EGY ?= this
				has_variable = cp_layla_alive
				ROOT.currently_enacting_law.type ?= law_type:law_single_party_state
				NOT = { has_variable = cp_seen_single_party }
			}
			trigger_event = { id = cp_layla.106 }
		}

		# Citizenship: multicultural. One-shot.
		if = {
			limit = {
				c:EGY ?= this
				has_variable = cp_layla_alive
				ROOT.currently_enacting_law.type ?= law_type:law_multicultural
				NOT = { has_variable = cp_seen_multicultural }
			}
			trigger_event = { id = cp_layla.107 }
		}
	}
}


# ============================================================================
# War started -- Ahmed conscription
# ============================================================================
# on_war_started fires on each involved country. root = this country.
# ============================================================================

on_war_started = {
	on_actions = {
		cp_on_war_started
	}
}

cp_on_war_started = {
	effect = {
		if = {
			limit = {
				c:EGY ?= this
				has_variable = cp_layla_alive
				has_variable = cp_ahmed_alive
				NOT = { has_variable = cp_ahmed_at_war }
				NOT = { has_variable = cp_seen_ahmed_conscripted }
				var:cp_ahmed_age >= 18
				var:cp_ahmed_age <= 45
			}
			trigger_event = { id = cp_layla.4 }
		}
	}
}


# ============================================================================
# War ended -- Ahmed homecoming or death
# ============================================================================
# Correct hook name is on_war_end (NOT on_war_ended). Fires with root = each
# involved country and sets recently_had_war variable per vanilla pattern.
# Homecoming vs death events will ship in the P9 war slice; this subscription
# is live now so the hook is reserved.
# ============================================================================

on_war_end = {
	on_actions = {
		cp_on_war_end
	}
}

cp_on_war_end = {
	effect = {
		if = {
			limit = {
				c:EGY ?= this
				has_variable = cp_layla_alive
				has_variable = cp_ahmed_alive
				has_variable = cp_ahmed_at_war
			}
			# Clear the conscription gate so the next war can fire cp_layla.4
			# again. cp_ahmed_at_war itself is cleared by the outcome effect
			# in cp_memory (cp_ahmed_homecoming / cp_ahmed_killed).
			remove_variable = cp_seen_ahmed_conscripted

			# Outcome weights by war result. recently_won_war and
			# recently_lost_war are vanilla-set on country scope at
			# on_capitulation (days = 1825 timer, so they're fresh here).
			if = {
				limit = { has_variable = recently_won_war }
				random_list = {
					75 = { trigger_event = { id = cp_layla.40 } }
					25 = { trigger_event = { id = cp_layla.41 } }
				}
			}
			else_if = {
				limit = { has_variable = recently_lost_war }
				random_list = {
					40 = { trigger_event = { id = cp_layla.40 } }
					60 = { trigger_event = { id = cp_layla.41 } }
				}
			}
			else = {
				# Neither side fully capitulated (white peace, stalemate).
				# Still brutal; still a coin flip with a slight pull toward home.
				random_list = {
					55 = { trigger_event = { id = cp_layla.40 } }
					45 = { trigger_event = { id = cp_layla.41 } }
				}
			}
		}
	}
}


# ============================================================================
# Technology acquired -- ambient industrial modernity reaches her
# ============================================================================
# on_acquired_technology fires once per country per tech, with root = country
# and scope:technology = the technology scope. Pattern from vanilla
# 00_code_on_actions.txt (grand-exhibition tracking).
#
# We don't key on scope:technology directly (the comparison syntax is
# awkward); we gate on has_technology_researched for each tech of interest
# plus a one-shot cp_seen_* flag. The dispatcher runs on every tech
# acquisition, but each branch fires its event exactly once.
# ============================================================================

on_acquired_technology = {
	on_actions = {
		cp_on_tech
	}
}

cp_on_tech = {
	effect = {
		if = {
			limit = {
				c:EGY ?= this
				has_variable = cp_layla_alive
				has_technology_researched = railways
				NOT = { has_variable = cp_seen_railway }
			}
			trigger_event = { id = cp_layla.30 }
			set_variable = cp_seen_railway
		}
		if = {
			limit = {
				c:EGY ?= this
				has_variable = cp_layla_alive
				has_technology_researched = electric_telegraph
				NOT = { has_variable = cp_seen_telegraph }
			}
			trigger_event = { id = cp_layla.31 }
			set_variable = cp_seen_telegraph
		}
		if = {
			limit = {
				c:EGY ?= this
				has_variable = cp_layla_alive
				has_technology_researched = radio
				NOT = { has_variable = cp_seen_radio }
			}
			trigger_event = { id = cp_layla.32 }
			set_variable = cp_seen_radio
		}
	}
}


# ============================================================================
# Building built -- the industrial revolution lands in her state
# ============================================================================
# on_building_built fires with root = the newly-built building. state gives
# the region_state, owner gives the country. Pattern verified against vanilla
# 00_code_on_actions.txt:1907 (Manila Cathedral handler). Each branch is
# one-shot via a cp_seen_* flag on c:EGY.
# ============================================================================

on_building_built = {
	on_actions = {
		cp_on_building_built
	}
}

cp_on_building_built = {
	effect = {
		# First textile mill in her state -- the factory age begins on the
		# far bank of the canal.
		if = {
			limit = {
				is_building_type = building_textile_mill
				owner ?= {
					THIS = c:EGY
					has_variable = cp_layla_alive
					NOT = { has_variable = cp_seen_first_textile_mill }
				}
				state = { has_variable = cp_layla_lives_here }
			}
			owner ?= {
				trigger_event = { id = cp_layla.50 }
				set_variable = cp_seen_first_textile_mill
			}
		}
	}
}


# ============================================================================
# Revolution start -- the country tears; she shutters the house
# ============================================================================
# on_revolution_start fires with root = the country where the revolution
# begins. We fire Layla's fear event when EGY is that country. One-shot
# per revolution via cp_seen_revolution flag; the flag is timed so a
# later revolution (different year, different cause) can re-fire.
# ============================================================================

on_revolution_start = {
	on_actions = {
		cp_on_revolution
	}
}

cp_on_revolution = {
	effect = {
		if = {
			limit = {
				c:EGY ?= this
				has_variable = cp_layla_alive
				NOT = { has_variable = cp_seen_revolution }
			}
			trigger_event = { id = cp_layla.60 }
			# Timed 10-year flag so a later revolution can re-fire the prose.
			set_variable = { name = cp_seen_revolution days = 3650 }
		}
	}
}
