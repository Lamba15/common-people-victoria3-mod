###############################################################################
# Common People -- shared person-parameterised sensors (r5)
#
# Sensors answer yes/no questions about the world or about a person's state.
# The shared ones take $person$ (and sometimes $culture$/$religion$/$poptype$)
# so they generalise across persons; per-person sensors in
# cp_<person>_sensors.txt wrap them with literals.
#
# Sensors never mutate state. They only READ.
###############################################################################

	has_global_variable = cp_person_$person$_alive
	has_variable = cp_$person$_alive
	var:cp_$person$_alive > 0

# ============================================================================
# Home-state sensors
# ============================================================================
# A person can only be directly affected by local buildings in the state that
# carries cp_<person>_lives_here. Country-wide technologies and laws can still
# matter nationally, but building events must pass through these local checks.

cp_building_is_in_person_home_state = {
	state = { has_variable = cp_$person$_lives_here }
}

cp_building_affects_person_workplace = {
	cp_building_is_in_person_home_state = { person = $person$ }
	OR = {
		AND = {
			owner ?= { var:cp_$person$_workplace_agrarian > 0 }
			OR = {
				is_building_group = bg_agriculture
				is_building_group = bg_plantations
				is_building_type = building_food_industry
			}
		}
		AND = {
			owner ?= { var:cp_$person$_workplace_manufacturing > 0 }
			is_building_group = bg_manufacturing
		}
		AND = {
			owner ?= { var:cp_$person$_workplace_machine > 0 }
			OR = {
				is_building_type = building_tooling_workshop
				is_building_type = building_steel_mill
				is_building_type = building_motor_industry
			}
		}
		AND = {
			owner ?= { var:cp_$person$_workplace_rail > 0 }
			is_building_type = building_railway
		}
		AND = {
			owner ?= { var:cp_$person$_workplace_port > 0 }
			OR = {
				is_building_type = building_port
				is_building_type = building_trade_center
			}
		}
		AND = {
			owner ?= { var:cp_$person$_workplace_public_service > 0 }
			OR = {
				is_building_type = building_government_administration
				is_building_type = building_university
			}
		}
		AND = {
			owner ?= { var:cp_$person$_workplace_electric > 0 }
			OR = {
				is_building_type = building_electrics_industry
				is_building_type = building_power_plant
			}
		}
		AND = {
			owner ?= { var:cp_$person$_workplace_urban > 0 }
			OR = {
				is_building_type = building_construction_sector
				is_building_type = building_trade_center
				is_building_type = building_urban_center
			}
		}
		AND = {
			owner ?= { var:cp_$person$_workplace_military > 0 }
			OR = {
				is_building_type = building_arms_industry
				is_building_type = building_munition_plant
				is_building_type = building_barrack
			}
		}
	}
}

# ============================================================================
# Personal-SoL band sensors
# ============================================================================
# V3's canonical SoL tiers (per r4 §0.5): Destitute 1-4, Struggling 5-9,
# Impoverished 10-14, Middling 15-19, Secure 20-24, Prosperous 25-29,
# Affluent 30-39, Wealthy 40-49, Lavish 50-59, Opulent 60+.

cp_person_destitute      = { var:cp_$person$_sol <= 4 }
cp_person_struggling     = { var:cp_$person$_sol >= 5   var:cp_$person$_sol <= 9 }
cp_person_impoverished   = { var:cp_$person$_sol >= 10  var:cp_$person$_sol <= 14 }
cp_person_middling       = { var:cp_$person$_sol >= 15  var:cp_$person$_sol <= 19 }
cp_person_secure         = { var:cp_$person$_sol >= 20  var:cp_$person$_sol <= 24 }
cp_person_prosperous     = { var:cp_$person$_sol >= 25  var:cp_$person$_sol <= 29 }
cp_person_affluent       = { var:cp_$person$_sol >= 30  var:cp_$person$_sol <= 39 }
cp_person_wealthy_plus   = { var:cp_$person$_sol >= 40 }

# ============================================================================
# Country/time context sensors
# ============================================================================
# These keep person-selection weights data-driven. They are intentionally broad:
# the monthly router should decide whose life is most legible in the current
# world, while each person's own ambient pool still decides which beat can fire.

cp_era_before_1850 = {
	game_date < 1850.1.1
}

cp_era_1850_to_1880 = {
	game_date >= 1850.1.1
	game_date < 1880.1.1
}

cp_era_after_1880 = {
	game_date >= 1880.1.1
}

cp_country_has_manufacturing = {
	any_scope_state = {
		is_incorporated = yes
		any_scope_building = {
			is_building_group = bg_manufacturing
		}
	}
}

cp_country_has_factory_migration = {
	OR = {
		AND = {
			cp_country_has_manufacturing = yes
			cp_country_has_urban_growth = yes
		}
		AND = {
			cp_country_has_manufacturing = yes
			cp_era_1850_to_1880 = yes
		}
		AND = {
			cp_country_has_manufacturing = yes
			cp_era_after_1880 = yes
		}
		AND = {
			cp_country_has_machine_industry = yes
			cp_country_has_urban_growth = yes
		}
	}
}

cp_country_has_machine_industry = {
	OR = {
		has_technology_researched = mechanical_tools
		has_technology_researched = rotary_valve_engine

			is_incorporated = yes
