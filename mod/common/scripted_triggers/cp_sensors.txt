###############################################################################
# Common People -- sensors
#
# The only layer that asks questions of the V3 world. Events consume sensors;
# sensors never mutate state. Each sensor reads its answer from the live game
# state (market, pops, laws, institutions, decrees, buildings, techs) and
# returns a boolean.
#
# Conventions:
#   - Every sensor is named cp_sensor_<short_question>.
#   - Most run in country scope (root = c:EGY inside a country_event).
#   - State-scoped sensors wrap their state lookup explicitly.
#
# Consumption basket: organized by popneed_* (V3's pop consumption groups).
# Rationale: Layla is statistically a peasant pop -- what she cares about is
# what pops in her wealth tier actually buy. See documentation/research/
# engine/buy_packages.md and /media/.../game/common/pop_needs/00_pop_needs.txt.
#
# At wealth_5 (her starting tier) she buys: simple_clothing, crude_items,
# basic_food, heating, intoxicants. wealth_6 adds stimulants. wealth_10 adds
# services, household_items, standard_clothing, free_movement. Higher tiers
# unlock luxury_food, luxury_items, luxury_drinks, communication, leisure.
#
# Reference: documentation/research/engine/ for trigger names.
###############################################################################

# ============================================================================
# popneed_basic_food -- the plate in front of her every day
# Maps: grain (bread), fish (Friday), meat (rare), fruit, groceries (urban)
# ============================================================================

cp_sensor_grain_is_expensive = {
	c:EGY.market = { mg:grain = { market_goods_pricier > 0.20 } }
}

cp_sensor_grain_is_cheap = {
	c:EGY.market = { mg:grain = { market_goods_cheaper >= 0.10 } }
}

cp_sensor_grain_is_scarce = {
	c:EGY.market = { mg:grain = { market_goods_shortage_ratio > 0.10 } }
}

cp_sensor_fish_is_affordable = {
	c:EGY.market = { mg:fish = { market_goods_buy_orders > 50 } }
	c:EGY.market = { mg:fish = { NOT = { market_goods_pricier > 0.20 } } }
}

cp_sensor_fish_is_scarce = {
	c:EGY.market = { mg:fish = { market_goods_shortage_ratio > 0.10 } }
}

cp_sensor_meat_is_a_rare_treat = {
	c:EGY.market = { mg:meat = { market_goods_cheaper >= 0.15 } }
}

cp_sensor_meat_is_too_dear = {

}

cp_sensor_fruit_is_affordable = {
	c:EGY.market = { mg:fruit = { market_goods_buy_orders > 50 } }
