###############################################################################
# Common People -- shared roster action-button availability
#
# These triggers mirror the player-click action pools. Keeping the availability
# checks separate lets the shared JE disable a person's button when every
# currently eligible scene has already been seen.
###############################################################################

cp_shared_action_layla_available = {
	cp_person_is_alive = { person = layla }
	OR = {
		AND = { has_variable = cp_layla_ahmed_at_war NOT = { has_variable = cp_layla_seen_letter } }
		AND = { has_variable = cp_layla_owns_land NOT = { has_variable = cp_layla_seen_walking_field } }
		AND = { has_variable = cp_layla_under_serfdom NOT = { has_variable = cp_layla_seen_under_beys_eye } }
		NOT = { has_variable = cp_layla_seen_no_wind }
		NOT = { has_variable = cp_layla_seen_neighbours_boy }
		NOT = { has_variable = cp_layla_seen_mothers_hand }
		NOT = { has_variable = cp_layla_seen_hen }
		NOT = { has_variable = cp_layla_seen_scale }
		NOT = { has_variable = cp_layla_seen_river_low }
		NOT = { has_variable = cp_layla_seen_old_woman }
		NOT = { has_variable = cp_layla_seen_bread }
		NOT = { has_variable = cp_layla_seen_dust }
		NOT = { has_variable = cp_layla_seen_brother }
		NOT = { has_variable = cp_layla_seen_mule }
		AND = { has_variable = cp_layla_may_vote NOT = { has_variable = cp_layla_has_voted } }
		var:cp_layla_sol <= 10
		has_variable = cp_layla_owns_land
		AND = { has_variable = cp_layla_daughter_at_school NOT = { has_variable = cp_layla_seen_daughter_reads } }
		AND = {
			OR = {
				has_law = law_type:law_debt_slavery
				has_law = law_type:law_legacy_slavery
				has_law = law_type:law_slave_trade
			}
			NOT = { has_variable = cp_layla_seen_slaves_pulse }
		}
		NOT = { has_variable = cp_layla_seen_ramadan }
		NOT = { has_variable = cp_layla_seen_flood }
		NOT = { has_variable = cp_layla_seen_hajj }
		NOT = { has_variable = cp_layla_seen_midwife_lamp }
		NOT = { has_variable = cp_layla_seen_date_harvest }
		NOT = { has_variable = cp_layla_seen_coffee_arrival }
		NOT = { has_variable = cp_layla_seen_sugar_cheap }
		NOT = { has_variable = cp_layla_seen_cloth_affordable }
		AND = {
			OR = {
				has_variable = cp_layla_profession_laborers
				has_variable = cp_layla_ahmed_profession_laborer
			}
			OR = {
				has_law = law_type:law_right_to_associate
				has_law = law_type:law_corporatized_unions
				has_law = law_type:law_factory_councils
			}
			NOT = { has_variable = cp_layla_seen_strike }
		}
		NOT = { has_variable = cp_layla_seen_imam_visit }
		NOT = { has_variable = cp_layla_seen_small_victory }
		NOT = { has_variable = cp_layla_seen_mariam_letter }
	}
}
