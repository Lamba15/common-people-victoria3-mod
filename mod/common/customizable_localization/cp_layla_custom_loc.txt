###############################################################################
# Common People -- customizable localization
#
# V3's customizable_localization lets a single loc key resolve to different
# text based on conditions. We use it to make Layla's JE "reason" prose
# reactive to her current state -- as her profession, home state, SoL, and
# ownership change, the prose describing her life changes with her.
#
# Pattern (vanilla, see 00_characters.txt:stepping_down_reason):
#   <name> = {
#     type = <scope type>
#     text = {
#       trigger = { <condition> }
#       localization_key = <loc key to render>
#     }
#     # More specific triggers first; the first passing branch wins.
#   }
#
# Each block below drives one segment of cp_je_layla_reason. The loc keys
# the branches point to live in cp_layla_l_english.yml.
###############################################################################


# ============================================================================
# SoL DELTA LABEL -- color-coded year-over-year change, hidden when zero
#
# The actual numeric delta is cp_layla_sol_delta. This customizable_loc picks
# one of three render variants: green up, red down, or empty (no change).
# Updated once per year in cp_on_yearly.
# ============================================================================
cp_layla_sol_delta_label = {
	type = country

	text = {
		trigger = { has_variable = cp_layla_sol_delta  var:cp_layla_sol_delta > 0 }
		localization_key = cp_layla_sol_delta_up
	}
	text = {
		trigger = { has_variable = cp_layla_sol_delta  var:cp_layla_sol_delta < 0 }
		localization_key = cp_layla_sol_delta_down
	}
	text = {
		localization_key = cp_layla_sol_delta_flat
	}
}

# ============================================================================
# HOME STATE LABEL -- the V3 state she lives in, pulled from vanilla
#
# Her residence is marked by var:cp_layla_lives_here on the region_state
# she inhabits. The label scope-jumps to that state and renders its
# vanilla name via GetName, so the string localises and renames
# automatically for any state in the game.
# ============================================================================
cp_layla_home_state_label = {
	type = country

	text = {
		trigger = {
			any_country = {
				any_scope_state = { has_variable = cp_layla_lives_here }
			}
		}
		localization_key = cp_layla_home_state_dynamic
	}

	# Fallback for saves that somehow lost the marker.
	text = {
		localization_key = STATE_LOWER_EGYPT
	}
}

# ============================================================================
# LITERACY LABEL -- binary, matching how V3 treats an individual
#
# V3 pops carry literacy as a percentage (0..100) = "approximately what
# percent of individuals in that pop can read and write". For a single
# persona like Layla, the percentage doesn't map cleanly — she either
# can or cannot read. We gate at 10% and render a binary word.
# ============================================================================
cp_layla_literacy_label = {
	type = country

	text = {
		trigger = { has_variable = cp_layla_literacy  var:cp_layla_literacy >= 10 }
		localization_key = cp_layla_literacy_label_literate
	}
	text = {
		localization_key = cp_layla_literacy_label_illiterate
	}
}


# ============================================================================
# POLITICAL STANCE -- V3's canonical three states
#
# Per V3 wiki: "Individuals have one of three stances towards the country
# and its government: loyalist, neutral, or radical." The two numeric
# variables cp_layla_loyalist and cp_layla_radical are mutually exclusive (V3 rule);
# a non-zero value in either places her in that state.
# ============================================================================
cp_layla_stance_label = {
	type = country

	text = {
		trigger = { has_variable = cp_layla_loyalist  var:cp_layla_loyalist > 0 }
		localization_key = cp_layla_stance_loyalist
	}
	text = {
		trigger = { has_variable = cp_layla_radical  var:cp_layla_radical > 0 }
		localization_key = cp_layla_stance_radical
	}
	text = {
		localization_key = cp_layla_stance_neutral
	}
}


# ============================================================================
# HOUSING -- where and how she lives
# ============================================================================
cp_layla_housing_desc = {
	type = country

	# Urban, upper-strata professions
	text = {
		trigger = {
			has_variable = cp_layla_in_cairo
			OR = { has_variable = cp_layla_profession_shopkeepers  has_variable = cp_layla_profession_clerks }
		}
		localization_key = cp_layla_housing_cairo_middling
	}
	text = {
		trigger = {
			has_variable = cp_layla_in_alexandria
			OR = { has_variable = cp_layla_profession_shopkeepers  has_variable = cp_layla_profession_clerks }
		}
		localization_key = cp_layla_housing_alexandria_middling
	}

	# Urban labourer
	text = {
		trigger = { has_variable = cp_layla_in_cairo  has_variable = cp_layla_profession_laborers }
		localization_key = cp_layla_housing_cairo_tenement
	}
	text = {
		trigger = { has_variable = cp_layla_in_alexandria  has_variable = cp_layla_profession_laborers }
		localization_key = cp_layla_housing_alexandria_tenement
	}

	# Rural variants
	text = {
		trigger = { has_variable = cp_layla_owns_land }
		localization_key = cp_layla_housing_homesteader
	}
	text = {
		trigger = { has_variable = cp_layla_under_serfdom }
		localization_key = cp_layla_housing_serf_restored
	}

	# Default: tenant's wife in a Delta village (the starting state)
	text = {
		localization_key = cp_layla_housing_tenant_default
	}
}
