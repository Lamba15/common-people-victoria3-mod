# Samier is Person 4: a Misri factory worker whose life enters when Egypt's
# industrialization has produced factories and unprotected labor.

namespace = cp_samier


# cp_samier.1 -- hidden setup
cp_samier.1 = {
	type = country_event
	hidden = yes

	trigger = {
		c:EGY ?= this
		has_global_variable = cp_person_samier_alive
		NOT = { has_variable = cp_samier_alive }
	}

	immediate = {
		set_variable = { name = cp_samier_alive value = 1 }
		set_variable = { name = cp_samier_age value = 19 }
		set_variable = { name = cp_samier_hope value = 6 }
		set_variable = { name = cp_samier_exhaustion value = 4 }
		set_variable = { name = cp_samier_sol value = 8 }
		set_variable = { name = cp_samier_literacy value = 3 }
		set_variable = { name = cp_samier_radical value = 1 }
		set_variable = { name = cp_samier_loyalist value = 0 }
		set_variable = { name = cp_samier_anger value = 4 }
		set_variable = cp_samier_profession_laborers
		set_variable = { name = cp_samier_w_labor_laws value = 10 }
		set_variable = { name = cp_samier_w_revolution value = 8 }
		set_variable = { name = cp_samier_w_economy value = 8 }
		set_variable = { name = cp_samier_w_education value = 6 }
		set_variable = { name = cp_samier_w_religion value = 3 }

		cp_set_person_home_state_from_spawn_marker_or_default = { person = samier country = EGY state = STATE_LOWER_EGYPT }
		cp_set_person_home_place = { person = samier place = bulaq }
		cp_set_person_workplace_profile = { person = samier profile = manufacturing }
		cp_samier_refresh_sol = yes

		if = {
			limit = {
				NOT = { has_variable = cp_startup_conditional_scan }
			}
			random_list = {
				65 = { cp_try_fire_world_response = { event = cp_samier.10 person = samier } }
				35 = { }
			}
		}
	}
}


# cp_samier.10 -- "The Gate of Smoke"
cp_samier.10 = {
	type = country_event

	title = cp_samier.10.t
	desc = cp_samier.10.d
	flavor = cp_samier.10.f

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	event_image = {
		texture = "gfx/event_pictures/cp_samier_10_the_gate_of_smoke_v01.dds"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/MUSIC/Stingers/events/tranquil"

	duration = 3

	trigger = {
		has_variable = cp_samier_alive
		var:cp_samier_alive > 0
		NOT = { has_variable = cp_samier_seen_intro }
	}

	immediate = {
		cp_samier_mark_intro = yes
	}

	option = {
		name = cp_samier.10.a
		default_option = yes
	}
}
