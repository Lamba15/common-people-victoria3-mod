# cp_debug.8 -- Force-run the yearly delta computation. Uses the same
# math as cp_on_yearly but via a temp-variable + multiply-by-(-1) pattern,
# which is the safer syntax in case `subtract = var:X` silently no-ops.
cp_debug.8 = {
	type = country_event
	hidden = yes
	orphan = yes

	immediate = {
		c:EGY ?= {
			# delta = cp_sol - cp_sol_prev  (via multiply-by-(-1))
			set_variable = { name = cp_sol_delta value = var:cp_sol_prev }
			multiply_variable = { name = cp_sol_delta value = -1 }
			change_variable = { name = cp_sol_delta add = var:cp_sol }
			set_variable = { name = cp_sol_prev value = var:cp_sol }
		}
	}
}

# cp_debug.11 -- Read country-level average_sol directly into cp_sol.
# If this works, the readout will show Egypt's actual average SoL (probably
# ~8-10 at 1836). If it fails or returns 0, we know the direct-read path
# does not work for this value.
cp_debug.11 = {
	type = country_event
	hidden = yes
	orphan = yes

	immediate = {
		c:EGY ?= {
			# Pre-write a marker so we can tell if the next line fires.
			set_variable = { name = cp_sol value = 88 }
			# Attempt the read. If it lands, cp_sol becomes the real value.
			set_variable = { name = cp_sol value = average_sol }
		}
	}
}

# cp_debug.18 -- Read state-region average SoL via the Belle Epoque pattern:
# enter state-region -> random_scope_state -> save average_sol as a temp.
# Vanilla source: 00_belle_epoque.txt line 74-78 (random_scope_state inside
# s:STATE_X ?= {}) plus line 71/85 (average_sol readable on region_state).
# Markers: 66 = script never entered the save; 0 = filter found no state;
# real number (5-10ish) = success.
cp_debug.18 = {
	type = country_event
	hidden = yes
	orphan = yes

	immediate = {
		c:EGY ?= {
			set_variable = { name = cp_sol value = 66 }
			s:STATE_LOWER_EGYPT ?= {
				random_scope_state = {
					limit = { owner = c:EGY }
					save_temporary_scope_value_as = {
						name = tmp_state_sol
						value = average_sol
					}
				}
			}
			set_variable = { name = cp_sol value = scope:tmp_state_sol }
		}
	}
}

# cp_debug.19 -- Pop-group SoL via the same scope path: enter region_state
# via random_scope_state, then ordered_scope_pop picks the LARGEST matching
# pop (order_by = total_size), saves its standard_of_living.
# Markers: 66 = script never entered the save chain; 0 = no matching pop
# found OR SoL read returned 0; a real SoL number = success.
cp_debug.19 = {
	type = country_event
	hidden = yes
	orphan = yes

	immediate = {
		c:EGY ?= {
			set_variable = { name = cp_sol value = 66 }
			s:STATE_LOWER_EGYPT ?= {
				random_scope_state = {
					limit = { owner = c:EGY }
					ordered_scope_pop = {
						limit = {
							culture = cu:misri
							religion = rel:sunni
							is_pop_type = peasants
						}
						order_by = total_pops
						position = 0
						check_range_bounds = no
						save_temporary_scope_value_as = {
							name = tmp_pop_sol
							value = standard_of_living
						}
					}
				}
			}
			set_variable = { name = cp_sol value = scope:tmp_pop_sol }
		}
	}
}

# cp_debug.23 -- Stage-by-stage probe for cp_refresh_sol_from_pop.
# Writes a marker to cp_sol at each stage so we can see how far it
# gets if it silently fails somewhere.
#   11 = effect entered at all
#   22 = home-state marker found by every_scope_state
#   33 = diaspora fallback entered (local cohort missed)
#   44 = diaspora cohort saved (tmp_cohort_sol exists)
#   real SoL = fully committed value
cp_debug.23 = {
	type = country_event
	hidden = yes
	orphan = yes

	immediate = {
		c:EGY ?= {
			set_variable = { name = cp_sol value = 11 }

			# Stage A: home-state marker exists?
			every_scope_state = {
				limit = { has_variable = cp_layla_lives_here }
				root = { set_variable = { name = cp_sol value = 22 } }
			}

			# Stage B: diaspora fallback path
			s:STATE_LOWER_EGYPT ?= {
				random_scope_state = {
					limit = { owner = c:EGY }
					c:EGY = { set_variable = { name = cp_sol value = 33 } }
					cp_read_cohort_only = yes
				}
			}
			if = {
				limit = { exists = scope:tmp_cohort_sol }
				set_variable = { name = cp_sol value = 44 }
				# Overwrite 44 with the actual cohort value so we see it
				set_variable = { name = cp_sol value = scope:tmp_cohort_sol }
			}
		}
	}
}
