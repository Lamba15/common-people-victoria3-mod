###############################################################################
# Common People -- debug events
#
# Console-only test events. Each has a single, isolated job.
# Fire with `event cp_debug.N` in the in-game console.
#
# Why split them: we want to be able to verify each layer independently.
# If cp_debug.2 fires "The Deed" but cp_debug.5 does NOT, the dispatcher is
# broken. If BOTH fail, the event itself is broken.
#
#   cp_debug.1   -- Confirm console event firing works (visible, no data).
#   cp_debug.2   -- Directly fire cp_layla.2 (The Deed). Proves the event
#                   renders. BYPASSES the on_law_enactment_pass dispatcher.
#   cp_debug.3   -- Directly fire cp_layla.3 (The Bey Returns).
#   cp_debug.4   -- Directly fire cp_layla.4 (He Marches).
#   cp_debug.5   -- Activate Homesteading law (tests the dispatcher path).
#   cp_debug.6   -- Age Layla by 10 years (hidden; verify with JE panel).
###############################################################################

namespace = cp_debug

# cp_debug.1 -- "I'm alive" visible event. No variable substitution -- event
# text in V3 does not cleanly support Var() substitution. Values are read
# from the journal entry body (which DOES render substitutions in reason).
cp_debug.1 = {
	type = country_event
	orphan = yes
	placement = root

	title = cp_debug.1.t
	desc = cp_debug.1.d
	flavor = cp_debug.1.f

	duration = 3

	option = {
		name = cp_debug.1.a
		default_option = yes
	}
}

# cp_debug.2 -- Directly trigger The Deed, no law activation.
cp_debug.2 = {
	type = country_event
	hidden = yes
	orphan = yes

	immediate = {
		c:EGY ?= {
			trigger_event = { id = cp_layla.2 }
		}
	}
}

# cp_debug.3 -- Directly trigger The Bey Returns.
cp_debug.3 = {
	type = country_event
	hidden = yes
	orphan = yes

	immediate = {
		c:EGY ?= {
			trigger_event = { id = cp_layla.3 }
		}
	}
}

# cp_debug.4 -- Directly trigger He Marches.
cp_debug.4 = {
	type = country_event
	hidden = yes
	orphan = yes

	immediate = {
		c:EGY ?= {
			trigger_event = { id = cp_layla.4 }
		}
	}
}

# cp_debug.5 -- Activate Homesteading law. This exercises the
# on_law_enactment_pass dispatcher path. If cp_layla.2 fires, the
# dispatcher works; if not, compare to cp_debug.2.
cp_debug.5 = {
	type = country_event
	hidden = yes
	orphan = yes

	immediate = {
		c:EGY ?= {
			activate_law = law_type:law_homesteading
		}
	}
}

# cp_debug.6 -- Age Layla by 10 years. Hidden; confirm via JE panel.
cp_debug.6 = {
	type = country_event
	hidden = yes
	orphan = yes

	immediate = {
		c:EGY ?= {
			change_variable = { name = cp_age add = 10 }
		}
	}
}
