###############################################################################
# Common People -- Farid al-Haddad events
#
# Farid is Person 9: a Misri railway porter and errand runner whose wages,
# sleep, and small literacy make railways and telegraphy visible at ground
# level.
###############################################################################

namespace = cp_farid


# cp_farid.1 -- hidden setup
cp_farid.1 = {
	type = country_event
	hidden = yes

	trigger = {
		c:EGY ?= this
		has_global_variable = cp_person_farid_alive
		NOT = { has_variable = cp_farid_alive }
	}

	immediate = {
		set_variable = { name = cp_farid_alive value = 1 }
		set_variable = { name = cp_farid_age value = 17 }
		set_variable = { name = cp_farid_hope value = 6 }
		set_variable = { name = cp_farid_exhaustion value = 5 }
		set_variable = { name = cp_farid_sol value = 9 }
		set_variable = { name = cp_farid_literacy value = 8 }
		set_variable = { name = cp_farid_radical value = 0 }
		set_variable = { name = cp_farid_loyalist value = 0 }
		set_variable = { name = cp_farid_station_fear value = 4 }
		set_variable = cp_farid_profession_laborers
		set_variable = { name = cp_farid_w_infrastructure value = 10 }
		set_variable = { name = cp_farid_w_work value = 8 }
		set_variable = { name = cp_farid_w_city value = 7 }
		set_variable = { name = cp_farid_w_authority value = 6 }
		set_variable = { name = cp_farid_w_education value = 5 }
		set_variable = { name = cp_farid_w_schedule value = 6 }

		cp_set_person_home_state_from_spawn_marker_or_default = { person = farid country = EGY state = STATE_LOWER_EGYPT }
		cp_set_person_home_place = { person = farid place = tanta }
		cp_set_person_workplace_profile = { person = farid profile = rail }
		cp_farid_refresh_sol = yes

		if = {
			limit = {
				NOT = { has_variable = cp_startup_conditional_scan }
			}
			random_list = {
				60 = { cp_try_fire_world_response = { event = cp_farid.10 person = farid } }
				40 = { }
			}
		}
	}
}


# cp_farid.10 -- "The Platform Boy"
cp_farid.10 = {
	type = country_event

	title = cp_farid.10.t
	desc = cp_farid.10.d
	flavor = cp_farid.10.f

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	event_image = {
		texture = "gfx/event_pictures/cp_farid_10_the_platform_boy_v01.dds"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/MUSIC/Stingers/events/enthusiastic"

	duration = 3

	trigger = {
		has_variable = cp_farid_alive
		var:cp_farid_alive > 0
		NOT = { has_variable = cp_farid_seen_intro }
	}

	immediate = {
		cp_farid_mark_intro = yes
	}

	option = {
		name = cp_farid.10.a
		default_option = yes
	}
}
