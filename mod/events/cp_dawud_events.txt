###############################################################################
# Common People -- Dawud Hanna events
#
# Dawud is Person 10: a Misri Orthodox dockworker whose hands make ports,
# steamships, and cargo cranes part of ordinary Egyptian life.
###############################################################################

namespace = cp_dawud


# cp_dawud.1 -- hidden setup
cp_dawud.1 = {
	type = country_event
	hidden = yes

	trigger = {
		c:EGY ?= this
		has_global_variable = cp_person_dawud_alive
		NOT = { has_variable = cp_dawud_alive }
	}

	immediate = {
		set_variable = { name = cp_dawud_alive value = 1 }
		set_variable = { name = cp_dawud_age value = 28 }
		set_variable = { name = cp_dawud_hope value = 5 }
		set_variable = { name = cp_dawud_exhaustion value = 6 }
		set_variable = { name = cp_dawud_sol value = 10 }
		set_variable = { name = cp_dawud_literacy value = 7 }
		set_variable = { name = cp_dawud_radical value = 0 }
		set_variable = { name = cp_dawud_loyalist value = 0 }
		set_variable = { name = cp_dawud_rope_scars value = 3 }
		set_variable = cp_dawud_profession_laborers
		set_variable = { name = cp_dawud_w_trade value = 10 }
		set_variable = { name = cp_dawud_w_work value = 8 }
		set_variable = { name = cp_dawud_w_machinery value = 7 }
		set_variable = { name = cp_dawud_w_religion value = 5 }
		set_variable = { name = cp_dawud_w_war value = 4 }
		set_variable = { name = cp_dawud_w_politics value = 4 }

		cp_set_person_home_state_from_spawn_marker_or_default = { person = dawud country = EGY state = STATE_LOWER_EGYPT }
		cp_set_person_home_place = { person = dawud place = alexandria_port }
		cp_set_person_workplace_profile = { person = dawud profile = port }
		cp_dawud_refresh_sol = yes

		if = {
			limit = {
				NOT = { has_variable = cp_startup_conditional_scan }
			}
			random_list = {
				55 = { cp_try_fire_world_response = { event = cp_dawud.10 person = dawud } }
				45 = { }
			}
		}
	}
}


# cp_dawud.10 -- "The Rope Burn"
cp_dawud.10 = {
	type = country_event

	title = cp_dawud.10.t
	desc = cp_dawud.10.d
	flavor = cp_dawud.10.f

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	event_image = {
		texture = "gfx/event_pictures/cp_dawud_10_the_rope_burn_v01.dds"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/MUSIC/Stingers/events/tranquil"

	duration = 3

	trigger = {
		has_variable = cp_dawud_alive
		var:cp_dawud_alive > 0
		NOT = { has_variable = cp_dawud_seen_intro }
	}

	immediate = {
		cp_dawud_mark_intro = yes
	}

	option = {
		name = cp_dawud.10.a
		default_option = yes
	}
}
