namespace = cp_zaynab


# cp_zaynab.1 -- hidden setup
cp_zaynab.1 = {
	type = country_event
	hidden = yes

	trigger = {
		c:EGY ?= this
		has_global_variable = cp_person_zaynab_alive
		NOT = { has_variable = cp_zaynab_alive }
	}

	immediate = {
		set_variable = { name = cp_zaynab_alive value = 1 }
		set_variable = { name = cp_zaynab_age value = 43 }
		set_variable = { name = cp_zaynab_hope value = 5 }
		set_variable = { name = cp_zaynab_exhaustion value = 4 }
		set_variable = { name = cp_zaynab_sol value = 8 }
		set_variable = { name = cp_zaynab_literacy value = 2 }
		set_variable = { name = cp_zaynab_radical value = 0 }
		set_variable = { name = cp_zaynab_loyalist value = 0 }
		set_variable = cp_zaynab_profession_peasants
		set_variable = cp_zaynab_has_birth_basin
		set_variable = { name = cp_zaynab_w_health value = 10 }
		set_variable = { name = cp_zaynab_w_family value = 9 }
		set_variable = { name = cp_zaynab_w_religion value = 7 }
		set_variable = { name = cp_zaynab_w_work value = 8 }
		set_variable = { name = cp_zaynab_w_children value = 9 }

		cp_set_person_home_state = { person = zaynab country = EGY state = STATE_LOWER_EGYPT }
		cp_set_person_home_place = { person = zaynab place = mit_al_nakhla }
		cp_set_person_workplace_profile = { person = zaynab profile = public_service }
		cp_zaynab_refresh_sol = yes
	}
}


# cp_zaynab.10 -- "The Brass Basin"
cp_zaynab.10 = {
	type = country_event

	title = cp_zaynab.10.t
	desc = cp_zaynab.10.d
	flavor = cp_zaynab.10.f

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	event_image = {
		texture = "gfx/event_pictures/cp_zaynab_intro.dds"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/MUSIC/Stingers/events/spiritual"

	duration = 3

	trigger = {
		has_variable = cp_zaynab_alive
		var:cp_zaynab_alive > 0
		NOT = { has_variable = cp_zaynab_seen_intro }
	}

	immediate = {
		cp_zaynab_mark_intro = yes
	}

	option = {
		name = cp_zaynab.10.a
		default_option = yes
	}
}
