###############################################################################
# Common People -- startup events
#
# Fires once at game start for each country we have content for. Layla is not
# a V3 character; she is a set of variables on c:EGY. Future characters follow
# the same pattern, each with their own variable namespace on their home
# country's scope.
#
# Root scope: country (EGY) when cp_startup.1 fires.
###############################################################################

namespace = cp_startup

# cp_startup.1 -- Seed Egypt with Layla
#
# Initializes every country variable Layla needs and fires her intro event.
# Idempotent: cp_layla_alive guards re-entry across save/reload and across
# the on_game_started hook firing for the first time.
cp_startup.1 = {
	type = country_event
	hidden = yes

	trigger = {
		NOT = { has_variable = cp_layla_alive }
	}

	immediate = {
		# --- Life guard ---
		# cp_layla_alive = 1 while she lives. Goes to 0 at her death event
		# (which deactivates the JE). Every sensor / dispatcher / pulse
		# gates on this.
		set_variable = { name = cp_layla_alive    value = 1 }

		# --- Identity ---
		set_variable = { name = cp_age            value = 22 }

		# --- Profession as flag set ---
		# Exactly one cp_profession_* is set at a time. Transitions clear
		# the old one and set the new one. Flag-based because string
		# variables are brittle for trigger comparisons in V3.
		set_variable = cp_profession_peasants

		# --- Home state as a state-scope marker ---
		# Her residence is data-driven: the region_state she lives in
		# carries var:cp_layla_lives_here. Any EGY state at any time
		# (Delta, Hejaz, Sudan, Syria under Muhammad Ali, etc.) can host
		# her without hardcoding a flag list. At game start she lives in
		# the Delta. Must run BEFORE cp_refresh_sol_from_pop, which reads
		# from the marker.
		cp_move_to_state = { state = STATE_LOWER_EGYPT }

		# --- V3-mirroring canonical state (scales match V3 itself) ---
		# SoL 1..99. Seed from her cohort via the pop-group read, then
		# layer personal law/variance via the library.
		set_variable = { name = cp_sol            value = 5 }
		cp_refresh_sol_from_pop = yes
		# Year-over-year snapshot and delta. Updated in cp_on_yearly.
		set_variable = { name = cp_sol_prev       value = var:cp_sol }
		set_variable = { name = cp_sol_delta      value = 0 }
		# Literacy 0..100 (percent).
		set_variable = { name = cp_literacy       value = 0 }
		# Radicalism / loyalism 0..100. Mutually exclusive; effects enforce.
		set_variable = { name = cp_radical        value = 0 }
		set_variable = { name = cp_loyalist       value = 0 }
		# Pop-tick working vars. Overwritten yearly by cp_pop_tick_*;
		# seeded here so lint can see them before first yearly pulse.
		set_variable = { name = cp_cohort_literacy_target value = 0 }
		set_variable = { name = cp_expected_sol   value = 5 }
		set_variable = { name = cp_sol_gap        value = 0 }
		# Tax burden 0..10. Recomputed yearly. Year-over-year snapshot
		# lets cp_on_yearly detect sharp spikes and fire the tax-spike event.
		set_variable = { name = cp_tax_burden     value = 0 }
		cp_refresh_tax_burden = yes
		set_variable = { name = cp_tax_burden_prev value = var:cp_tax_burden }

		# --- Inner state (0..10) ---
		set_variable = { name = cp_hope           value = 5 }
		set_variable = { name = cp_exhaustion     value = 3 }

		# --- Monthly mood roll (1..10) ---
		# Rerolled monthly in cp_on_monthly. Drives which mood bucket her
		# status line falls into: low 1..3, mid 4..6, high 7..10.
		set_variable = { name = cp_mood_roll      value = 5 }

		# --- Monthly variant roll (1..3) ---
		# Rerolled alongside mood. Picks which of three prose variants
		# within the chosen mood bucket is displayed this month. Each
		# (situation × mood × variant) triple resolves to one triggered_desc.
		set_variable = { name = cp_status_variant value = 1 }

		# --- Personality weights (0..10) ---
		# Drift silently + via event effects. See documentation/characters/layla/README.md.
		set_variable = { name = cp_w_land_reform     value = 10 }
		set_variable = { name = cp_w_war             value = 7 }
		set_variable = { name = cp_w_revolution      value = 8 }
		set_variable = { name = cp_w_religion        value = 6 }
		set_variable = { name = cp_w_womens_rights   value = 5 }
		set_variable = { name = cp_w_economy         value = 5 }
		set_variable = { name = cp_w_france          value = 4 }
		set_variable = { name = cp_w_education       value = 3 }
		set_variable = { name = cp_w_labor_laws      value = 2 }
		set_variable = { name = cp_w_diplomacy       value = 1 }

		# --- Nation opinions (-100..+100) ---
		# Absence of a variable = total unawareness. She only has opinions
		# on nations she has heard of. New opinions get set when events
		# introduce them.
		set_variable = { name = cp_opinion_fra       value = 5 }
		set_variable = { name = cp_opinion_ott       value = 10 }
		set_variable = { name = cp_opinion_gbr       value = 0 }
		set_variable = { name = cp_opinion_egy       value = 10 }

		# --- Household (Ahmed is narrative; no V3 character for him either) ---
		set_variable = { name = cp_ahmed_alive       value = 1 }
		set_variable = { name = cp_ahmed_age         value = 26 }
		set_variable = cp_ahmed_profession_peasant
		set_variable = { name = cp_married           value = 1 }
		set_variable = { name = cp_children          value = 0 }

		# --- Debug tick ---
		# Incremented by the JE's on_monthly_pulse. Visible in the status
		# line during Phase A verification; removed from UI once the pulse
		# is proven to fire.
		set_variable = { name = cp_debug_tick        value = 0 }

		# --- Journal entry + intro ---
		add_journal_entry = { type = cp_je_layla }
		trigger_event = { id = cp_layla.1 }
	}
}
