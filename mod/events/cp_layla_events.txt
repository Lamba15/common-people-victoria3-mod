###############################################################################
# Common People -- Layla story events
#
# All Layla events are country_event (root = c:EGY). Layla's state lives as
# country variables on c:EGY, so reads are unqualified (`has_variable = X`,
# `var:cp_hope`) and writes go through cp_memory scripted_effects.
#
# Pattern B (image-led):
#   - event_image with custom DDS
#   - short desc (one sentence setup)
#   - long flavor (Mahfouz-register prose)
#   - no portrait, no gui_window
#
# Taxonomy:
#   - Reactions (fired by dispatchers): cp_layla.2 (homesteading),
#     cp_layla.3 (serfdom restored), cp_layla.4 (Ahmed conscripted).
#   - Pulses (fired by the JE's check-on button via weighted random):
#     cp_layla.10..15.
#   - Story beats (orphan until hooked): cp_layla.1 (intro, fired from
#     cp_startup.1), cp_layla.5 (childbirth), cp_layla.6 (reflection).
###############################################################################

namespace = cp_layla

# cp_layla.1 -- "The Farmer's Daughter" (intro)
#
# First meeting. Fires from cp_startup.1 immediately after variable init.
cp_layla.1 = {
	type = country_event

	title = cp_layla.1.t
	desc  = cp_layla.1.d
	flavor = cp_layla.1.f

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	event_image = {
		texture = "gfx/event_pictures/cp_layla_intro.dds"
	}

	duration = 3

	trigger = {
		has_variable = cp_layla_alive
	}

	option = {
		name = cp_layla.1.a
		default_option = yes
		set_variable = cp_met_player
	}
}

# cp_layla.90 -- "The First Envelope" (first welfare receipt)
#
# Fires once per life when any welfare law is active AND she qualifies.
# Three triggered_desc variants by which rung reached her first:
# pension (age 60+ + law_old_age_pension), subsidy (laborer household

	type = country_event

		first_valid = {
			triggered_desc = {

					var:cp_age >= 60

				desc = cp_layla.90.d.subsidy
				trigger = {
					has_law = law_type:law_wage_subsidies

	duration = 3

	trigger = {
		has_variable = cp_layla_alive
	}

	immediate = {
		set_variable = cp_receives_welfare
	}

	option = {
		name = cp_layla.90.a
		default_option = yes

		cp_shift_radical = { delta = 1 }

# =============================================================================
# Women's rights enactments -- cp_layla.93..96
# =============================================================================
# Four first-time-enactment reactions, one per rung above no_womens_rights.
# Each is gated on a permanent cp_seen_womens_rights_* flag so the event
# fires exactly once per life, even if the law is rolled back and re-enacted.
# Dispatcher lives in cp_on_law_enacted and checks the same flag before
# trigger_event; the flag is set inside the option so console-fired test
# runs still work without a dispatcher.
#
# The law ladder in lawgroup_rights_of_women:
#   law_no_womens_rights    (starting point, nothing to react to)
#   law_women_in_the_fields (quasi-corvée; a grim "recognition" by the state)
#   law_women_own_property  (she can hold a deed in her own name)
#   law_women_in_the_workplace (unlocks self-employment crossroads)
#   law_womens_suffrage     (her name on the rolls)
# =============================================================================

# cp_layla.93 -- "The Law Catches Up to Us" (law_women_in_the_fields)
#
# A grim "progression": the state now requires women in the fields. For a
# peasant like Layla, it's not liberation — it's the formalisation of what
# she already does. For a city household, it's a notice that her hours are
# claimed. Radicalism rises if she is already struggling; exhaustion always
# rises because this rung always means more work.
cp_layla.93 = {
	type = country_event

	title = cp_layla.93.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = cp_layla.93.d.peasant
				trigger = {
					OR = {
						has_variable = cp_profession_peasants
						has_variable = cp_profession_farmers
					}
				}
			}
			triggered_desc = {
				desc = cp_layla.93.d.city
				trigger = { has_variable = cp_in_city }
			}
			triggered_desc = {
				desc = cp_layla.93.d.default
				trigger = { always = yes }
			}
		}
	}
	flavor = cp_layla.93.f

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	event_image = {
		texture = "gfx/event_pictures/cp_layla_no_wind.dds"
	}

	duration = 3

	trigger = {
		has_variable = cp_layla_alive
		NOT = { has_variable = cp_seen_womens_rights_fields }
	}

	immediate = {
		set_variable = cp_seen_womens_rights_fields
	}

	# Option A: weary recognition. Yes. We were already doing it.
	option = {
		name = cp_layla.93.a
		default_option = yes
		cp_shift_exhaustion = { delta = 1 }
	}
	# Option B: the small anger. The state's name for her labour is still
	# not payment. Weight toward land_reform (she wants something deeper).
	option = {
		name = cp_layla.93.b
		cp_shift_radical = { delta = 1 }
		cp_shift_weight = { weight = land_reform delta = 1 }
	}
}

				trigger = {
					has_variable = cp_ahmed_alive
					var:cp_ahmed_alive = 0
				}
			}
			triggered_desc = {
				desc = cp_layla.97.f.poor
				trigger = { var:cp_sol <= 8 }
			}
			triggered_desc = {
				desc = cp_layla.97.f.chosen
				trigger = { always = yes }
			}
		}
	}

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	event_image = {
		texture = "gfx/event_pictures/cp_layla_mothers_hand.dds"
	}

	duration = 5

	trigger = {
		has_variable = cp_layla_alive
		has_variable = cp_may_work
		NOT = { has_variable = cp_seen_self_employment }
		NOT = { has_variable = cp_self_employed }
	}

	immediate = {
		set_variable = cp_seen_self_employment
	}

	# Option A: open a stall. Default option.
	# Prose register differs by forced/chosen but the mechanic is the
	# same: she is now a shopkeeper, her household's SoL lifts, her
	# exhaustion rises. Forced register ALSO sets cp_self_employed_forced
	# which downstream events can read for prose colour.
	option = {
		name = cp_layla.97.a
		default_option = yes
		# Branch on whether this is a forced entry.
		if = {
			limit = {
				OR = {
					AND = {
						has_variable = cp_ahmed_alive
						var:cp_ahmed_alive = 0
					}
					var:cp_sol <= 8
				}
			}
			cp_start_self_employed_forced = yes
		}
		else = {
			cp_start_self_employed_chosen = yes
		}
	}
	# Option B: decline. Only available when the crossroads is a choice
	# (not forced by widowhood or poverty). Prose acknowledges the
	# refused possibility.
	option = {
		name = cp_layla.97.b
		trigger = {
			NOT = {
				OR = {
					AND = {
						has_variable = cp_ahmed_alive
						var:cp_ahmed_alive = 0
					}
					var:cp_sol <= 8
				}
			}
		}
		cp_decline_self_employment = yes
	}
}
