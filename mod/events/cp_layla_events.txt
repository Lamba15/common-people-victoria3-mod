###############################################################################
# Common People -- Layla story events
#
# All Layla events follow Pattern B (image-led, NOT portrait):
#   - no gui_window, no left_icon
#   - event_image with custom DDS that depicts Layla in the scene
#   - short desc (one sentence setup)
#   - long flavor (the longer Mahfouz-style prose; renders as small italic)
#
# All events are country_event with root = c:EGY. They access Layla via the
# saved scope cp_layla, which cp_startup.1 saves on the country at game start.
# Every event guards on `exists = scope:cp_layla` so a missing/dead Layla
# fails silently rather than erroring.
#
# Console testing:
#   event cp_layla.1     -- intro
#   event cp_layla.2     -- The Deed (homesteading)
#   event cp_layla.3     -- The Bey Returns (serfdom restored)
#   event cp_layla.4     -- He Marches (Ahmed conscripted)
#   event cp_layla.5     -- The First Cry (childbirth)
#
# Note: V3 requires numeric event IDs (namespace.NUMBER). String IDs are
# rejected by the parser.
###############################################################################

namespace = cp_layla

# cp_layla.1 -- "The Farmer's Daughter"
#
# First meeting with Layla. Fires immediately from cp_startup.1.
cp_layla.1 = {
    type = country_event

    title = cp_layla.1.t
    desc  = cp_layla.1.d
    flavor = cp_layla.1.f

    icon = "gfx/interface/icons/event_icons/event_default.dds"

    event_image = {
        texture = "gfx/event_pictures/cp_layla_intro.dds"
    }

    duration = 3

    trigger = {
        exists = scope:cp_layla
    }

    option = {
        name = cp_layla.1.a
        default_option = yes
        scope:cp_layla = {
            set_variable = cp_met_player
        }
    }
}
