namespace = cp_nabil


# cp_nabil.1 -- hidden setup
cp_nabil.1 = {
	type = country_event
	hidden = yes

	trigger = {
		c:EGY ?= this
		has_global_variable = cp_person_nabil_alive
		NOT = { has_variable = cp_nabil_alive }
	}

	immediate = {
		set_variable = { name = cp_nabil_alive value = 1 }
		set_variable = { name = cp_nabil_age value = 31 }
		set_variable = { name = cp_nabil_hope value = 4 }
		set_variable = { name = cp_nabil_exhaustion value = 5 }
		set_variable = { name = cp_nabil_sol value = 9 }
		set_variable = { name = cp_nabil_literacy value = 4 }
		set_variable = { name = cp_nabil_radical value = 0 }
		set_variable = { name = cp_nabil_loyalist value = 1 }
		set_variable = { name = cp_nabil_night_eyes value = 2 }
		set_variable = { name = cp_nabil_just_spawned days = 30 }
		set_variable = cp_nabil_profession_laborers
		set_variable = { name = cp_nabil_w_order value = 10 }
		set_variable = { name = cp_nabil_w_uniform value = 7 }
		set_variable = { name = cp_nabil_w_fear value = 7 }
		set_variable = { name = cp_nabil_w_crowd value = 6 }
		set_variable = { name = cp_nabil_w_family value = 5 }

		cp_set_person_home_state = { person = nabil country = EGY state = STATE_LOWER_EGYPT }
		cp_set_person_home_place = { person = nabil place = bab_al_khalq }
		cp_set_person_workplace_profile = { person = nabil profile = urban }
		cp_nabil_refresh_sol = yes

		if = {
			limit = {
				NOT = { has_variable = cp_startup_conditional_scan }
			}
			random_list = {
				50 = { cp_try_fire_world_response = { event = cp_nabil.10 person = nabil } }
				50 = { }
			}
		}
	}
}


# cp_nabil.10 -- "The Lamp at the Corner"
cp_nabil.10 = {
	type = country_event

	title = cp_nabil.10.t
	desc = cp_nabil.10.d
	flavor = cp_nabil.10.f

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	event_image = {
		texture = "gfx/event_pictures/cp_nabil_10_the_lamp_at_the_corner_v01.dds"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/MUSIC/Stingers/events/tranquil"

	duration = 3

	trigger = {
		has_variable = cp_nabil_alive
		var:cp_nabil_alive > 0
		NOT = { has_variable = cp_nabil_seen_intro }
	}

	immediate = {
		cp_nabil_mark_intro = yes
	}

	option = {
		name = cp_nabil.10.a
		default_option = yes
	}
}
