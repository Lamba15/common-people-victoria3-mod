###############################################################################
# Common People -- Layla imagines an audience with the Pasha
#
# Framing: while kneading bread, she imagines walking into the palace in
# Cairo and standing before the ruler of Egypt. The player is the Pasha.
# At each turn the player picks what he says back. She responds according
# to which Pasha she faces tonight. The audience ends when the fantasy
# dissolves; she returns to the quern.
#
# Full design: documentation/characters/layla/conversation.md
#
# Event numbering:
#   cp_conversation.1         -- Root: she sets the scene, picks the topic
#   cp_conversation.10..19    -- The Land topic
#   cp_conversation.20..29    -- The Ahmed topic (reacts to his state)
#   cp_conversation.30..39    -- The Children topic (gated on cp_children > 0)
#   cp_conversation.40..49    -- The Room in the City (gated on cp_in_city)
#   cp_conversation.50..59    -- The Paper (gated on cp_owns_land)
#   cp_conversation.60..69    -- The Politics (gated on cp_radical or cp_loyalist)
#   cp_conversation.70..79    -- The Boss (gated on laborer + city)
#   cp_conversation.80..89    -- A Small Thing (always available fallback)
#   cp_conversation.90        -- Shared closer (fantasy dissolves)
#   cp_conversation.100..109  -- The Distance (gated on cp_sol >= 25 — wealth)
###############################################################################

namespace = cp_conversation


# ============================================================================
# cp_conversation.1 -- Root: she goes to Cairo in her mind
# ============================================================================

cp_conversation.1 = {
	type = country_event
	placement = root

	title = cp_conversation.1.t
	# Framing varies with her state. A widow entering Cairo in her mind is
	# not the same woman who enters it with her husband asleep in the next
	# room. first_valid picks the first matching variant.
	desc = {
		first_valid = {
			triggered_desc = {
				desc = cp_conversation.1.d.widow
				trigger = { var:cp_ahmed_alive = 0 }
			}
			triggered_desc = {
				desc = cp_conversation.1.d.at_war
				trigger = { has_variable = cp_ahmed_at_war }
			}
			triggered_desc = {
				desc = cp_conversation.1.d.default
				trigger = { always = yes }
			}
		}
	}
	flavor = cp_conversation.1.f

	event_image = {
		texture = "gfx/event_pictures/cp_layla_intro.dds"
	}

	duration = 5

	# About the land. Available whenever she is rural.
	option = {
		name = cp_conversation.1.a
		trigger = {
			OR = {
				has_variable = cp_profession_peasants
				has_variable = cp_profession_farmers
				has_variable = cp_owns_land
				has_variable = cp_under_serfdom
			}
			NOT = { has_variable = cp_seen_audience_land }
		}
		set_variable = { name = cp_seen_audience_land months = 18 }
		trigger_event = { id = cp_conversation.10 popup = yes }
	}

	# About Ahmed. Available whether he is home, at war, or dead —
	# the prose in cp_conversation.20 adapts to which.
	option = {
		name = cp_conversation.1.b
		trigger = {
			has_variable = cp_married
			NOT = { has_variable = cp_seen_audience_ahmed }
		}
		set_variable = { name = cp_seen_audience_ahmed months = 18 }
		trigger_event = { id = cp_conversation.20 popup = yes }
	}

	# About the children. Only if she has any.
	option = {
		name = cp_conversation.1.c
		trigger = {
			has_variable = cp_children
			var:cp_children > 0
			NOT = { has_variable = cp_seen_audience_children }
		}
		set_variable = { name = cp_seen_audience_children months = 18 }
		trigger_event = { id = cp_conversation.30 popup = yes }
	}

	# About the room in the city. Only if she lives urban now.
	option = {
		name = cp_conversation.1.d
		trigger = {
			has_variable = cp_in_city
			NOT = { has_variable = cp_seen_audience_city }
		}
		set_variable = { name = cp_seen_audience_city months = 18 }
		trigger_event = { id = cp_conversation.40 popup = yes }
	}

	# About the paper — the field now hers on paper. Only if she owns land.
	option = {
		name = cp_conversation.1.e
		trigger = {
			has_variable = cp_owns_land
			NOT = { has_variable = cp_seen_audience_paper }
		}
		set_variable = { name = cp_seen_audience_paper months = 18 }
		trigger_event = { id = cp_conversation.50 popup = yes }
	}

	# About politics. Only if she has a stance (radical or loyalist).
	option = {
		name = cp_conversation.1.f
		trigger = {
			OR = {
				AND = { has_variable = cp_radical   var:cp_radical   > 0 }
				AND = { has_variable = cp_loyalist  var:cp_loyalist  > 0 }
			}
			NOT = { has_variable = cp_seen_audience_politics }
		}
		set_variable = { name = cp_seen_audience_politics months = 18 }
		trigger_event = { id = cp_conversation.60 popup = yes }
	}

	# About the mill, and the boss. Only if she's an urban laborer.
	option = {
		name = cp_conversation.1.g
		trigger = {
			has_variable = cp_profession_laborers
			has_variable = cp_in_city
			NOT = { has_variable = cp_seen_audience_boss }
		}
		set_variable = { name = cp_seen_audience_boss months = 18 }
		trigger_event = { id = cp_conversation.70 popup = yes }
	}

	# About wealth — when it came, and what it feels like. Only if she's
	# crossed into Prosperous (SoL 25+).
	option = {
		name = cp_conversation.1.i
		trigger = {
			has_variable = cp_sol
			var:cp_sol >= 25
			NOT = { has_variable = cp_seen_audience_distance }
		}
		set_variable = { name = cp_seen_audience_distance months = 18 }
		trigger_event = { id = cp_conversation.100 popup = yes }
	}

	# About the envelope. Only if she has received welfare at least once.
	option = {
		name = cp_conversation.1.j
		trigger = {
			has_variable = cp_receives_welfare
			NOT = { has_variable = cp_seen_audience_envelope }
		}
		set_variable = { name = cp_seen_audience_envelope months = 18 }
		trigger_event = { id = cp_conversation.110 popup = yes }
	}

	# About the vote. Only if she has voted.
	option = {
		name = cp_conversation.1.k
		trigger = {
			has_variable = cp_has_voted
			NOT = { has_variable = cp_seen_audience_vote }
		}
		set_variable = { name = cp_seen_audience_vote months = 18 }
		trigger_event = { id = cp_conversation.120 popup = yes }
	}

	# About the stall. Only if she is self-employed.
	option = {
		name = cp_conversation.1.l
		trigger = {
			has_variable = cp_self_employed
			NOT = { has_variable = cp_seen_audience_stall }
		}
		set_variable = { name = cp_seen_audience_stall months = 18 }
		trigger_event = { id = cp_conversation.130 popup = yes }
	}

	# About the Nile. Available for rural/land-bound households.
	option = {
		name = cp_conversation.1.m
		trigger = {
			OR = {
				has_variable = cp_profession_peasants
				has_variable = cp_profession_farmers
				has_variable = cp_owns_land
				has_variable = cp_under_serfdom
			}
			NOT = { has_variable = cp_seen_audience_nile }
		}
		set_variable = { name = cp_seen_audience_nile months = 18 }
		trigger_event = { id = cp_conversation.140 popup = yes }
	}

	# About the child who did not live. Only if she has lost a child.
	# Long cooldown (36 months) — this is not a topic to raise lightly.
	option = {
		name = cp_conversation.1.n
		trigger = {
			has_variable = cp_lost_a_child
			NOT = { has_variable = cp_seen_audience_lost_child }
		}
		set_variable = { name = cp_seen_audience_lost_child months = 36 }
		trigger_event = { id = cp_conversation.150 popup = yes }
	}

	# About the school. Only if her daughter attends.
	option = {
		name = cp_conversation.1.o
		trigger = {
			has_variable = cp_daughter_at_school
			NOT = { has_variable = cp_seen_audience_school }
		}
		set_variable = { name = cp_seen_audience_school months = 18 }
		trigger_event = { id = cp_conversation.160 popup = yes }
	}

	# A small thing. Always available. Shorter cooldown (6 months) because
	# it's the "I don't have a speech tonight" option and should be reachable
	# more often than the big topics.
	option = {
		name = cp_conversation.1.h
		trigger = {
			NOT = { has_variable = cp_seen_audience_small_thing }
		}
		set_variable = { name = cp_seen_audience_small_thing months = 6 }
		trigger_event = { id = cp_conversation.80 popup = yes }
	}

	# Fallback so the root never has zero options.
	option = {
		name = cp_conversation.1.z
		default_option = yes
	}
}

# ============================================================================

	}
	# Pasha: cruel. Always available.

		name = cp_conversation.20.c
