###############################################################################
# Common People -- Layla imagines an audience with the Pasha
#
# Framing: while kneading bread, she imagines walking into the palace in
# Cairo and standing before the ruler of Egypt. The player is the Pasha.
# At each turn the player picks what he says back. She responds according
# to which Pasha she faces tonight. The audience ends when the fantasy
# dissolves; she returns to the quern.
#
# Full design: documentation/characters/layla/conversation.md
#
# Event numbering:
#   cp_conversation.1      -- Root: she sets the scene, picks the topic
#   cp_conversation.10..19 -- The Land topic (Phase 1, fully written)
#   cp_conversation.20..29 -- Ahmed topic (Phase 2)
#   cp_conversation.30..39 -- The Bey topic (Phase 2)
#   cp_conversation.40..49 -- The Water topic (Phase 2)
#   cp_conversation.50..59 -- Herself topic (Phase 3)
#   cp_conversation.90     -- Shared closer (fantasy dissolves)
###############################################################################

namespace = cp_conversation


# ============================================================================
# cp_conversation.1 -- Root: she goes to Cairo in her mind
# ============================================================================

cp_conversation.1 = {
	type = country_event
	placement = root

	title = cp_conversation.1.t
	# Framing varies with her state. A widow entering Cairo in her mind is
	# not the same woman who enters it with her husband asleep in the next
	# room. first_valid picks the first matching variant.
	desc = {
		first_valid = {
			triggered_desc = {
				desc = cp_conversation.1.d.widow
				trigger = { var:cp_ahmed_alive = 0 }
			}
			triggered_desc = {
				desc = cp_conversation.1.d.at_war
				trigger = { has_variable = cp_ahmed_at_war }
			}
			triggered_desc = {
				desc = cp_conversation.1.d.default
				trigger = { always = yes }
			}
		}
	}
	flavor = cp_conversation.1.f

	event_image = {
		texture = "gfx/event_pictures/cp_layla_intro.dds"
	}

	duration = 5

	# About the land. Available whenever she is rural.
	option = {
		name = cp_conversation.1.a
		trigger = {
			OR = {
				has_variable = cp_profession_peasants
				has_variable = cp_profession_farmers
				has_variable = cp_owns_land
				has_variable = cp_under_serfdom
			}
			NOT = { has_variable = cp_seen_audience_land }
		}
		set_variable = { name = cp_seen_audience_land months = 18 }
		trigger_event = { id = cp_conversation.10 popup = yes }
	}

	# About Ahmed. Available whether he is home, at war, or dead —
	# the prose in cp_conversation.20 adapts to which.
	option = {
		name = cp_conversation.1.b
		trigger = {
			has_variable = cp_married
			NOT = { has_variable = cp_seen_audience_ahmed }
		}
		set_variable = { name = cp_seen_audience_ahmed months = 18 }
		trigger_event = { id = cp_conversation.20 popup = yes }
	}

	# Fallback so the root never has zero options.
	option = {
		name = cp_conversation.1.z
		default_option = yes
	}
}


# ============================================================================
# cp_conversation.10 -- Her opening speech (THE LAND)
# ============================================================================
# Her opening monologue to the Pasha. Options are what the Pasha says back.
# Three mood branches: dismissive, engaged, cruel.
# ============================================================================

cp_conversation.10 = {
	type = country_event
	placement = root

	title = cp_conversation.10.t
	# Her opening about the land varies with her condition. A serf and a
	# homesteader do not enter the same room.
	desc = {
		first_valid = {
			triggered_desc = {
				desc = cp_conversation.10.d.serf
				trigger = { has_variable = cp_under_serfdom }
			}
			triggered_desc = {
				desc = cp_conversation.10.d.owner
				trigger = { has_variable = cp_owns_land }
			}
			triggered_desc = {
				desc = cp_conversation.10.d.default
				trigger = { always = yes }
			}
		}
	}
	flavor = cp_conversation.10.f

	event_image = {
		texture = "gfx/event_pictures/cp_layla_homesteading.dds"
	}

	duration = 5

	immediate = {
		# Topic-memory flag. Set here (not in .90) so save-across-branch
		# preserves the fact that she has spoken of land at least once.
		set_variable = cp_has_spoken_of_land
	}

	# Pasha: dismissive
	option = {
		name = cp_conversation.10.a
		trigger_event = { id = cp_conversation.11 popup = yes }
	}
	# Pasha: engaged
	option = {
		name = cp_conversation.10.b
		trigger_event = { id = cp_conversation.12 popup = yes }
	}
	# Pasha: cruel
	option = {
		name = cp_conversation.10.c
		trigger_event = { id = cp_conversation.13 popup = yes }
	}
}

# ============================================================================
# cp_conversation.20 -- Her opening speech (AHMED)
# ============================================================================
# The Ahmed topic adapts to his state: alive-at-home, conscripted-at-war,
# or dead. Three distinct opening monologues keyed on his condition;
# three Pasha responses, one of which is only available if he still lives.
# ============================================================================

cp_conversation.20 = {
	type = country_event
	placement = root

	title = cp_conversation.20.t

	desc = {
		first_valid = {
			triggered_desc = {
				desc = cp_conversation.20.d.widow
				trigger = { var:cp_ahmed_alive = 0 }
			}
			triggered_desc = {
				desc = cp_conversation.20.d.at_war
				trigger = { has_variable = cp_ahmed_at_war }
			}
			triggered_desc = {
				desc = cp_conversation.20.d.home
				trigger = { always = yes }
			}
		}
	}
	flavor = cp_conversation.20.f

	event_image = {
		texture = "gfx/event_pictures/cp_layla_ahmed_conscripted.dds"
	}

	duration = 5

	immediate = {
		set_variable = cp_has_spoken_of_ahmed
	}

	# Pasha: compassionate. Always available.
	option = {
		name = cp_conversation.20.a
		trigger_event = { id = cp_conversation.21 popup = yes }
	}
	# Pasha: political. Only meaningful if Ahmed is alive — the Pasha
	# cannot ask more of a dead man.
	option = {
		name = cp_conversation.20.b
		trigger = {
			has_variable = cp_ahmed_alive
			var:cp_ahmed_alive > 0
		}
		trigger_event = { id = cp_conversation.22 popup = yes }
	}
	# Pasha: cruel. Always available.
	option = {
		name = cp_conversation.20.c
		trigger_event = { id = cp_conversation.23 popup = yes }
	}
}

# ============================================================================
# cp_conversation.90 -- Closer (the fantasy dissolves)
# ============================================================================

cp_conversation.90 = {
	type = country_event
	placement = root

	title = cp_conversation.90.t
	# Her exit from the fantasy is colored by which topic she just
	# carried. Widow walks back into a different room than a wife does.
	desc = {
		first_valid = {
			triggered_desc = {
				desc = cp_conversation.90.d.widow
				trigger = {
					has_variable = cp_has_spoken_of_ahmed
					var:cp_ahmed_alive = 0
				}
			}
			triggered_desc = {
				desc = cp_conversation.90.d.default
				trigger = { always = yes }
			}
		}
	}
	flavor = cp_conversation.90.f

	event_image = {
		texture = "gfx/event_pictures/cp_layla_intro.dds"
	}

	duration = 5

	immediate = {
		# Permanent flag set by ANY completed audience. Per-topic flags
		# are now set in each topic's opening event (.10, .20) so
		# save-across-branch preserves which topic she spoke of.
		set_variable = cp_has_stood_before_pasha
		# Cooldown set here (not on the button) so mid-audience save/quit
		# does not lock the player out.
		set_variable = { name = cp_audience_cooldown days = 90 }
	}

	option = {
		name = cp_conversation.90.a
		default_option = yes
	}
}
