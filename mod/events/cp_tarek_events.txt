namespace = cp_tarek


# cp_tarek.1 -- hidden setup
cp_tarek.1 = {
	type = country_event
	hidden = yes

	trigger = {
		c:EGY ?= this
		has_global_variable = cp_person_tarek_alive
		NOT = { has_variable = cp_tarek_alive }
	}

	immediate = {
		set_variable = { name = cp_tarek_alive value = 1 }
		set_variable = { name = cp_tarek_age value = 26 }
		set_variable = { name = cp_tarek_hope value = 7 }
		set_variable = { name = cp_tarek_exhaustion value = 3 }
		set_variable = { name = cp_tarek_sol value = 28 }
		set_variable = { name = cp_tarek_literacy value = 45 }
		set_variable = { name = cp_tarek_radical value = 0 }
		set_variable = { name = cp_tarek_loyalist value = 2 }
		set_variable = { name = cp_tarek_certainty value = 6 }
		set_variable = { name = cp_tarek_profit_pressure value = 3 }
		set_variable = cp_tarek_profession_capitalists
		set_variable = { name = cp_tarek_w_industrialization value = 10 }
		set_variable = { name = cp_tarek_w_economy value = 10 }
		set_variable = { name = cp_tarek_w_labor_laws value = 8 }
		set_variable = { name = cp_tarek_w_free_trade value = 9 }
		set_variable = { name = cp_tarek_w_land_reform value = 6 }

		cp_set_person_home_state_from_spawn_marker_or_default = { person = tarek country = EGY state = STATE_LOWER_EGYPT }
		cp_set_person_home_place = { person = tarek place = bulaq }
		cp_set_person_workplace_profile = { person = tarek profile = manufacturing }
		cp_tarek_refresh_sol = yes

		if = {
			limit = {
				NOT = { has_variable = cp_startup_conditional_scan }
			}
			random_list = {
				55 = { cp_try_fire_world_response = { event = cp_tarek.10 person = tarek } }
				45 = { }
			}
		}
	}
}


# cp_tarek.10 -- "The Mill Ledger"
cp_tarek.10 = {
	type = country_event

	title = cp_tarek.10.t
	desc = cp_tarek.10.d
	flavor = cp_tarek.10.f

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	event_image = {
		texture = "gfx/event_pictures/cp_tarek_10_the_mill_ledger_v01.dds"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/MUSIC/Stingers/events/tranquil"

	duration = 3

	trigger = {
		has_variable = cp_tarek_alive
		var:cp_tarek_alive > 0
		NOT = { has_variable = cp_tarek_seen_intro }
	}

	immediate = {
		cp_tarek_mark_intro = yes
	}

	option = {
		name = cp_tarek.10.a
		default_option = yes
	}
}
