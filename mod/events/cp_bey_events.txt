namespace = cp_bey


# cp_bey.1 -- hidden setup
cp_bey.1 = {
	type = country_event
	hidden = yes

	trigger = {
		c:EGY ?= this
		has_global_variable = cp_person_bey_alive
		NOT = { has_variable = cp_bey_alive }
	}

	immediate = {
		set_variable = { name = cp_bey_alive value = 1 }
		set_variable = { name = cp_bey_age value = 52 }
		set_variable = { name = cp_bey_hope value = 6 }
		set_variable = { name = cp_bey_exhaustion value = 2 }
		set_variable = { name = cp_bey_sol value = 30 }
		set_variable = { name = cp_bey_literacy value = 35 }
		set_variable = { name = cp_bey_radical value = 0 }
		set_variable = { name = cp_bey_loyalist value = 5 }
		set_variable = { name = cp_bey_power value = 8 }
		set_variable = { name = cp_bey_certainty value = 8 }
		set_variable = cp_bey_profession_aristocrats
		set_variable = { name = cp_bey_w_land_reform value = 10 }
		set_variable = { name = cp_bey_w_industrialization value = 9 }
		set_variable = { name = cp_bey_w_landowners value = 10 }
		set_variable = { name = cp_bey_w_monarchy value = 8 }
		set_variable = { name = cp_bey_w_education value = 7 }

		cp_set_person_home_state = { person = bey country = EGY state = STATE_LOWER_EGYPT }
		cp_set_person_home_place = { person = bey place = sayyid_estate }
		cp_set_person_workplace_profile = { person = bey profile = manufacturing }
		cp_bey_refresh_sol = yes
	}
}


# cp_bey.10 -- "The Master's Window"
cp_bey.10 = {
	type = country_event

	title = cp_bey.10.t
	desc = cp_bey.10.d
	flavor = cp_bey.10.f

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	event_image = {
		texture = "gfx/event_pictures/cp_bey_intro.dds"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/MUSIC/Stingers/events/tranquil"

	duration = 3

	trigger = {
		has_variable = cp_bey_alive
		var:cp_bey_alive > 0
		NOT = { has_variable = cp_bey_seen_intro }
	}

	immediate = {
		cp_mark_event_seen = { person = bey id = intro days = 3650 }
	}

	option = {
		name = cp_bey.10.a
		default_option = yes
	}
}
