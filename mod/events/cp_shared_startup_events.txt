###############################################################################
# Common People -- shared startup events (r5)
#
# Single entry point for setting up always-on persons and sampling eligible
# conditional persons before the first visible contact. Called from cp_on_start
# (on_game_started subscription). Conditional entrants still use their
# cp_<person>_try_spawn effect from yearly/tech/building/law hooks later, but
# startup can now register them too if the selected start date or save state
# already satisfies their world conditions. The visible first contact is
# randomized in cp_shared_startup.2 so no person owns the start of every
# campaign.
###############################################################################

namespace = cp_shared_startup


# cp_shared_startup.1 -- "Register all persons"
#
# Iterates each always-on person (via a fixed list — V3 cannot loop over
# arbitrary names). Each person registers through cp_register_person, then
# triggers that person's hidden setup event if their local variable block does
# not exist yet.
#
# Adding a new always-on person: add one cp_register_person call here, create
# their cp_<person>_memory.txt + cp_<person>_events.txt, and add one lean
# if-block per hook to cp_on_actions.txt. Conditional persons instead need a
# called cp_<person>_try_spawn effect. No existing person file is modified.

cp_shared_startup.1 = {
	type = country_event
	hidden = yes

	trigger = {
		c:EGY ?= this
	}

	immediate = {
		if = {
			limit = { NOT = { has_variable = cp_common_people_je_added } }
			add_journal_entry = { type = cp_je_common_people }
			set_variable = cp_common_people_je_added
		}

		cp_layla_dispatch_startup = yes

		# Person 1 -- Layla al-Sharif of Mit al-Nakhla, Lower Egypt.
		# Misri Sunni peasant, 22 in 1836. Her country-variable block lives
		# in cp_layla_setup.1.
		if = {
			limit = {
				OR = {
					NOT = { has_variable = cp_layla_alive }
					AND = {
						has_variable = cp_layla_alive
						var:cp_layla_alive > 0
					}
				}
			}
			cp_register_person = {
				person = layla
				country = EGY
			}
		}
		trigger_event = { id = cp_layla_setup.1 }

		# Person 2 -- Hassan Bey al-Sayyid, a Misri aristocrat in
		# Lower Egypt. His country-variable block lives in cp_bey.1.
		cp_bey_dispatch_startup = yes
		if = {
			limit = {
				OR = {
					NOT = { has_variable = cp_bey_alive }
					AND = {
						has_variable = cp_bey_alive
						var:cp_bey_alive > 0
					}
				}
			}
			cp_register_person = {
				person = bey
				country = EGY
			}
		}
		trigger_event = { id = cp_bey.1 }

		# Person 3 -- Yusuf ibn Mahmud, a Misri soldier in Lower Egypt.
		# His country-variable block lives in cp_soldier.1.
		cp_soldier_dispatch_startup = yes
		if = {
			limit = {
				OR = {
					NOT = { has_variable = cp_soldier_alive }
					AND = {
						has_variable = cp_soldier_alive
						var:cp_soldier_alive > 0
					}
				}
			}
			cp_register_person = {
				person = soldier
				country = EGY
			}
		}
		trigger_event = { id = cp_soldier.1 }

		# Person 6 -- Nour, a Misri shopkeeper's widow in Lower Egypt.
		# Her country-variable block lives in cp_nour.1.
		cp_nour_dispatch_startup = yes
		if = {
			limit = {
				OR = {
					NOT = { has_variable = cp_nour_alive }
					AND = {
						has_variable = cp_nour_alive
						var:cp_nour_alive > 0
					}
				}
			}
			cp_register_person = {
				person = nour
				country = EGY
			}
		}
		trigger_event = { id = cp_nour.1 }

		# Person 7 -- Mina, a Misri Orthodox printer's apprentice in Cairo.
		# His country-variable block lives in cp_mina.1.
		cp_mina_dispatch_startup = yes
		if = {
			limit = {
				OR = {
					NOT = { has_variable = cp_mina_alive }
					AND = {
						has_variable = cp_mina_alive
						var:cp_mina_alive > 0
					}
				}
			}
			cp_register_person = {
				person = mina
				country = EGY
			}
		}
		trigger_event = { id = cp_mina.1 }

		# Person 8 -- Zaynab, a Misri village midwife in Lower Egypt.
		# Her country-variable block lives in cp_zaynab.1.
		cp_zaynab_dispatch_startup = yes
		if = {
			limit = {
				OR = {
					NOT = { has_variable = cp_zaynab_alive }
					AND = {
						has_variable = cp_zaynab_alive
						var:cp_zaynab_alive > 0
					}
				}
			}
			cp_register_person = {
				person = zaynab
				country = EGY
			}
		}
		trigger_event = { id = cp_zaynab.1 }

		# If a later start date or loaded scenario already satisfies a
		# conditional person's world gates, register them before first contact.
		# Their setup events see this one-day flag and suppress their own
		# immediate visible intro, letting cp_shared_startup.2 pick one opener.
		set_variable = { name = cp_startup_conditional_scan days = 1 }
		cp_samier_try_spawn = yes
		cp_tarek_try_spawn = yes
		cp_farid_try_spawn = yes
		cp_dawud_try_spawn = yes
		cp_rashid_try_spawn = yes
		cp_salma_try_spawn = yes
		cp_karim_try_spawn = yes
		cp_huda_try_spawn = yes
		cp_nabil_try_spawn = yes
		cp_mansur_try_spawn = yes

		if = {
			limit = { NOT = { has_variable = cp_active_roster_v0_cleaned } }
			cp_clear_all_active_people = yes
			set_variable = cp_active_roster_v0_cleaned
		}
		cp_refresh_active_roster = yes
		trigger_event = { id = cp_shared_startup.2 }
	}

}
