###############################################################################
# Common People -- Layla setup events
#
# Layla is not a V3 character; she is a person-object represented by country
# variables on c:EGY. This file owns her hidden setup path so Layla follows the
# same register-then-setup pattern as every other person in the roster.
#
# Root scope: country (EGY) when cp_layla_setup.1 fires.
###############################################################################

namespace = cp_layla_setup


# cp_layla_setup.1 -- hidden setup
#
# Initializes every country variable Layla needs. Her visible intro is selected
# by cp_shared_startup.2's first-contact roll or by later person-owned routing,
# just like the rest of the roster.
# Idempotent: cp_layla_alive guards re-entry across save/reload and across the
# on_game_started hook firing for the first time.
cp_layla_setup.1 = {
	type = country_event
	hidden = yes

	trigger = {
		c:EGY ?= this
		has_global_variable = cp_person_layla_alive
		NOT = { has_variable = cp_layla_alive }
	}

	immediate = {
		# --- Life guard ---
		# cp_layla_alive = 1 while she lives. Goes to 0 at her death event
		# (which deactivates the JE). Every sensor / dispatcher / pulse
		# gates on this.
		set_variable = { name = cp_layla_alive value = 1 }

		# --- Identity ---
		set_variable = { name = cp_layla_age value = 22 }

		# --- Profession as flag set ---
		# Exactly one cp_profession_* is set at a time. Transitions clear
		# the old one and set the new one. Flag-based because string
		# variables are brittle for trigger comparisons in V3.
		set_variable = cp_layla_profession_peasants

		# --- Home state as a state-scope marker ---
		# Her residence is data-driven: the region_state she lives in
		# carries var:cp_layla_lives_here. Any EGY state at any time
		# (Delta, Hejaz, Sudan, Syria under Muhammad Ali, etc.) can host
		# her without hardcoding a flag list. At game start she lives in
		# the Delta. Must run BEFORE cp_refresh_sol_from_pop, which reads
		# from the marker.
		cp_move_to_state = { state = STATE_LOWER_EGYPT }
		cp_set_person_home_place = { person = layla place = mit_al_nakhla }
		cp_set_person_workplace_profile = { person = layla profile = agrarian }

		# --- V3-mirroring canonical state (scales match V3 itself) ---
		# SoL 1..99. Seed from her cohort via the pop-group read, then
		# layer personal law/variance via the library.
		set_variable = { name = cp_layla_sol value = 5 }
		cp_layla_refresh_sol = yes
		# Year-over-year snapshot and delta. Updated in cp_on_yearly.
		set_variable = { name = cp_layla_sol_prev value = var:cp_layla_sol }
		set_variable = { name = cp_layla_sol_delta value = 0 }
		# Literacy 0..100 (percent).
		set_variable = { name = cp_layla_literacy value = 0 }
		# Radicalism / loyalism 0..100. Mutually exclusive; effects enforce.
		set_variable = { name = cp_layla_radical value = 0 }
		set_variable = { name = cp_layla_loyalist value = 0 }
		# Pop-tick working vars. Overwritten yearly by cp_pop_tick_*;
		# seeded here so lint can see them before first yearly pulse.
		set_variable = { name = cp_layla_cohort_literacy_target value = 0 }
		set_variable = { name = cp_layla_expected_sol value = 5 }
		set_variable = { name = cp_layla_sol_gap value = 0 }
		# Mortality pressures, refreshed yearly.
		set_variable = { name = cp_layla_mortality_pressure value = 0 }
		set_variable = { name = cp_layla_worker_risk value = 0 }
		# Welfare-ladder tier (0..4), snapshot via cp_refresh_welfare_tier.
		# Tracked so cp_on_law_enacted can detect up/down transitions.
		set_variable = { name = cp_layla_welfare_tier value = 0 }
		set_variable = { name = cp_layla_welfare_tier_prev value = 0 }
		cp_refresh_welfare_tier = yes
		# Tax burden 0..10. Recomputed yearly. Year-over-year snapshot
		# lets cp_on_yearly detect sharp spikes and fire the tax-spike event.
		set_variable = { name = cp_layla_tax_burden value = 0 }
		cp_refresh_tax_burden = yes
		set_variable = { name = cp_layla_tax_burden_prev value = var:cp_layla_tax_burden }

		# --- Inner state (0..10) ---
		set_variable = { name = cp_layla_hope value = 5 }
		set_variable = { name = cp_layla_exhaustion value = 3 }

		# --- Monthly mood roll (1..10) ---
		# Rerolled monthly in cp_on_monthly. Drives which mood bucket her
		# status line falls into: low 1..3, mid 4..6, high 7..10.
		set_variable = { name = cp_layla_mood_roll value = 5 }

		# --- Monthly variant roll (1..3) ---
		# Rerolled alongside mood. Picks which of three prose variants
		# within the chosen mood bucket is displayed this month. Each
		# (situation x mood x variant) triple resolves to one triggered_desc.
		set_variable = { name = cp_layla_status_variant value = 1 }

		# --- Personality weights (0..10) ---
		# Drift silently + via event effects. See documentation/characters/layla/README.md.
		set_variable = { name = cp_layla_w_land_reform value = 10 }
		set_variable = { name = cp_layla_w_war value = 7 }
		set_variable = { name = cp_layla_w_revolution value = 8 }
		set_variable = { name = cp_layla_w_religion value = 6 }
		set_variable = { name = cp_layla_w_womens_rights value = 5 }
		set_variable = { name = cp_layla_w_economy value = 5 }
		set_variable = { name = cp_layla_w_france value = 4 }
		set_variable = { name = cp_layla_w_education value = 3 }
		set_variable = { name = cp_layla_w_labor_laws value = 2 }
		set_variable = { name = cp_layla_w_diplomacy value = 1 }

		# --- Nation opinions (-100..+100) ---
		# Absence of a variable = total unawareness. She only has opinions
		# on nations she has heard of. New opinions get set when events
		# introduce them.
		set_variable = { name = cp_layla_opinion_fra value = 5 }
		set_variable = { name = cp_layla_opinion_ott value = 10 }
		set_variable = { name = cp_layla_opinion_gbr value = 0 }
		set_variable = { name = cp_layla_opinion_egy value = 10 }

		# --- Household (Ahmed is narrative; no V3 character for him either) ---
		set_variable = { name = cp_layla_ahmed_alive value = 1 }
		set_variable = { name = cp_layla_ahmed_age value = 26 }
		set_variable = cp_layla_ahmed_profession_peasant
		set_variable = { name = cp_layla_married value = 1 }
		set_variable = { name = cp_layla_children value = 0 }

		# --- Debug tick ---
		# Incremented by the JE's on_monthly_pulse. Visible in the status
		# line during Phase A verification; removed from UI once the pulse
		# is proven to fire.
		set_variable = { name = cp_debug_tick value = 0 }

		# The persistent journal UI is now shared by the roster in
		# cp_je_common_people. Layla keeps her own variables and events, but
		# she no longer owns the player's journal surface.
	}
}
