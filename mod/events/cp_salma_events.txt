###############################################################################
# Common People -- Salma Farag events
#
# Salma is Person 12: a Misri switchboard clerk whose night shifts, listening
# posture, and tired hands make telephones and electric service human-scale.
###############################################################################

namespace = cp_salma


# cp_salma.1 -- hidden setup
cp_salma.1 = {
	type = country_event
	hidden = yes

	trigger = {
		c:EGY ?= this
		has_global_variable = cp_person_salma_alive
		NOT = { has_variable = cp_salma_alive }
	}

	immediate = {
		set_variable = { name = cp_salma_alive value = 1 }
		set_variable = { name = cp_salma_age value = 19 }
		set_variable = { name = cp_salma_hope value = 6 }
		set_variable = { name = cp_salma_exhaustion value = 4 }
		set_variable = { name = cp_salma_sol value = 11 }
		set_variable = { name = cp_salma_literacy value = 18 }
		set_variable = { name = cp_salma_radical value = 0 }
		set_variable = { name = cp_salma_loyalist value = 0 }
		set_variable = { name = cp_salma_earache value = 2 }
		set_variable = cp_salma_profession_clerks
		set_variable = { name = cp_salma_w_wires value = 10 }
		set_variable = { name = cp_salma_w_electricity value = 8 }
		set_variable = { name = cp_salma_w_distance value = 8 }
		set_variable = { name = cp_salma_w_family value = 6 }
		set_variable = { name = cp_salma_w_work value = 6 }

		cp_set_person_home_state_from_spawn_marker_or_default = { person = salma country = EGY state = STATE_LOWER_EGYPT }
		cp_set_person_home_place = { person = salma place = ataba_cairo }
		cp_set_person_workplace_profile = { person = salma profile = electric }
		cp_salma_refresh_sol = yes

		if = {
			limit = {
				NOT = { has_variable = cp_startup_conditional_scan }
			}
			random_list = {
				55 = { cp_try_fire_world_response = { event = cp_salma.10 person = salma } }
				45 = { }
			}
		}
	}
}


# cp_salma.10 -- "The Switchboard"
cp_salma.10 = {
	type = country_event

	title = cp_salma.10.t
	desc = cp_salma.10.d
	flavor = cp_salma.10.f

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	event_image = {
		texture = "gfx/event_pictures/cp_salma_10_the_switchboard_v01.dds"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/MUSIC/Stingers/events/civil"

	duration = 3

	trigger = {
		has_variable = cp_salma_alive
		var:cp_salma_alive > 0
		NOT = { has_variable = cp_salma_seen_intro }
	}

	immediate = {
		cp_salma_mark_intro = yes
	}

	option = {
		name = cp_salma.10.a
		default_option = yes
	}
}
